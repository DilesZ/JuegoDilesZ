   171→  /* ---------- Spawns ---------- */

   173→  private scaling() {
   174→    const lv = this.player.level;
   175→    const eMul = this.endless ? 1.35 : 1;
   176→    return {
   177→      hp: (1 + 0.08 * (lv - 1)) * eMul,
   178→      dmg: (1 + 0.05 * (lv - 1)) * eMul,
   179→    };

   182→  private spawnEnemy(type: EnemyType, x: number, z: number, guardianOf = -1): Enemy {
   183→    const s = this.scaling();

   190→  private spawnGuardians() {
   191→    const compositions: EnemyType[][] = [
   192→      ['goblin', 'goblin', 'goblin', 'archer'],
   193→      ['goblin', 'goblin', 'archer', 'archer'],
   194→      ['orc', 'goblin', 'goblin', 'archer'],

   196→    compositions.forEach((comp, idx) => {

   200→        const r = rand(4.5, 7.5);
   201→        this.spawnEnemy(type, s.x + Math.cos(a) * r, s.z + Math.sin(a) * r, idx);
   202→      });
   203→    });

   206→  private spawnRoamer(initial = false) {
   207→    if (this.enemies.filter(e => e.alive && e.guardianOf === -1 && !e.isBoss).length >= ROAMER_TARGET + 4) return;
   208→    for (let tries = 0; tries < 20; tries++) {
   209→      const a = rand(0, Math.PI * 2);
   210→      const r = rand(24, WORLD.radius - 12);
   211→      const x = Math.cos(a) * r, z = Math.sin(a) * r;
   212→      const dPlayer = Math.hypot(x - this.player.pos.x, z - this.player.pos.z);
   213→      const dBonfire = Math.hypot(x - WORLD.bonfire.x, z - WORLD.bonfire.z);
   214→      const nearShrine = WORLD.shrines.some((s, i) => !this.world.shrines[i].cleansed && Math.hypot(x - s.x, z - s.z) < 16);
   215→      if (dPlayer > (initial ? 20 : 26) && dBonfire > 16 && !nearShrine) {
   216→        // tipos según nivel
   217→        const roll = Math.random();
   218→        let type: EnemyType = 'goblin';
   219→        if (roll > 0.82 - Math.min(0.15, this.player.level * 0.01)) type = 'orc';
   220→        else if (roll > 0.5) type = 'archer';
   221→        this.spawnEnemy(type, x, z);
   222→        return;
   223→      }
   224→    }
   225→  }

   256→    this.player.kills++;

   258→    const gold = randInt(cfg.gold[0], cfg.gold[1]);
   259→    this.player.gold += gold;
   260→    this.player.gainXp(cfg.xp, this.ctx);
   261→    // orbes de alma
   262→    for (let i = 0; i < 10; i++) {
   263→      this.particles.spawn({
   264→        x: e.pos.x + rand(-0.6, 0.6), y: e.pos.y + 0.6, z: e.pos.z + rand(-0.6, 0.6),
   265→        color: 0x6ae8ff, size: 0.26, life: 2.2, glow: 2.2, shrink: false,
   266→        seek: { getPos: () => this.player.pos, speed: 6 },
   267→      });
   268→    }
   269→    // drop de poción
   270→    if (Math.random() < cfg.potionChance) {
   271→      this.pickups.push(new Pickup(e.pos.clone(), 'potion'));
   272→      this.scene.add(this.pickups[this.pickups.length - 1].root);
   273→    }
   274→    // guardianes → santuario purificable
   275→    if (e.guardianOf >= 0) {
   276→      const alive = this.enemies.filter(o => o.guardianOf === e.guardianOf && o.alive && o !== e).length;
   277→      if (alive === 0) {
   278→        // todos los guardias muertos
   279→        void e.guardianOf;
   280→      }
   281→    }
   282→    if (e.isBoss) {
   283→      this.bossActive = false;
   284→      this.bossDefeated = true;
   285→      this.endless = true;
   286→      this.world.activateSigil();
   287→      this.world.sigilReady();
   288→      this.slowmoT = 2.2;
   289→      this.timeScale = 0.3;
   290→      this.victoryDelay = 2.0;
   291→      this.audio.setIntensity(0.2);
   292→      this.audio.cleanse();
   293→      this.ctx.shake(1);
   294→      this.particles.burst({
   295→        x: e.pos.x, y: e.pos.y + 2, z: e.pos.z,
   296→        count: 90, speed: 10, color: 0xffc84a, size: 0.5, life: 2, gravity: -1, drag: 0.8, glow: 2.6,
   297→      });
   298→    }
   299→  }

   301→  /* ---------- Interacción ---------- */
   302→
   303→  private updateInteract() {
   304→    this.interactTarget = null;
   305→    const p = this.player.pos;
   306→    const dB = Math.hypot(p.x - WORLD.bonfire.x, p.z - WORLD.bonfire.z);
   307→    if (dB < 3.4) {
   308→      this.interactTarget = { kind: 'bonfire' };
   309→      return;
   310→    }
   311→    for (const sh of this.world.shrines) {
   312→      if (sh.cleansed) continue;
   313→      const guardians = this.enemies.filter(e => e.guardianOf === sh.idx && e.alive);
   314→      const d = Math.hypot(p.x - sh.pos.x, p.z - sh.pos.z);
   315→      if (d < 5 && guardians.length === 0) {
   316→        this.interactTarget = { kind: 'shrine', idx: sh.idx };
   317→        return;
   318→      }
   319→    }
   320→    if (this.world.sigil && this.world.sigil.group.visible && this.bossDefeated) {
   321→      const A = WORLD.arena;
   322→      const d = Math.hypot(p.x - A.x, p.z - A.z);
   323→      if (d < 4.5) this.interactTarget = { kind: 'sigil' };
   324→    }
   325→  }

   327→  private doInteract() {
   328→    const t = this.interactTarget;
   329→    if (!t) return;
   330→    if (t.kind === 'bonfire') {
   331→      const p = this.player;
   332→      p.hp = p.maxHp; p.stamina = p.maxStamina; p.potions = p.maxPotions;
   333→      this.audio.bonfireRest();
   334→      this.particles.burst({
   335→        x: p.pos.x, y: p.pos.y + 0.6, z: p.pos.z,
   336→        count: 26, speed: 2.6, color: 0xffb347, size: 0.3, life: 1.1, gravity: -1.5, drag: 1.5, glow: 2,
   337→      });
   338→      this.addDamageNumber(new THREE.Vector3(p.pos.x, p.pos.y + 2.3, p.pos.z), 'REPOSO', '#ffb347');
   339→    } else if (t.kind === 'shrine' && t.idx !== undefined) {
   340→      const sh = this.world.shrines[t.idx];
   341→      this.world.cleanseShrine(t.idx);
   342→      this.shrinesCleansed++;
   343→      this.audio.cleanse();
   344→      this.player.gainXp(60, this.ctx);
   345→      this.player.gold += 40;
   346→      this.particles.burst({
   347→        x: sh.pos.x, y: sh.pos.y + 4, z: sh.pos.z,
   348→        count: 70, speed: 7, color: 0x37d8c8, size: 0.42, life: 1.8, gravity: -0.5, drag: 0.9, glow: 2.4,
   349→      });
   350→      this.ctx.shake(0.5);
   351→      if (this.shrinesCleansed >= 3 && !this.bossDefeated && !this.bossActive) {
   352→        this.awakenTimer = 2.5;
   353→      }
   354→      this.emitHud();
   355→    } else if (t.kind === 'sigil') {
   356→      this.world.sigilReady();
   357→      this.spawnBoss(1.35 ** this.bossKills());
   358→      this.addDamageNumber(new THREE.Vector3(this.player.pos.x, this.player.pos.y + 2.3, this.player.pos.z), 'EL JEFE DESPIERTA DE NUEVO', '#ff5a4e', true);
   359→    }
   360→  }

   366→  /* ---------- Lock-on ---------- */

   368→  private toggleLockOn() {
   369→    if (this.lockEnemy) { this.lockEnemy = null; this.player.lockTarget = null; return; }
   370→    let best: Enemy | null = null, bestD = 34;
   371→    for (const e of this.enemies) {
   372→      if (!e.alive || e.state === 'spawn') continue;
   373→      const d = e.pos.distanceTo(this.player.pos);
   374→      if (d < bestD) { bestD = d; best = e; }
   375→    }
   376→    this.lockEnemy = best;
   377→    this.player.lockTarget = best;
   378→  }

   380→  /* ---------- Bucle principal ---------- */

   382→  start() {
   383→    if (this.running) return;
   384→    this.running = true;
   385→    this.lastT = performance.now();
   386→    this.loop(this.lastT);
   387→  }
   388→
   389→  private loop = (t: number) => {
   390→    if (!this.running) return;
   391→    this.raf = requestAnimationFrame(this.loop);
   392→    let dt = Math.min(0.05, (t - this.lastT) / 1000);
   393→    this.lastT = t;
   394→    this.fps = this.fps * 0.95 + (1 / Math.max(0.0001, dt)) * 0.05;
   395→
   396→    if (this.phase === 'playing') {
   397→      // hit-stop y cámara lenta
   398→      if (this.hitStopT > 0) {
   399→        this.hitStopT -= dt;
   400→        this.updateWorld(dt * 0.06);
   401→      } else if (this.slowmoT > 0) {
   402→        this.slowmoT -= dt;
   403→        if (this.slowmoT <= 0) this.timeScale = 1;
   404→        this.updateWorld(dt * this.timeScale);
   405→      } else {
   406→        this.updateWorld(dt);
   407→      }
   408→    } else if (this.phase === 'dead' || this.phase === 'victory') {
   409→      // el mundo sigue con animaciones reducidas
   410→      this.updateWorld(dt * 0.25, true);
   411→    }
   412→
   413→    this.updateCamera(dt);
   414→    this.updateEffects(dt);
   415→    this.composer.render();
   416→
   417→    this.hudTimer -= dt;
   418→    if (this.hudTimer <= 0) { this.hudTimer = 0.08; this.emitHud(); this.drawMinimap(); }
   419→  };
   420→
   421→  private updateWorld(dt: number, frozen = false) {
   422→    this.elapsed += frozen ? 0 : dt;
   423→
   424→    // input → estado
   425→    const k = this.keys;
   426→    this.inputState.fwd = k.has('KeyW') || k.has('ArrowUp');
   427→    this.inputState.back = k.has('KeyS') || k.has('ArrowDown');
   428→    this.inputState.left = k.has('KeyA') || k.has('ArrowLeft');
   429→    this.inputState.right = k.has('KeyD') || k.has('ArrowRight');
   430→    this.inputState.sprint = k.has('ShiftLeft') || k.has('ShiftRight');
   431→
   432→    // jugador
   433→    this.ctx.input = frozen ? this.frozenInput : this.inputState;
   434→    this.ctx.camYaw = this.camYaw;
   435→    this.player.update(dt, this.ctx);
   436→    this.world.resolve(this.player.pos, this.player.radius);
   437→    this.worldClamp(this.player.pos, this.player.radius);
   438→
   439→    // resolver golpe pendiente del jugador
   440→    if (this.player.pendingStrike) {
   441→      this.resolvePlayerStrike(this.player.pendingStrike);
   442→      this.player.pendingStrike = null;
   443→    }
   444→
   445→    // lock-on: limpiar si muere o se aleja
   446→    if (this.lockEnemy && (!this.lockEnemy.alive || this.lockEnemy.pos.distanceTo(this.player.pos) > 38)) {
   447→      this.lockEnemy = null;
   448→      this.player.lockTarget = null;
   449→    }
   450→
   451→    // enemigos
   452→    for (const e of this.enemies) {
   453→      if (e.removable) continue;
   454→      e.update(dt, this.ctx);
   455→      if (e.alive) {
   456→        this.world.resolve(e.pos, e.radius);
   457→        this.worldClamp(e.pos, e.radius);
   458→      }
   459→    }
   460→    // separación entre enemigos
   461→    const live = this.enemies.filter(e => e.alive && e.state !== 'spawn');
   462→    for (let i = 0; i < live.length; i++) {
   463→      for (let j = i + 1; j < live.length; j++) {
   464→        const a = live[i], b = live[j];
   465→        const dx = b.pos.x - a.pos.x, dz = b.pos.z - a.pos.z;
   466→        const minD = a.radius + b.radius;
   467→        const d2 = dx * dx + dz * dz;
   468→        if (d2 < minD * minD && d2 > 1e-5) {
   469→          const d = Math.sqrt(d2);
   470→          const push = (minD - d) / d * 0.5;
   471→          a.pos.x -= dx * push; a.pos.z -= dz * push;
   472→          b.pos.x += dx * push; b.pos.z += dz * push;
   473→        }
   474→      }
   475→      // el jugador empuja y es empujado ligeramente
   476→      const p = this.player;
   477→      const dx = p.pos.x - live[i].pos.x, dz = p.pos.z - live[i].pos.z;
   478→      const minD = live[i].radius + p.radius;
   479→      const d2 = dx * dx + dz * dz;
   480→      if (d2 < minD * minD && d2 > 1e-5) {
   481→        const d = Math.sqrt(d2);
   482→        const push = (minD - d) / d * 0.6;
   483→        live[i].pos.x -= dx * push; live[i].pos.z -= dz * push;
   484→        p.pos.x += dx * push * 0.3; p.pos.z += dz * push * 0.3;
   485→      }
   486→    }
   487→    // limpiar cadáveres
   488→    if (this.enemies.some(e => e.removable)) {
   489→      for (const e of this.enemies) {
   490→        if (e.removable) { this.scene.remove(e.root); }
   491→      }
   492→      this.enemies = this.enemies.filter(e => !e.removable);
   493→    }
   494→
   495→    // proyectiles y drops
   496→    for (const pr of this.projectiles) pr.update(dt, this.ctx);
   497→    this.projectiles = this.projectiles.filter(pr => {
   498→      if (pr.dead) { this.scene.remove(pr.root); return false; }
   499→      return true;
   500→    });
   501→    for (const pk of this.pickups) {
   502→      pk.update(dt, this.ctx);
   503→      this.world.resolve(pk.pos, 0.3);
   504→    }
   505→    this.pickups = this.pickups.filter(pk => {
   506→      if (pk.dead) { this.scene.remove(pk.root); return false; }
   507→      return true;
   508→    });
   509→
   510→    // mundo visual
   511→    this.world.update(dt, this.camera, (x, y, z) => {
   512→      this.particles.spawn({
   513→        x, y, z, vx: rand(-0.3, 0.3), vy: rand(1, 2.2), vz: rand(-0.3, 0.3),
   514→        color: 0xff7a2a, size: rand(0.1, 0.2), life: rand(0.8, 1.6), glow: 2.2, drag: 0.5,
   515→      });
   516→    });
   517→
   518→    if (frozen) return;
   519→
   520→    // estela de espada
   521→    if (this.player.state === 'attack') {
   522→      const b = new THREE.Vector3(), ti = new THREE.Vector3();
   523→      this.player.swordPoints(b, ti);
   524→      this.trail.emit(b, ti);
   525→    }
   526→    this.trail.update(dt, this.player.state === 'attack');
   527→
   528→    // despawn/respawn de errantes
   529→    this.roamerTimer -= dt;
   530→    if (this.roamerTimer <= 0) {
   531→      this.roamerTimer = rand(3.5, 6);
   532→      const aliveRoamers = this.enemies.filter(e => e.alive && e.guardianOf === -1 && !e.isBoss).length;
   533→      if (aliveRoamers < ROAMER_TARGET) this.spawnRoamer();
   534→    }
   535→
   536→    // despertar del jefe
   537→    if (this.awakenTimer > 0) {
   538→      this.awakenTimer -= dt;
   539→      if (this.awakenTimer <= 0) this.spawnBoss();
   540→    }
   541→    // reaparición del jefe tras morir el jugador en plena lucha
   542→    if (this.bossRespawnT > 0) {
   543→      this.bossRespawnT -= dt;
   544→      if (this.bossRespawnT <= 0 && this.shrinesCleansed >= 3) this.spawnBoss();
   545→    }
   546→
   547→    // interacción
   548→    this.updateInteract();
   549→
   550→    // música: intensidad
   551→    if (this.elapsed % 1 < dt) {
   552→      let intensity = 0.12;
   553→      if (this.bossActive) intensity = 1;
   554→      else {
   555→        const near = this.enemies.some(e => e.alive && e.aggroed && e.pos.distanceTo(this.player.pos) < 30);
   556→        intensity = near ? 0.75 : 0.12;
   557→      }
   558→      this.audio.setIntensity(intensity);
   559→    }
   560→
   561→    // victoria
   562→    if (this.victoryDelay > 0) {
   563→      this.victoryDelay -= dt;
   564→      if (this.victoryDelay <= 0) {
   565→        this.setPhase('victory');
   566→      }
   567→    }
   568→  }

   577→  private resolvePlayerStrike(def: AttackDef) {
   578→    const p = this.player;
   579→    const heavy = def === PLAYER_HEAVY;
   580→    let hitAny = false;
   581→    for (const e of this.enemies) {
   582→      if (!e.alive || e.state === 'spawn') continue;
   583→      const to = new THREE.Vector3().subVectors(e.pos, p.pos).setY(0);
   584→      const d = to.length() - e.radius;
   585→      if (d > def.range) continue;
   586→      const ang = Math.atan2(to.x, to.z);
   587→      let diff = Math.abs(((ang - p.yaw + Math.PI * 3) % (Math.PI * 2)) - Math.PI);
   588→      if (diff <= def.arc / 2) {
   589→        const dmg = def.dmg * p.dmgMul;
   590→        e.takeDamage(dmg, p.pos, this.ctx, heavy);
   591→        hitAny = true;
   592→        this.particles.burst({
   593→          x: e.pos.x - to.x * 0.3, y: e.pos.y + e.rig.height * 0.55, z: e.pos.z - to.z * 0.3,
   594→          count: heavy ? 14 : 8, speed: 5, color: 0xffd9a0, size: 0.2, life: 0.35, drag: 2, glow: 2.4, gravity: 4,
   595→        });
   596→      }
   597→    }
   598→    if (hitAny) {
   599→      this.hitStopT = Math.max(this.hitStopT, heavy ? 0.11 : 0.07);
   600→      this.shakeAmt = Math.min(1.2, this.shakeAmt + (heavy ? 0.4 : 0.16));
   601→      if (heavy) this.audio.heavyHit();
   602→      else this.audio.hitMetal();
   603→    }
   604→  }

   607→    const r = Math.hypot(pos.x, pos.z);
   608→    const max = WORLD.radius - 2 - radius;

   610→      pos.x *= max / r;

   612→    }

   617→  private updateCamera(dt: number) {
   618→    // ratón → rotación
   619→    if (this.locked && this.phase === 'playing') {
   620→      this.camYaw -= this.mouseDX * 0.0023;
   621→      this.camPitch += this.mouseDY * 0.0018;
   622→      this.camPitch = clamp(this.camPitch, -0.25, 1.15);
   623→    }
   624→    this.mouseDX = 0; this.mouseDY = 0;
   625→
   626→    const target = this.player.pos;
   627→    const head = new THREE.Vector3(target.x, target.y + 1.55, target.z);
   628→
   629→    if (this.lockEnemy && this.lockEnemy.alive) {
   630→      // cámara de objetivo fijado
   631→      const desired = Math.atan2(target.x - this.lockEnemy.pos.x, target.z - this.lockEnemy.pos.z);
   632→      this.camYaw = dampAngle(this.camYaw, desired, 3.2, dt);
   633→      this.camPitch = damp(this.camPitch, 0.3, 3, dt);
   634→    }
   635→
   636→    const dist = this.camDist;
   637→    const cp = Math.cos(this.camPitch), sp = Math.sin(this.camPitch);
   638→    const off = new THREE.Vector3(Math.sin(this.camYaw) * cp, sp, Math.cos(this.camYaw) * cp);
   639→    const desiredPos = head.clone().addScaledVector(off, dist);
   640→    // hombro
   641→    const right = new THREE.Vector3(-Math.cos(this.camYaw), 0, Math.sin(this.camYaw));
   642→    desiredPos.addScaledVector(right, 0.55);
   643→    // colisión con terreno
   644→    const groundH = terrainHeight(desiredPos.x, desiredPos.z) + 0.5;
   645→    if (desiredPos.y < groundH) desiredPos.y = groundH;
   646→
   647→    this.camPos.lerp(desiredPos, 1 - Math.exp(-(this.lockEnemy ? 9 : 12) * dt));
   648→
   649→    // sacudida
   650→    const sh = this.shakeAmt;
   651→    this.camera.position.copy(this.camPos);
   652→    if (sh > 0.001) {
   653→      this.camera.position.x += rand(-1, 1) * sh * 0.18;
   654→      this.camera.position.y += rand(-1, 1) * sh * 0.18;
   655→      this.camera.position.z += rand(-1, 1) * sh * 0.18;
   656→      this.shakeAmt = Math.max(0, sh - dt * 2.4);
   657→    }
   658→    const lookAt = head.clone();
   659→    if (this.lockEnemy && this.lockEnemy.alive) {
   660→      lookAt.lerp(new THREE.Vector3(this.lockEnemy.pos.x, this.lockEnemy.pos.y + 1.2, this.lockEnemy.pos.z), 0.22);
   661→    }
   662→    this.camera.lookAt(lookAt);
   663→
   664→    // FOV dinámico al esprintar
   665→    const targetFov = this.baseFov + (this.player.sprinting ? 5 : 0);
   666→    if (Math.abs(this.camera.fov - targetFov) > 0.05) {
   667→      this.camera.fov = damp(this.camera.fov, targetFov, 6, dt);
   668→      this.camera.updateProjectionMatrix();
   669→    }
   670→  }

   672→  private updateEffects(dt: number) {
   673→    this.particles.update(dt, this.camera.position);
   674→    // números de daño
   675→    const w = window.innerWidth, h = window.innerHeight;
   676→    const v = new THREE.Vector3();
   677→    for (const d of this.dmgPool) {
   678→      if (!d.active) continue;
   679→      d.life -= dt;
   680→      if (d.life <= 0) { d.active = false; d.el.style.display = 'none'; continue; }
   681→      d.worldPos.y += d.vy * dt;
   682→      d.vy = Math.max(0.2, d.vy - dt * 1.4);
   683→      v.copy(d.worldPos).project(this.camera);
   684→      if (v.z > 1) { d.el.style.display = 'none'; continue; }
   685→      d.el.style.display = 'block';
   686→      d.el.style.left = `${(v.x * 0.5 + 0.5) * w}px`;
   687→      d.el.style.top = `${(-v.y * 0.5 + 0.5) * h}px`;
   688→      d.el.style.opacity = `${Math.min(1, d.life * 2.5)}`;
   689→    }
   690→    // viñeta de daño
   691→    this.hurtFlash = Math.max(0, this.hurtFlash - dt * 2.2);
   692→    const lowHp = this.player.hp / this.player.maxHp < 0.3 && this.player.alive;
   693→    const pulse = lowHp ? 0.22 + Math.sin(this.elapsed * 5) * 0.08 : 0;
   694→    this.refs.vignette.style.opacity = `${Math.min(1, this.hurtFlash * 0.9 + pulse)}`;
   695→  }
