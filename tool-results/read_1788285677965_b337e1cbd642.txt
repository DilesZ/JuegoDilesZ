    21→/* Grading final estilo anime: limpio, saturado, sin grano notable */
    22→const GradeShader = {
    23→  uniforms: {
    24→    tDiffuse: { value: null as THREE.Texture | null },
    25→    uTime: { value: 0 },
    26→    uVignette: { value: 0.30 },
    27→    uGrain: { value: 0.016 },
    28→    uCA: { value: 0.0006 },
    29→    uSat: { value: 1.18 },
    30→    uCon: { value: 1.03 },
    31→  },
    32→  vertexShader: `varying vec2 vUv; void main(){ vUv = uv; gl_Position = projectionMatrix * modelViewMatrix * vec4(position,1.0); }`,
    33→  fragmentShader: `
    34→    uniform sampler2D tDiffuse; uniform float uTime; uniform float uVignette;
    35→    uniform float uGrain; uniform float uCA; uniform float uSat; uniform float uCon;
    36→    varying vec2 vUv;
    37→    float hash(vec2 p){ return fract(sin(dot(p, vec2(12.9898,78.233))) * 43758.5453); }
    38→    void main(){
    39→      vec2 uv = vUv;
    40→      vec2 d = uv - 0.5;
    41→      float r2 = dot(d, d);
    42→      vec2 off = d * r2 * uCA * 8.0;
    43→      vec3 col;
    44→      col.r = texture2D(tDiffuse, uv + off).r;
    45→      col.g = texture2D(tDiffuse, uv).g;
    46→      col.b = texture2D(tDiffuse, uv - off).b;
    47→      float l = dot(col, vec3(0.299, 0.587, 0.114));
    48→      col = mix(vec3(l), col, uSat);
    49→      col = (col - 0.5) * uCon + 0.5;
    50→      col *= 1.0 - uVignette * smoothstep(0.12, 0.72, r2);
    51→      col += (hash(uv * vec2(1287.0, 731.0) + fract(uTime) * 43.7) - 0.5) * uGrain;
    52→      gl_FragColor = vec4(clamp(col, 0.0, 1.0), 1.0);
    53→    }`,
    54→};

   461→  private spawnGuardians() {
   462→    const compositions: EnemyType[][] = [

   464→      ['goblin', 'goblin', 'archer', 'archer'],

   471→        const r = rand(4.5, 7.5);
   472→        this.spawnEnemy(type, s.x + Math.cos(a) * r, s.z + Math.sin(a) * r, idx);
   473→      });
   474→    });

   477→  private spawnRoamer(initial = false) {
   478→    if (this.enemies.filter(e => e.alive && e.guardianOf === -1 && !e.isBoss).length >= ROAMER_TARGET + 4) return;
   479→    for (let tries = 0; tries < 20; tries++) {
   480→      const a = rand(0, Math.PI * 2);
   481→      const r = rand(24, WORLD.radius - 12);
   482→      const x = Math.cos(a) * r, z = Math.sin(a) * r;
   483→      const dPlayer = Math.hypot(x - this.player.pos.x, z - this.player.pos.z);
   484→      const dBonfire = Math.hypot(x - WORLD.bonfire.x, z - WORLD.bonfire.z);
   485→      const nearShrine = WORLD.shrines.some((s, i) => !this.world.shrines[i].cleansed && Math.hypot(x - s.x, z - s.z) < 16);
   486→      if (dPlayer > (initial ? 20 : 26) && dBonfire > 16 && !nearShrine) {
   487→        // tipos según nivel
   488→        const roll = Math.random();
   489→        let type: EnemyType = 'goblin';

   665→  start() {
   666→    if (this.running) return;
   667→    this.running = true;
   668→    this.lastT = performance.now();
   669→    this.loop(this.lastT);

   671→
   672→  private loop = (t: number) => {
   673→    if (!this.running) return;
   674→    this.raf = requestAnimationFrame(this.loop);
   675→    const dt = Math.min(0.05, (t - this.lastT) / 1000);
   676→    this.lastT = t;
   677→    this.fps = this.fps * 0.95 + (1 / Math.max(0.0001, dt)) * 0.05;
   678→
   679→    if (this.phase === 'menu') {
   680→      this.menuT += dt;
   681→      this.cycle.update(dt); // el mundo vive detrás del menú (con ciclo día/noche)
   682→      this.updateMenuScene(dt);
   683→      this.composer.render();
   684→      this.hudTimer -= dt;
   685→      if (this.hudTimer <= 0) { this.hudTimer = 0.3; this.emitHud(); }
   686→      return;
   687→    }
   688→
   689→    if (this.phase === 'playing') {
   690→      this.autoQuality(dt);
   691→      // ciclo día/noche (los enemigos son más rápidos de noche)
   692→      this.cycle.update(dt);
   693→      this.ctx.nightFactor = this.cycle.nightFactor;
   694→      if (this.noticeT > 0) { this.noticeT -= dt; if (this.noticeT <= 0) this.notice = ''; }
   695→      // pasos del jugador sobre el terreno
   696→      const stepDx = this.player.pos.x - this.lastStepPos.x;
   697→      const stepDz = this.player.pos.z - this.lastStepPos.z;
   698→      this.stepDist += Math.hypot(stepDx, stepDz);
   699→      this.lastStepPos.copy(this.player.pos);
   700→      if (this.stepDist > 2.2 && this.player.state !== 'roll' && this.player.state !== 'dead') {
   701→        this.stepDist = 0;
   702→        this.audio.footstep();
   703→      }
   704→      if (this.uiOpen) {
   705→        // inventario abierto: mundo congelado, escena viva
   706→        this.updateWorld(0.0001, true);
   707→      } else if (this.hitStopT > 0) {
   708→        // hit-stop y cámara lenta
   709→        this.hitStopT -= dt;
   710→        this.updateWorld(dt * 0.06);
   711→      } else if (this.slowmoT > 0) {
   712→        this.slowmoT -= dt;
   713→        if (this.slowmoT <= 0) this.timeScale = 1;
   714→        this.updateWorld(dt * this.timeScale);
   715→      } else {
   716→        this.updateWorld(dt);
   717→      }
   718→    } else if (this.phase === 'dead' || this.phase === 'victory') {
   719→      // el mundo sigue con animaciones reducidas
   720→      this.cycle.update(dt * 0.25);
   721→      this.updateWorld(dt * 0.25, true);
   722→    }
   723→
   724→    this.updateCamera(dt);
   725→    this.updateEffects(dt);
   726→    this.composer.render();
   727→
   728→    this.hudTimer -= dt;
   729→    if (this.hudTimer <= 0) { this.hudTimer = 0.08; this.emitHud(); this.drawMinimap(); }
   730→  };

   769→  private updateWorld(dt: number, frozen = false) {
   770→    this.elapsed += frozen ? 0 : dt;
   771→
   772→    // input → estado
   773→    const k = this.keys;
   774→    this.inputState.fwd = k.has('KeyW') || k.has('ArrowUp');
   775→    this.inputState.back = k.has('KeyS') || k.has('ArrowDown');
   776→    this.inputState.left = k.has('KeyA') || k.has('ArrowLeft');
   777→    this.inputState.right = k.has('KeyD') || k.has('ArrowRight');
   778→    this.inputState.sprint = k.has('ShiftLeft') || k.has('ShiftRight');
   779→
   780→    // jugador
   781→    this.ctx.input = frozen ? this.frozenInput : this.inputState;
   782→    this.ctx.camYaw = this.camYaw;
   783→    this.player.update(dt, this.ctx);
   784→    this.world.resolve(this.player.pos, this.player.radius);
   785→    this.worldClamp(this.player.pos, this.player.radius);
   786→
   787→    // resolver golpe pendiente del jugador
   788→    if (this.player.pendingStrike) {
   789→      this.resolvePlayerStrike(this.player.pendingStrike);
   790→      this.player.pendingStrike = null;
   791→    }
   792→
   793→    // lock-on: limpiar si muere o se aleja
   794→    if (this.lockEnemy && (!this.lockEnemy.alive || this.lockEnemy.pos.distanceTo(this.player.pos) > 38)) {
   795→      this.lockEnemy = null;
   796→      this.player.lockTarget = null;
   797→    }
   798→
   799→    // enemigos
   800→    for (const e of this.enemies) {
   801→      if (e.removable) continue;
   802→      e.update(dt, this.ctx);
   803→      if (e.alive) {
   804→        this.world.resolve(e.pos, e.radius);
   805→        this.worldClamp(e.pos, e.radius);
   806→      }
   807→    }
   808→    // separación entre enemigos
   809→    const live = this.enemies.filter(e => e.alive && e.state !== 'spawn');
   810→    for (let i = 0; i < live.length; i++) {
   811→      for (let j = i + 1; j < live.length; j++) {
   812→        const a = live[i], b = live[j];
   813→        const dx = b.pos.x - a.pos.x, dz = b.pos.z - a.pos.z;
   814→        const minD = a.radius + b.radius;
   815→        const d2 = dx * dx + dz * dz;
   816→        if (d2 < minD * minD && d2 > 1e-5) {
   817→          const d = Math.sqrt(d2);
   818→          const push = (minD - d) / d * 0.5;
   819→          a.pos.x -= dx * push; a.pos.z -= dz * push;
   820→          b.pos.x += dx * push; b.pos.z += dz * push;
   821→        }
   822→      }
   823→      // el jugador empuja y es empujado ligeramente
   824→      const p = this.player;
   825→      const dx = p.pos.x - live[i].pos.x, dz = p.pos.z - live[i].pos.z;
   826→      const minD = live[i].radius + p.radius;
   827→      const d2 = dx * dx + dz * dz;
   828→      if (d2 < minD * minD && d2 > 1e-5) {
   829→        const d = Math.sqrt(d2);
   830→        const push = (minD - d) / d * 0.6;
   831→        live[i].pos.x -= dx * push; live[i].pos.z -= dz * push;
   832→        p.pos.x += dx * push * 0.3; p.pos.z += dz * push * 0.3;
   833→      }
   834→    }
   835→    // limpiar cadáveres
   836→    if (this.enemies.some(e => e.removable)) {
   837→      for (const e of this.enemies) {
   838→        if (e.removable) { this.scene.remove(e.root); }
   839→      }
   840→      this.enemies = this.enemies.filter(e => !e.removable);
   841→    }
   842→
   843→    // proyectiles y drops
   844→    for (const pr of this.projectiles) pr.update(dt, this.ctx);
   845→    this.projectiles = this.projectiles.filter(pr => {
   846→      if (pr.dead) { this.scene.remove(pr.root); return false; }
   847→      return true;
   848→    });
   849→    for (const pk of this.pickups) {
   850→      pk.update(dt, this.ctx);
   851→      this.world.resolve(pk.pos, 0.3);
   852→    }
   853→    this.pickups = this.pickups.filter(pk => {
   854→      if (pk.dead) { this.scene.remove(pk.root); return false; }
   855→      return true;
   856→    });
   857→
   858→    // mundo visual
   859→    this.world.update(dt, this.camera, (x, y, z) => {
   860→      this.particles.spawn({
   861→        x, y, z, vx: rand(-0.3, 0.3), vy: rand(1, 2.2), vz: rand(-0.3, 0.3),
   862→        color: 0xff7a2a, size: rand(0.1, 0.2), life: rand(0.8, 1.6), glow: 2.2, drag: 0.5,
   863→      });
   864→    });
   865→
   866→    if (frozen) return;
   867→
   868→    // estela de espada
   869→    if (this.player.state === 'attack') {
   870→      const b = new THREE.Vector3(), ti = new THREE.Vector3();
   871→      this.player.swordPoints(b, ti);
   872→      this.trail.emit(b, ti);
   873→    }
   874→    this.trail.update(dt, this.player.state === 'attack');
   875→
   876→    // despawn/respawn de errantes
   877→    this.roamerTimer -= dt;
   878→    if (this.roamerTimer <= 0) {
   879→      this.roamerTimer = rand(3.5, 6);
   880→      const aliveRoamers = this.enemies.filter(e => e.alive && e.guardianOf === -1 && !e.isBoss).length;
   881→      if (aliveRoamers < ROAMER_TARGET) this.spawnRoamer();
   882→    }
   883→
   884→    // despertar del jefe
   885→    if (this.awakenTimer > 0) {
   886→      this.awakenTimer -= dt;
   887→      if (this.awakenTimer <= 0) this.spawnBoss();
   888→    }
   889→    // reaparición del jefe tras morir el jugador en plena lucha
   890→    if (this.bossRespawnT > 0) {
   891→      this.bossRespawnT -= dt;
   892→      if (this.bossRespawnT <= 0 && this.shrinesCleansed >= 3) this.spawnBoss();
   893→    }
   894→
   895→    // interacción
   896→    this.updateInteract();
   897→
   898→    // música: intensidad
   899→    if (this.elapsed % 1 < dt) {
   900→      let intensity = 0.12;
   901→      if (this.bossActive) intensity = 1;
   902→      else {
   903→        const near = this.enemies.some(e => e.alive && e.aggroed && e.pos.distanceTo(this.player.pos) < 30);
   904→        intensity = near ? 0.75 : 0.12;
   905→      }
   906→      this.audio.setIntensity(intensity);
   907→    }
   908→
   909→    // victoria
   910→    if (this.victoryDelay > 0) {
   911→      this.victoryDelay -= dt;
   912→      if (this.victoryDelay <= 0) {
   913→        this.setPhase('victory');
   914→      }
   915→    }
   916→  }

   925→  private resolvePlayerStrike(def: AttackDef) {
   926→    const p = this.player;
   927→    const heavy = def === PLAYER_HEAVY;
   928→    let hitAny = false;
   929→    for (const e of this.enemies) {
   930→      if (!e.alive || e.state === 'spawn') continue;
   931→      const to = new THREE.Vector3().subVectors(e.pos, p.pos).setY(0);
   932→      const d = to.length() - e.radius;
   933→      if (d > def.range) continue;
   934→      const ang = Math.atan2(to.x, to.z);
   935→      let diff = Math.abs(((ang - p.yaw + Math.PI * 3) % (Math.PI * 2)) - Math.PI);
   936→      if (diff <= def.arc / 2) {
   937→        const dmg = def.dmg * p.dmgMul;
   938→        e.takeDamage(dmg, p.pos, this.ctx, heavy);
   939→        hitAny = true;
   940→        this.particles.burst({
   941→          x: e.pos.x - to.x * 0.3, y: e.pos.y + e.rig.height * 0.55, z: e.pos.z - to.z * 0.3,
   942→          count: heavy ? 14 : 8, speed: 5, color: 0xffd9a0, size: 0.2, life: 0.35, drag: 2, glow: 2.4, gravity: 4,
   943→        });
   944→      }
   945→    }
   946→    if (hitAny) {
   947→      this.hitStopT = Math.max(this.hitStopT, heavy ? 0.11 : 0.07);
   948→      this.shakeAmt = Math.min(1.2, this.shakeAmt + (heavy ? 0.4 : 0.16));
   949→      if (heavy) this.audio.heavyHit();
   950→      else this.audio.hitMetal();
   951→    }
   952→  }

   954→  private worldClamp(pos: THREE.Vector3, radius: number) {
   955→    const r = Math.hypot(pos.x, pos.z);
   956→    const max = WORLD.radius - 2 - radius;
   957→    if (r > max) {
   958→      pos.x *= max / r;
   959→      pos.z *= max / r;
   960→    }

   965→  private updateCamera(dt: number) {
   966→    // ratón → rotación
   967→    if (this.locked && this.phase === 'playing' && !this.uiOpen) {
   968→      this.camYaw -= this.mouseDX * 0.0023;
   969→      this.camPitch += this.mouseDY * 0.0018;
   970→      this.camPitch = clamp(this.camPitch, -0.25, 1.15);
   971→    }
   972→    this.mouseDX = 0; this.mouseDY = 0;
   973→
   974→    const target = this.player.pos;
   975→    const head = new THREE.Vector3(target.x, target.y + 1.55, target.z);
   976→
   977→    if (this.lockEnemy && this.lockEnemy.alive) {
   978→      // cámara de objetivo fijado
   979→      const desired = Math.atan2(target.x - this.lockEnemy.pos.x, target.z - this.lockEnemy.pos.z);
   980→      this.camYaw = dampAngle(this.camYaw, desired, 3.2, dt);
   981→      this.camPitch = damp(this.camPitch, 0.3, 3, dt);
   982→    }
   983→
   984→    // zoom suave hacia la distancia objetivo (rueda del ratón)
   985→    this.camDist = damp(this.camDist, this.camDistTarget, 9, dt);
   986→    const dist = this.camDist;
   987→    const cp = Math.cos(this.camPitch), sp = Math.sin(this.camPitch);
   988→    const off = new THREE.Vector3(Math.sin(this.camYaw) * cp, sp, Math.cos(this.camYaw) * cp);
   989→    const desiredPos = head.clone().addScaledVector(off, dist);
   990→    // hombro
   991→    const right = new THREE.Vector3(-Math.cos(this.camYaw), 0, Math.sin(this.camYaw));
   992→    desiredPos.addScaledVector(right, 0.55);
   993→    // colisión con terreno
   994→    const groundH = terrainHeight(desiredPos.x, desiredPos.z) + 0.5;
   995→    if (desiredPos.y < groundH) desiredPos.y = groundH;
   996→
   997→    this.camPos.lerp(desiredPos, 1 - Math.exp(-(this.lockEnemy ? 9 : 12) * dt));
   998→
   999→    // sacudida
  1000→    const sh = this.shakeAmt;
  1001→    this.camera.position.copy(this.camPos);
  1002→    if (sh > 0.001) {
  1003→      this.camera.position.x += rand(-1, 1) * sh * 0.18;
  1004→      this.camera.position.y += rand(-1, 1) * sh * 0.18;
  1005→      this.camera.position.z += rand(-1, 1) * sh * 0.18;
  1006→      this.shakeAmt = Math.max(0, sh - dt * 2.4);
  1007→    }
  1008→    const lookAt = head.clone();
  1009→    if (this.lockEnemy && this.lockEnemy.alive) {
  1010→      lookAt.lerp(new THREE.Vector3(this.lockEnemy.pos.x, this.lockEnemy.pos.y + 1.2, this.lockEnemy.pos.z), 0.22);
  1011→    }
  1012→    this.camera.lookAt(lookAt);
  1013→
  1014→    // FOV dinámico al esprintar
  1015→    const targetFov = this.baseFov + (this.player.sprinting ? 5 : 0);
  1016→    if (Math.abs(this.camera.fov - targetFov) > 0.05) {
  1017→      this.camera.fov = damp(this.camera.fov, targetFov, 6, dt);
  1018→      this.camera.updateProjectionMatrix();
  1019→    }
  1020→  }

  1022→  private updateEffects(dt: number) {
  1023→    this.particles.update(dt, this.camera.position);
  1024→    // números de daño
  1025→    const w = window.innerWidth, h = window.innerHeight;
  1026→    const v = new THREE.Vector3();
  1027→    for (const d of this.dmgPool) {
  1028→      if (!d.active) continue;
  1029→      d.life -= dt;
  1030→      if (d.life <= 0) { d.active = false; d.el.style.display = 'none'; continue; }
  1031→      d.worldPos.y += d.vy * dt;
  1032→      d.vy = Math.max(0.2, d.vy - dt * 1.4);
  1033→      v.copy(d.worldPos).project(this.camera);
  1034→      if (v.z > 1) { d.el.style.display = 'none'; continue; }
  1035→      d.el.style.display = 'block';
  1036→      d.el.style.left = `${(v.x * 0.5 + 0.5) * w}px`;
  1037→      d.el.style.top = `${(-v.y * 0.5 + 0.5) * h}px`;
  1038→      d.el.style.opacity = `${Math.min(1, d.life * 2.5)}`;
  1039→    }
  1040→    // viñeta de daño + grano/tiempo del grading
  1041→    this.hurtFlash = Math.max(0, this.hurtFlash - dt * 2.2);
  1042→    const lowHp = this.player.hp / this.player.maxHp < 0.3 && this.player.alive;
  1043→    const pulse = lowHp ? 0.22 + Math.sin(this.elapsed * 5) * 0.08 : 0;
  1044→    this.refs.vignette.style.opacity = `${Math.min(1, this.hurtFlash * 0.9 + pulse)}`;
  1045→    if (this.grade) this.grade.uniforms.uTime.value = this.elapsed;
  1046→  }

  1234→  private onKeyDown = (e: KeyboardEvent) => {
  1235→    if (e.code === 'Tab' || e.code === 'Space') e.preventDefault();
  1236→    // con la mochila abierta solo se permite cerrarla
  1237→    if (this.uiOpen) {
  1238→      if (e.code === 'Escape' || e.code === 'KeyI' || e.code === 'KeyB') this.toggleInventory();
  1239→      return;
  1240→    }
  1241→    if (this.phase !== 'playing') return;
  1242→    this.keys.add(e.code);
  1243→    switch (e.code) {
  1244→      case 'Tab': this.toggleLockOn(); break;
  1245→      case 'Space': this.queued.roll = true; break;
  1246→      case 'KeyF': this.queued.potion = true; break;
  1247→      case 'KeyE': this.doInteract(); break;
  1248→      case 'KeyI': case 'KeyB': this.toggleInventory(); break;
  1249→    }
  1250→  };
