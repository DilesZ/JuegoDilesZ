    43→    43→const AURORA_FRAG = /* glsl */`
    44→    44→varying vec2 vUv;
    45→    45→uniform float uTime; uniform float uPhase; uniform vec3 uColA; uniform vec3 uColB;
    46→    46→void main(){
    47→    47→  float x = vUv.x * 6.28318;
    48→    48→  float band = sin(x*2.0 + uTime*0.11 + uPhase) * 0.5
    49→    49→             + sin(x*5.0 - uTime*0.17 + uPhase*2.1) * 0.3
    50→    50→             + sin(x*11.0 + uTime*0.29) * 0.2;
    51→    51→  float y = vUv.y;
    52→    52→  float curtain = smoothstep(0.02, 0.3, y) * (1.0 - smoothstep(0.45, 0.98, y));
    53→    53→  float rays = 0.5 + 0.5 * sin(x*26.0 + band*4.0 + uTime*0.4 + uPhase*3.0);
    54→    54→  float i = curtain * (0.35 + 0.65 * rays);
    55→    55→  vec3 col = mix(uColA, uColB, clamp(y * 1.5, 0.0, 1.0));
    56→    56→  gl_FragColor = vec4(col * i * 0.9, i);
    57→    57→}`;

   100→   100→  resolve(pos: THREE.Vector3, radius: number) {
   101→   101→    for (const c of this.colliders) {
   102→   102→      const dx = pos.x - c.x, dz = pos.z - c.z;
   103→   103→      const d2 = dx * dx + dz * dz;
   104→   104→      const minD = c.r + radius;
   105→   105→      if (d2 < minD * minD && d2 > 1e-6) {
   106→   106→        const d = Math.sqrt(d2);
   107→   107→        const push = (minD - d) / d;
   108→   108→        pos.x += dx * push;
   109→   109→        pos.z += dz * push;
   110→   110→      }
   111→   111→    }

   273→   273→  private buildTerrain() {
   274→   274→    const seg = 150;
   275→   275→    const geo = new THREE.PlaneGeometry(WORLD.size, WORLD.size, seg, seg);
   276→   276→    geo.rotateX(-Math.PI / 2);
   277→   277→    const pos = geo.getAttribute('position') as THREE.BufferAttribute;
   278→   278→    const colors = new Float32Array(pos.count * 3);
   279→   279→    const c = new THREE.Color();
   280→   280→    for (let i = 0; i < pos.count; i++) {
   281→   281→      const x = pos.getX(i), z = pos.getZ(i);
   282→   282→      const h = terrainHeight(x, z);
   283→   283→      pos.setY(i, h);
   284→   284→      // sombreado por pendiente + variación
   285→   285→      const n = fbm(x * 0.11, z * 0.11, 3) * 0.5 + 0.5;
   286→   286→      const slope = 1 - Math.min(1, Math.abs(fbm(x * 0.05 + 31, z * 0.05 - 12, 2)) * 1.4);
   287→   287→      const shade = (0.86 + n * 0.22) * lerp(0.82, 1, slope);
   288→   288→      c.setRGB(shade, shade, shade);
   289→   289→      colors[i * 3] = c.r; colors[i * 3 + 1] = c.g; colors[i * 3 + 2] = c.b;
   290→   290→    }
   291→   291→    geo.setAttribute('color', new THREE.BufferAttribute(colors, 3));
   292→   292→    geo.computeVertexNormals();
   293→   293→    const mat = new THREE.MeshStandardMaterial({
   294→   294→      map: terrainSplat(),
   295→   295→      normalMap: terrainDetailNormal(),
   296→   296→      normalScale: new THREE.Vector2(0.9, 0.9),
   297→   297→      vertexColors: true,
   298→   298→      roughness: 0.96, metalness: 0,
   299→   299→      envMapIntensity: 0.35,
   300→   300→    });
   301→   301→    const terrain = new THREE.Mesh(geo, mat);
   302→   302→    terrain.receiveShadow = true;
   303→   303→    terrain.name = 'terrain';
   304→   304→    this.scene.add(terrain);
   305→   305→  }

   539→   539→    for (const s of WORLD.shrines) if (Math.hypot(x - s.x, z - s.z) < margin + 3) return true;
   540→   540→    if (Math.hypot(x - WORLD.arena.x, z - WORLD.arena.z) < WORLD.arena.r + margin - 4) return true;
   541→   541→    return false;

   573→   573→  private buildFireflies() {
   574→   574→    const rng = mulberry32(555);
   575→   575→    const N = 130;
   576→   576→    const pos = new Float32Array(N * 3);
   577→   577→    const phase = new Float32Array(N);
   578→   578→    const speed = new Float32Array(N);
   579→   579→    const size = new Float32Array(N);
   580→   580→    let i = 0, guard = 0;
   581→   581→    while (i < N && guard++ < 3000) {
   582→   582→      const a = rng() * Math.PI * 2;
   583→   583→      const r = 8 + rng() * (WORLD.radius - 16);
   584→   584→      const x = Math.cos(a) * r, z = Math.sin(a) * r;
   585→   585→      const h = terrainHeight(x, z);
   586→   586→      if (h > 6) continue;
   587→   587→      pos[i * 3] = x; pos[i * 3 + 1] = h + 0.4 + rng() * 1.4; pos[i * 3 + 2] = z;
   588→   588→      phase[i] = rng() * Math.PI * 2;
   589→   589→      speed[i] = 0.5 + rng() * 0.9;
   590→   590→      size[i] = 3 + rng() * 4;
   591→   591→      i++;
   592→   592→    }
   593→   593→    const geo = new THREE.BufferGeometry();
   594→   594→    geo.setAttribute('position', new THREE.BufferAttribute(pos, 3));
   595→   595→    geo.setAttribute('aPhase', new THREE.BufferAttribute(phase, 1));
   596→   596→    geo.setAttribute('aSpeed', new THREE.BufferAttribute(speed, 1));
   597→   597→    geo.setAttribute('aSize', new THREE.BufferAttribute(size, 1));
   598→   598→    const mat = new THREE.ShaderMaterial({
   599→   599→      transparent: true, depthWrite: false, blending: THREE.AdditiveBlending,
   600→   600→      uniforms: { uTime: this.skyTime },
   601→   601→      vertexShader: `attribute float aPhase; attribute float aSpeed; attribute float aSize;
   602→   602→        uniform float uTime; varying float vA;
   603→   603→        void main(){
   604→   604→          vec3 p = position;
   605→   605→          p.x += sin(uTime * aSpeed + aPhase) * 1.3;
   606→   606→          p.y += sin(uTime * aSpeed * 0.7 + aPhase * 2.0) * 0.55;
   607→   607→          p.z += cos(uTime * aSpeed * 0.85 + aPhase) * 1.3;
   608→   608→          vec4 mv = modelViewMatrix * vec4(p, 1.0);
   609→   609→          gl_PointSize = clamp(aSize * (150.0 / max(0.1, -mv.z)), 1.0, 40.0);
   610→   610→          float nearFade = smoothstep(2.2, 6.0, -mv.z);
   611→   611→          vA = (pow(0.5 + 0.5 * sin(uTime * 2.1 + aPhase * 3.0), 2.2) * 0.9 + 0.06) * nearFade;
   612→   612→          gl_Position = projectionMatrix * mv; }`,
   613→   613→      fragmentShader: `varying float vA;
   614→   614→        void main(){ vec2 uv = gl_PointCoord - 0.5; float d = length(uv);
   615→   615→          float a = smoothstep(0.5, 0.06, d) * vA;
   616→   616→          if (a < 0.01) discard;
   617→   617→          gl_FragColor = vec4(vec3(0.82, 1.0, 0.55) * 1.6, a); }`,
   618→   618→    });
   619→   619→    const pts = new THREE.Points(geo, mat);
   620→   620→    pts.frustumCulled = false;
   621→   621→    pts.renderOrder = 6;
   622→   622→    this.scene.add(pts);
   623→   623→  }
