    29→interface WindEntry { mat: THREE.Material; amp: number }

    31→const flameMats: THREE.ShaderMaterial[] = [];
    32→const _time = { value: 0 };
    33→
    34→export function registerWind(mat: THREE.Material, amp: number, anchor: 'top' | 'bottom' = 'top') {
    35→  if (windMats.some(w => w.mat === mat)) return;
    36→  windMats.push({ mat, amp });
    37→  mat.onBeforeCompile = (shader) => {
    38→    shader.uniforms.uTime = _time;
    39→    shader.uniforms.uAmp = { value: amp };
    40→    shader.vertexShader = shader.vertexShader
    41→      .replace('#include <common>', '#include <common>\nuniform float uTime;\nuniform float uAmp;')
    42→      .replace('#include <begin_vertex>', `#include <begin_vertex>
    43→        {
    44→          float wx = 0.0, wz = 0.0;
    45→          #ifdef USE_INSTANCING
    46→            wx = instanceMatrix[3][0]; wz = instanceMatrix[3][2];
    47→          #endif
    48→          float swayF = pow(${anchor === 'top' ? 'max(uv.y, 0.0)' : 'max(1.0 - uv.y, 0.0)'}, 1.5);
    49→          float sway = sin(uTime * 1.6 + wx * 0.35 + wz * 0.5) + 0.45 * sin(uTime * 2.9 + wx * 1.4 + wz * 0.8);
    50→          transformed.x += sway * uAmp * swayF;
    51→          transformed.z += sway * uAmp * 0.6 * swayF;
    52→        }`);
    53→  };
    54→  mat.customProgramCacheKey = () => `wind_${amp}_${anchor}`;
    55→}

    62→/** Material de llama procedural (plano con forma de fuego animado) */
    63→export function makeFlameMaterial(core: number, outer: number, scaleY = 1): THREE.ShaderMaterial {
    64→  const mat = new THREE.ShaderMaterial({
    65→    transparent: true, depthWrite: false, blending: THREE.AdditiveBlending, side: THREE.DoubleSide,
    66→    uniforms: {
    67→      uTime: { value: 0 },
    68→      uCore: { value: new THREE.Color(core) },
    69→      uOuter: { value: new THREE.Color(outer) },
    70→      uScaleY: { value: scaleY },
    71→    },
    72→    vertexShader: `varying vec2 vUv;
    73→      void main(){ vUv = uv; gl_Position = projectionMatrix * modelViewMatrix * vec4(position,1.0); }`,
    74→    fragmentShader: `varying vec2 vUv; uniform float uTime; uniform vec3 uCore; uniform vec3 uOuter; uniform float uScaleY;
    75→      float hash(vec2 p){ return fract(sin(dot(p, vec2(127.1,311.7))) * 43758.5453); }
    76→      float noise(vec2 p){ vec2 i = floor(p), f = fract(p); f = f*f*(3.0-2.0*f);
    77→        return mix(mix(hash(i), hash(i+vec2(1,0)), f.x), mix(hash(i+vec2(0,1)), hash(i+vec2(1,1)), f.x), f.y); }
    78→      void main(){
    79→        vec2 uv = vUv; uv.y *= uScaleY;
    80→        float t = uTime;
    81→        // ondulación horizontal creciente con la altura
    82→        float wob = (noise(vec2(uv.y*3.0 - t*2.4, t*0.8)) - 0.5) * 0.34 * uv.y;
    83→        float x = uv.x - 0.5 + wob;
    84→        // silueta de llama: ancho decreciente hacia arriba
    85→        float width = mix(0.42, 0.02, pow(uv.y, 0.72));
    86→        float body = 1.0 - smoothstep(width*0.15, width, abs(x));
    87→        // flicker vertical
    88→        float flick = 0.82 + 0.18 * noise(vec2(t*3.1, uv.y*2.0));
    89→        body *= flick * smoothstep(0.0, 0.10, uv.y) * (1.0 - smoothstep(0.72, 1.0, uv.y + abs(x)*0.8));
    90→        if (body <= 0.004) discard;
    91→        float core = smoothstep(0.12, 0.75, body) * (1.0 - uv.y*0.55);
    92→        vec3 col = mix(uOuter, uCore, clamp(core, 0.0, 1.0));
    93→        col *= 1.35 + 0.65 * body;
    94→        gl_FragColor = vec4(col, body);
    95→      }`,
    96→  });
    97→  flameMats.push(mat);
    98→  return mat;
    99→}

   102→export function flameGeometry(w: number, h: number): THREE.BufferGeometry {
   103→  const p1 = new THREE.PlaneGeometry(w, h);
   104→  p1.translate(0, h / 2, 0);

   169→/** Clona una geometría desplazando cada vértice a lo largo de su normal */
   170→function expandGeometry(geo: THREE.BufferGeometry, t: number): THREE.BufferGeometry {
   171→  const g = geo.clone();
   172→  const pos = g.getAttribute('position') as THREE.BufferAttribute;
   173→  const nor = g.getAttribute('normal') as THREE.BufferAttribute | null;
   174→  for (let i = 0; i < pos.count; i++) {
   175→    const nx = nor ? nor.getX(i) : 0, ny = nor ? nor.getY(i) : 1, nz = nor ? nor.getZ(i) : 0;
   176→    pos.setXYZ(i, pos.getX(i) + nx * t, pos.getY(i) + ny * t, pos.getZ(i) + nz * t);
   177→  }
   178→  g.computeBoundingSphere();
   179→  return g;
   180→}

   182→/**
   183→ * Añade contornos de tinta estilo anime a todos los meshes de un grupo.
   184→ * Omite materiales transparentes, aditivos, basic (ojos/glow) y emisivos.
   185→ */
   186→export function addOutlines(root: THREE.Object3D, thickness = 0.025) {
   187→  const targets: THREE.Mesh[] = [];
   188→  root.traverse(o => {
   189→    if (!(o instanceof THREE.Mesh)) return;
   190→    const m = o.material as THREE.Material & { transparent?: boolean; emissiveIntensity?: number };
   191→    if (!m || Array.isArray(m)) return;
   192→    if ((m as THREE.MeshBasicMaterial).isMeshBasicMaterial) return;
   193→    if (m.transparent) return;
   194→    if (typeof m.emissiveIntensity === 'number' && m.emissiveIntensity > 0.01) return;
   195→    if (o.userData.noOutline) return;
   196→    targets.push(o);
   197→  });
   198→  for (const mesh of targets) {
   199→    const out = new THREE.Mesh(expandGeometry(mesh.geometry, thickness), outlineMat());
   200→    out.castShadow = false;
   201→    out.receiveShadow = false;
   202→    out.renderOrder = mesh.renderOrder;
   203→    mesh.add(out); // hereda transformación (sigue animaciones)
   204→  }
   205→}

   207→/** Añade rim light (fresnel) a un material toon/estándar */
   208→export function addRim(mat: THREE.MeshStandardMaterial | THREE.MeshToonMaterial, color: number, strength = 0.35, power = 3.2) {
   209→  const c = new THREE.Color(color);
   210→  mat.onBeforeCompile = (shader) => {
   211→    shader.uniforms.uRimColor = { value: c };
   212→    shader.uniforms.uRimStrength = { value: strength };
   213→    shader.uniforms.uRimPower = { value: power };
   214→    shader.fragmentShader = shader.fragmentShader
   215→      .replace('#include <common>', `#include <common>
   216→        uniform vec3 uRimColor; uniform float uRimStrength; uniform float uRimPower;`)
   217→      .replace('#include <emissivemap_fragment>', `#include <emissivemap_fragment>
   218→        {
   219→          float rimF = pow(1.0 - saturate(dot(normalize(vViewPosition), normal)), uRimPower);
   220→          totalEmissiveRadiance += uRimColor * rimF * uRimStrength;
   221→        }`);
   222→  };
   223→  mat.customProgramCacheKey = () => `rim_${color}_${strength}_${power}`;
   224→  return mat;
   225→}

   236→/** Hoja de sección romboidal (más creíble que una caja) */
   237→function bladeGeo(len: number, wBase: number, tipFrac = 0.22): THREE.BufferGeometry {
   238→  const g = new THREE.CylinderGeometry(wBase * 0.22, wBase, len * (1 - tipFrac), 4, 1);
   239→  g.translate(0, len * (1 - tipFrac) / 2, 0);
   240→  const tip = new THREE.ConeGeometry(wBase * 0.22, len * tipFrac, 4);
   241→  tip.rotateY(Math.PI / 4);
   242→  tip.translate(0, len * (1 - tipFrac) + (len * tipFrac) / 2, 0);
   243→  const merged = mergeGeometries([g.toNonIndexed(), tip.toNonIndexed()])!;
   244→  merged.rotateY(Math.PI / 4);
   245→  merged.scale(1, 1, 0.42);
   246→  merged.computeVertexNormals();
   247→  return merged;
   248→}

   250→export function buildSword(scale = 1): THREE.Group {
   251→  const g = new THREE.Group();
   252→  const steel = forgedMat(0xcdd6e2);
   253→  const gold = forgedMat(0xc7a24a);
   254→  const blade = mesh(bladeGeo(0.85 * scale, 0.05 * scale), steel, 0, 0.1 * scale, 0);
   255→  // ranura luminosa (runa encendida)
   256→  const rune = mesh(new THREE.BoxGeometry(0.012 * scale, 0.5 * scale, 0.024 * scale),
   257→    emisMat(0x54e0ff, 1.5), 0, 0.5 * scale, 0, false);
   258→  const guard = mesh(new THREE.TorusGeometry(0.085 * scale, 0.02 * scale, 6, 12, Math.PI), gold, 0, 0.1 * scale, 0);
   259→  guard.rotation.z = Math.PI;
   260→  const grip = mesh(new THREE.CylinderGeometry(0.026 * scale, 0.03 * scale, 0.2 * scale, 8), woodMat(), 0, -0.02 * scale, 0);
   261→  const pommel = mesh(new THREE.OctahedronGeometry(0.045 * scale, 0), gold, 0, -0.14 * scale, 0);
   262→  g.add(blade, rune, guard, grip, pommel);
   263→  return g;
   264→}

   266→export function buildGreatsword(scale = 1): THREE.Group {
   267→  const g = new THREE.Group();
   268→  const dark = forgedMat(0x4a4a58);
   269→  const blade = mesh(bladeGeo(1.62 * scale, 0.15 * scale, 0.16), dark, 0, 0.12 * scale, 0);
   270→  // grietas incandescentes
   271→  const crackMat = emisMat(0xff3a1a, 1.8);
   272→  for (let i = 0; i < 3; i++) {
   273→    const crack = mesh(new THREE.BoxGeometry(0.016 * scale, 0.3 * scale - i * 0.04 * scale, 0.05 * scale),
   274→      crackMat, 0, (0.45 + i * 0.35) * scale, 0, false);
   275→    crack.rotation.z = (i - 1) * 0.18;
   276→    g.add(crack);
   277→  }
   278→  // púas del lomo
   279→  for (let i = 0; i < 3; i++) {
   280→    const spike = mesh(new THREE.ConeGeometry(0.03 * scale, 0.12 * scale, 4), dark,
   281→      0.07 * scale, (0.35 + i * 0.4) * scale, 0);
   282→    spike.rotation.z = -1.9;
   283→    g.add(spike);
   284→  }
   285→  const guard = mesh(new THREE.BoxGeometry(0.44 * scale, 0.07 * scale, 0.1 * scale), forgedMat(0x2a2a34), 0, 0.1 * scale, 0);
   286→  const grip = mesh(new THREE.CylinderGeometry(0.04 * scale, 0.05 * scale, 0.34 * scale, 8), woodMat(), 0, -0.1 * scale, 0);
   287→  const skull = mesh(new THREE.IcosahedronGeometry(0.075 * scale, 0), emisMat(0xff2a1a, 1.6), 0, -0.3 * scale, 0);
   288→  g.add(blade, guard, grip, skull);
   289→  return g;
   290→}

   397→export function buildHumanoid(o: HumanoidOpts): HumanoidRig {
   398→  const s = o.scale ?? 1;
   399→  const root = new THREE.Group();
   400→
   401→  const mkBody = (color: number) => {
   402→    const m = toonMat(color, { gradientMap: toonRamp4() });
   403→    if (o.rim) addRim(m, o.rim.color, o.rim.strength);
   404→    return m;
   405→  };
   406→  const skin = mkBody(o.skin);
   407→  const torsoMat = mkBody(o.torso);
   408→  const legMat = mkBody(o.legs);
   409→  const armMat = mkBody(o.arms ?? o.skin);
   410→
   411→  const legLen = 0.52, torsoH = 0.55, headR = o.bigHead ? 0.21 : 0.155;
   412→
   413→  const body = new THREE.Group();
   414→  body.position.y = legLen;
   415→  root.add(body);
   416→
   417→  const pelvis = mesh(new THREE.BoxGeometry(0.3, 0.16, 0.2), legMat, 0, 0.04, 0);
   418→  body.add(pelvis);
   419→
   420→  const torso = new THREE.Group();
   421→  body.add(torso);
   422→  const chest = mesh(new THREE.BoxGeometry(0.34, torsoH, 0.22), torsoMat, 0, torsoH / 2, 0);
   423→  torso.add(chest);
   424→  if (o.chest !== null && o.chest !== undefined) {
   425→    torso.add(mesh(new THREE.BoxGeometry(0.37, torsoH * 0.55, 0.25), forgedMat(o.chest), 0, torsoH * 0.62, 0));
   426→  }
   427→  if (o.hunched) torso.rotation.x = 0.35;
   428→
   429→  // Capa
   430→  if (o.cape !== null && o.cape !== undefined) {
   431→    const capeGroup = new THREE.Group();
   432→    capeGroup.position.set(0, torsoH * 0.9, -0.15);
   433→    capeGroup.rotation.x = 0.14;
   434→    capeGroup.rotation.y = Math.PI;
   435→    const geo = new THREE.PlaneGeometry(0.5, 0.68, 4, 6);
   436→    geo.translate(0, -0.34, 0);
   437→    if (o.capeTattered) {
   438→      const pos = geo.getAttribute('position') as THREE.BufferAttribute;
   439→      for (let i = 0; i < pos.count; i++) {
   440→        if (pos.getY(i) < -0.3) {
   441→          pos.setY(i, pos.getY(i) - Math.random() * 0.1);
   442→          pos.setX(i, pos.getX(i) + (Math.random() - 0.5) * 0.06);
   443→        }
   444→      }
   445→      geo.computeVertexNormals();
   446→    }
   447→    const capeMat = stdMat(o.cape, { side: THREE.DoubleSide });

   612→        if (mm && mm.emissive && mm.emissiveIntensity > 0) weaponMat = mm;
   613→      }
   614→    });

   622→    sh.rotation.x = Math.PI / 2;
   623→    sh.rotation.y = -0.25;
   624→    aL.hand.add(sh);
   625→  }
   626→
   627→  root.scale.setScalar(s);
   628→  if (o.outline) addOutlines(root, o.outline);
   629→  return {
   630→    root, body, torso, head,
   631→    armL: aL.arm, armR: aR.arm, legL, legR,
   632→    handR: aR.hand, handL: aL.hand, weapon, weaponMat: weaponMat ?? null,
   633→    height: (legLen + torsoH + headR * 2.4) * s,
   634→  };
   635→}

   823→/* ---------- Hierba: mata de 3 hojas curvadas (textura alpha) ---------- */
   824→
   825→export function grassGeometry(): THREE.BufferGeometry {
   826→  const blades: THREE.BufferGeometry[] = [];
   827→  const H = 0.6;
   828→  for (let i = 0; i < 3; i++) {
   829→    const p = new THREE.PlaneGeometry(0.2, H, 1, 3);
   830→    p.translate(0, H / 2, 0);
   831→    // curvatura de la hoja
   832→    const pos = p.getAttribute('position') as THREE.BufferAttribute;
   833→    for (let v = 0; v < pos.count; v++) {
   834→      const y = pos.getY(v);
   835→      const k = Math.pow(y / H, 2);
   836→      pos.setZ(v, k * 0.16);
   837→      pos.setX(v, pos.getX(v) * (1 - k * 0.35));
   838→    }
   839→    p.rotateY((i / 3) * Math.PI + 0.3);
   840→    p.rotateX(((i * 37) % 10 - 5) * 0.02);
   841→    blades.push(p);
   842→  }
   843→  const merged = mergeGeometries(blades)!;
   844→  merged.computeVertexNormals();
   845→  return merged;
   846→}
