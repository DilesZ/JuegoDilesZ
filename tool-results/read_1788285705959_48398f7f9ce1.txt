   145→/** Líneas de estadísticas de un objeto, tipo "+26% Daño" */
   146→function ItemStatLines({ item }: { item: ItemView }) {
   147→  const s = item.stats;
   148→  const rows: string[] = [];
   149→  if (s.dmg) rows.push(`+${Math.round(s.dmg * 100)}% Daño`);
   150→  if (s.hp) rows.push(`+${s.hp} Vida`);
   151→  if (s.def) rows.push(`+${s.def} Defensa`);
   152→  if (s.speed) rows.push(`+${Math.round(s.speed * 100)}% Velocidad`);
   153→  if (s.stam) rows.push(`+${Math.round(s.stam * 100)}% Aguante`);
   154→  if (s.crit) rows.push(`+${Math.round(s.crit * 100)}% Crítico`);
   155→  if (!rows.length && item.useText) rows.push(item.useText);
   156→  if (!rows.length) return null;
   157→  return (
   158→    <ul className="space-y-0.5">
   159→      {rows.map(r => (
   160→        <li key={r} className="text-[12px] text-emerald-300/90 font-semibold">{r}</li>
   161→      ))}
   162→    </ul>
   163→  );
   164→}

   166→/** Hueco de equipo o de mochila */
   167→function ItemSlot({
   168→  item, label, onClick, onHover, size = 'w-16 h-16', iconSize = 'text-2xl',
   169→}: {
   170→  item: ItemView | null; label?: string; onClick?: () => void;
   171→  onHover?: (it: ItemView | null) => void; size?: string; iconSize?: string;
   172→}) {
   173→  const rc = item ? RARITY_CSS[item.rarity] : null;
   174→  return (
   175→    <button
   176→      onClick={onClick}
   177→      onMouseEnter={() => onHover?.(item)}
   178→      onMouseLeave={() => onHover?.(null)}
   179→      disabled={!onClick}
   180→      className={`relative ${size} shrink-0 rounded-md border-2 bg-stone-950/85 transition-all duration-150 flex items-center justify-center
   181→        ${item && onClick ? 'cursor-pointer hover:scale-[1.06] hover:z-10' : ''} ${!item && onClick ? 'cursor-default' : ''}`}
   182→      style={rc ? { borderColor: rc, boxShadow: `0 0 10px ${rc}44, inset 0 0 8px ${rc}22` } : { borderColor: 'rgba(120,113,108,0.35)' }}
   183→      title={item ? item.name : (label ?? '')}
   184→    >
   185→      {item ? (
   186→        <span className={`${iconSize} leading-none drop-shadow-[0_2px_3px_rgba(0,0,0,0.8)]`}>{item.icon}</span>
   187→      ) : (
   188→        label && <span className="text-[9px] uppercase tracking-widest text-stone-600 px-1 text-center leading-tight">{label}</span>
   189→      )}
   190→      {item && item.count > 1 && (
   191→        <span className="absolute bottom-0.5 right-1 text-[10px] font-bold text-amber-100 drop-shadow-[0_1px_2px_rgba(0,0,0,1)]">×{item.count}</span>
   192→      )}
   193→      {item && rc && (item.rarity === 'epico' || item.rarity === 'legendario') && (
   194→        <span className="absolute -top-1 -right-1 w-2 h-2 rotate-45" style={{ background: rc, boxShadow: `0 0 6px ${rc}` }} />
   195→      )}
   196→    </button>

   381→  };

   383→  const hpFrac = hud.hp / Math.max(1, hud.maxHp);
   384→
   385→  return (

   407→      {/* ============ VELO DE CARGA ============ */}
