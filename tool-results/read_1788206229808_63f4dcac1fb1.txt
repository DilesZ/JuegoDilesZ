    19→/* Grading final: viñeta, grano fílmico, aberración cromática y color */
    20→const GradeShader = {
    21→  uniforms: {
    22→    tDiffuse: { value: null as THREE.Texture | null },
    23→    uTime: { value: 0 },
    24→    uVignette: { value: 0.44 },
    25→    uGrain: { value: 0.038 },
    26→    uCA: { value: 0.0015 },
    27→    uSat: { value: 1.08 },
    28→    uCon: { value: 1.04 },
    29→  },
    30→  vertexShader: `varying vec2 vUv; void main(){ vUv = uv; gl_Position = projectionMatrix * modelViewMatrix * vec4(position,1.0); }`,
    31→  fragmentShader: `
    32→    uniform sampler2D tDiffuse; uniform float uTime; uniform float uVignette;
    33→    uniform float uGrain; uniform float uCA; uniform float uSat; uniform float uCon;
    34→    varying vec2 vUv;
    35→    float hash(vec2 p){ return fract(sin(dot(p, vec2(12.9898,78.233))) * 43758.5453); }
    36→    void main(){
    37→      vec2 uv = vUv;
    38→      vec2 d = uv - 0.5;
    39→      float r2 = dot(d, d);
    40→      vec2 off = d * r2 * uCA * 8.0;
    41→      vec3 col;
    42→      col.r = texture2D(tDiffuse, uv + off).r;
    43→      col.g = texture2D(tDiffuse, uv).g;
    44→      col.b = texture2D(tDiffuse, uv - off).b;
    45→      float l = dot(col, vec3(0.299, 0.587, 0.114));
    46→      col = mix(vec3(l), col, uSat);
    47→      col = (col - 0.5) * uCon + 0.5;
    48→      col *= 1.0 - uVignette * smoothstep(0.12, 0.72, r2);
    49→      col += (hash(uv * vec2(1287.0, 731.0) + fract(uTime) * 43.7) - 0.5) * uGrain;
    50→      gl_FragColor = vec4(clamp(col, 0.0, 1.0), 1.0);
    51→    }`,
    52→};

   312→  private spawnRoamer(initial = false) {
   313→    if (this.enemies.filter(e => e.alive && e.guardianOf === -1 && !e.isBoss).length >= ROAMER_TARGET + 4) return;
   314→    for (let tries = 0; tries < 20; tries++) {
   315→      const a = rand(0, Math.PI * 2);
   316→      const r = rand(24, WORLD.radius - 12);
   317→      const x = Math.cos(a) * r, z = Math.sin(a) * r;
   318→      const dPlayer = Math.hypot(x - this.player.pos.x, z - this.player.pos.z);
   319→      const dBonfire = Math.hypot(x - WORLD.bonfire.x, z - WORLD.bonfire.z);
   320→      const nearShrine = WORLD.shrines.some((s, i) => !this.world.shrines[i].cleansed && Math.hypot(x - s.x, z - s.z) < 16);
   321→      if (dPlayer > (initial ? 20 : 26) && dBonfire > 16 && !nearShrine) {
   322→        // tipos según nivel
   323→        const roll = Math.random();
   324→        let type: EnemyType = 'goblin';
   325→        if (roll > 0.82 - Math.min(0.15, this.player.level * 0.01)) type = 'orc';
   326→        else if (roll > 0.5) type = 'archer';
   327→        this.spawnEnemy(type, x, z);
   328→        return;
   329→      }
   330→    }
   331→  }

   361→  private handleEnemyDied(e: Enemy) {
   362→    this.player.kills++;
   363→    const cfg = ENEMY_CFG[e.type];
   364→    const gold = randInt(cfg.gold[0], cfg.gold[1]);
   365→    this.player.gold += gold;
   366→    this.player.gainXp(cfg.xp, this.ctx);
   367→    // orbes de alma
   368→    for (let i = 0; i < 10; i++) {
   369→      this.particles.spawn({
   370→        x: e.pos.x + rand(-0.6, 0.6), y: e.pos.y + 0.6, z: e.pos.z + rand(-0.6, 0.6),
   371→        color: 0x6ae8ff, size: 0.26, life: 2.2, glow: 2.2, shrink: false,
   372→        seek: { getPos: () => this.player.pos, speed: 6 },
   373→      });
   374→    }
   375→    // drop de poción
   376→    if (Math.random() < cfg.potionChance) {
   377→      this.pickups.push(new Pickup(e.pos.clone(), 'potion'));
   378→      this.scene.add(this.pickups[this.pickups.length - 1].root);
   379→    }
   380→    // guardianes → santuario purificable
   381→    if (e.guardianOf >= 0) {
   382→      const alive = this.enemies.filter(o => o.guardianOf === e.guardianOf && o.alive && o !== e).length;
   383→      if (alive === 0) {
   384→        // todos los guardias muertos
   385→        void e.guardianOf;
   386→      }
   387→    }
   388→    if (e.isBoss) {
   389→      this.bossActive = false;
   390→      this.bossDefeated = true;
   391→      this.endless = true;
   392→      this.world.activateSigil();
   393→      this.world.sigilReady();
   394→      this.slowmoT = 2.2;
   395→      this.timeScale = 0.3;
   396→      this.victoryDelay = 2.0;
   397→      this.audio.setIntensity(0.2);
   398→      this.audio.cleanse();
   399→      this.ctx.shake(1);
   400→      this.particles.burst({
   401→        x: e.pos.x, y: e.pos.y + 2, z: e.pos.z,
   402→        count: 90, speed: 10, color: 0xffc84a, size: 0.5, life: 2, gravity: -1, drag: 0.8, glow: 2.6,
   403→      });
   404→    }
   405→  }

   407→  /* ---------- Interacción ---------- */
   408→
   409→  private updateInteract() {
   410→    this.interactTarget = null;
   411→    const p = this.player.pos;
   412→    const dB = Math.hypot(p.x - WORLD.bonfire.x, p.z - WORLD.bonfire.z);
   413→    if (dB < 3.4) {
   414→      this.interactTarget = { kind: 'bonfire' };
   415→      return;
   416→    }
   417→    for (const sh of this.world.shrines) {
   418→      if (sh.cleansed) continue;
   419→      const guardians = this.enemies.filter(e => e.guardianOf === sh.idx && e.alive);
   420→      const d = Math.hypot(p.x - sh.pos.x, p.z - sh.pos.z);
   421→      if (d < 5 && guardians.length === 0) {
   422→        this.interactTarget = { kind: 'shrine', idx: sh.idx };
   423→        return;
   424→      }
   425→    }
   426→    if (this.world.sigil && this.world.sigil.group.visible && this.bossDefeated) {
   427→      const A = WORLD.arena;
   428→      const d = Math.hypot(p.x - A.x, p.z - A.z);
   429→      if (d < 4.5) this.interactTarget = { kind: 'sigil' };
   430→    }
   431→  }
   432→
   433→  private doInteract() {
   434→    const t = this.interactTarget;
   435→    if (!t) return;
   436→    if (t.kind === 'bonfire') {
   437→      const p = this.player;
   438→      p.hp = p.maxHp; p.stamina = p.maxStamina; p.potions = p.maxPotions;
   439→      this.audio.bonfireRest();
   440→      this.particles.burst({
   441→        x: p.pos.x, y: p.pos.y + 0.6, z: p.pos.z,
   442→        count: 26, speed: 2.6, color: 0xffb347, size: 0.3, life: 1.1, gravity: -1.5, drag: 1.5, glow: 2,
   443→      });
   444→      this.addDamageNumber(new THREE.Vector3(p.pos.x, p.pos.y + 2.3, p.pos.z), 'REPOSO', '#ffb347');
   445→    } else if (t.kind === 'shrine' && t.idx !== undefined) {
   446→      const sh = this.world.shrines[t.idx];
   447→      this.world.cleanseShrine(t.idx);
   448→      this.shrinesCleansed++;
   449→      this.audio.cleanse();
   450→      this.player.gainXp(60, this.ctx);
   451→      this.player.gold += 40;
   452→      this.particles.burst({
   453→        x: sh.pos.x, y: sh.pos.y + 4, z: sh.pos.z,
   454→        count: 70, speed: 7, color: 0x37d8c8, size: 0.42, life: 1.8, gravity: -0.5, drag: 0.9, glow: 2.4,
   455→      });
   456→      this.ctx.shake(0.5);
   457→      if (this.shrinesCleansed >= 3 && !this.bossDefeated && !this.bossActive) {
   458→        this.awakenTimer = 2.5;
   459→      }
   460→      this.emitHud();
   461→    } else if (t.kind === 'sigil') {
   462→      this.world.sigilReady();
   463→      this.spawnBoss(1.35 ** this.bossKills());
   464→      this.addDamageNumber(new THREE.Vector3(this.player.pos.x, this.player.pos.y + 2.3, this.player.pos.z), 'EL JEFE DESPIERTA DE NUEVO', '#ff5a4e', true);
   465→    }
   466→  }

   474→  private toggleLockOn() {
   475→    if (this.lockEnemy) { this.lockEnemy = null; this.player.lockTarget = null; return; }
   476→    let best: Enemy | null = null, bestD = 34;
   477→    for (const e of this.enemies) {
   478→      if (!e.alive || e.state === 'spawn') continue;
   479→      const d = e.pos.distanceTo(this.player.pos);
   480→      if (d < bestD) { bestD = d; best = e; }
   481→    }
   482→    this.lockEnemy = best;
   483→    this.player.lockTarget = best;
   484→  }

   486→  /* ---------- Bucle principal ---------- */
   487→
   488→  start() {
   489→    if (this.running) return;
   490→    this.running = true;
   491→    this.lastT = performance.now();
   492→    this.loop(this.lastT);
   493→  }
   494→
   495→  private loop = (t: number) => {
   496→    if (!this.running) return;
   497→    this.raf = requestAnimationFrame(this.loop);
   498→    const dt = Math.min(0.05, (t - this.lastT) / 1000);
   499→    this.lastT = t;
   500→    this.fps = this.fps * 0.95 + (1 / Math.max(0.0001, dt)) * 0.05;
   501→
   502→    if (this.phase === 'menu') {
   503→      this.menuT += dt;
   504→      this.updateMenuScene(dt);
   505→      this.composer.render();
   506→      this.hudTimer -= dt;
   507→      if (this.hudTimer <= 0) { this.hudTimer = 0.3; this.emitHud(); }
   508→      return;
   509→    }
   510→
   511→    if (this.phase === 'playing') {
   512→      this.autoQuality(dt);
   513→      // hit-stop y cámara lenta
   514→      if (this.hitStopT > 0) {
   515→        this.hitStopT -= dt;
   516→        this.updateWorld(dt * 0.06);
   517→      } else if (this.slowmoT > 0) {
   518→        this.slowmoT -= dt;
   519→        if (this.slowmoT <= 0) this.timeScale = 1;
   520→        this.updateWorld(dt * this.timeScale);
   521→      } else {
   522→        this.updateWorld(dt);
   523→      }
   524→    } else if (this.phase === 'dead' || this.phase === 'victory') {
   525→      // el mundo sigue con animaciones reducidas
   526→      this.updateWorld(dt * 0.25, true);
   527→    }
   528→
   529→    this.updateCamera(dt);
   530→    this.updateEffects(dt);
   531→    this.composer.render();
   532→
   533→    this.hudTimer -= dt;
   534→    if (this.hudTimer <= 0) { this.hudTimer = 0.08; this.emitHud(); this.drawMinimap(); }
   535→  };

   574→  private updateWorld(dt: number, frozen = false) {
   575→    this.elapsed += frozen ? 0 : dt;
   576→
   577→    // input → estado
   578→    const k = this.keys;
   579→    this.inputState.fwd = k.has('KeyW') || k.has('ArrowUp');
   580→    this.inputState.back = k.has('KeyS') || k.has('ArrowDown');
   581→    this.inputState.left = k.has('KeyA') || k.has('ArrowLeft');
   582→    this.inputState.right = k.has('KeyD') || k.has('ArrowRight');
   583→    this.inputState.sprint = k.has('ShiftLeft') || k.has('ShiftRight');
   584→
   585→    // jugador
   586→    this.ctx.input = frozen ? this.frozenInput : this.inputState;
   587→    this.ctx.camYaw = this.camYaw;
   588→    this.player.update(dt, this.ctx);
   589→    this.world.resolve(this.player.pos, this.player.radius);
   590→    this.worldClamp(this.player.pos, this.player.radius);
   591→
   592→    // resolver golpe pendiente del jugador
   593→    if (this.player.pendingStrike) {
   594→      this.resolvePlayerStrike(this.player.pendingStrike);
   595→      this.player.pendingStrike = null;
   596→    }
   597→
   598→    // lock-on: limpiar si muere o se aleja
   599→    if (this.lockEnemy && (!this.lockEnemy.alive || this.lockEnemy.pos.distanceTo(this.player.pos) > 38)) {
   600→      this.lockEnemy = null;
   601→      this.player.lockTarget = null;
   602→    }
   603→
   604→    // enemigos
   605→    for (const e of this.enemies) {
   606→      if (e.removable) continue;
   607→      e.update(dt, this.ctx);
   608→      if (e.alive) {
   609→        this.world.resolve(e.pos, e.radius);
   610→        this.worldClamp(e.pos, e.radius);
   611→      }
   612→    }
   613→    // separación entre enemigos
   614→    const live = this.enemies.filter(e => e.alive && e.state !== 'spawn');
   615→    for (let i = 0; i < live.length; i++) {
   616→      for (let j = i + 1; j < live.length; j++) {
   617→        const a = live[i], b = live[j];
   618→        const dx = b.pos.x - a.pos.x, dz = b.pos.z - a.pos.z;
   619→        const minD = a.radius + b.radius;
   620→        const d2 = dx * dx + dz * dz;
   621→        if (d2 < minD * minD && d2 > 1e-5) {
   622→          const d = Math.sqrt(d2);
   623→          const push = (minD - d) / d * 0.5;
   624→          a.pos.x -= dx * push; a.pos.z -= dz * push;
   625→          b.pos.x += dx * push; b.pos.z += dz * push;
   626→        }
   627→      }
   628→      // el jugador empuja y es empujado ligeramente
   629→      const p = this.player;
   630→      const dx = p.pos.x - live[i].pos.x, dz = p.pos.z - live[i].pos.z;
   631→      const minD = live[i].radius + p.radius;
   632→      const d2 = dx * dx + dz * dz;
   633→      if (d2 < minD * minD && d2 > 1e-5) {
   634→        const d = Math.sqrt(d2);
   635→        const push = (minD - d) / d * 0.6;
   636→        live[i].pos.x -= dx * push; live[i].pos.z -= dz * push;
   637→        p.pos.x += dx * push * 0.3; p.pos.z += dz * push * 0.3;
   638→      }
   639→    }
   640→    // limpiar cadáveres
   641→    if (this.enemies.some(e => e.removable)) {
   642→      for (const e of this.enemies) {
   643→        if (e.removable) { this.scene.remove(e.root); }
   644→      }
   645→      this.enemies = this.enemies.filter(e => !e.removable);
   646→    }
   647→
   648→    // proyectiles y drops
   649→    for (const pr of this.projectiles) pr.update(dt, this.ctx);
   650→    this.projectiles = this.projectiles.filter(pr => {
   651→      if (pr.dead) { this.scene.remove(pr.root); return false; }
   652→      return true;
   653→    });
   654→    for (const pk of this.pickups) {
   655→      pk.update(dt, this.ctx);
   656→      this.world.resolve(pk.pos, 0.3);
   657→    }
   658→    this.pickups = this.pickups.filter(pk => {
   659→      if (pk.dead) { this.scene.remove(pk.root); return false; }
   660→      return true;
   661→    });
   662→
   663→    // mundo visual
   664→    this.world.update(dt, this.camera, (x, y, z) => {
   665→      this.particles.spawn({
   666→        x, y, z, vx: rand(-0.3, 0.3), vy: rand(1, 2.2), vz: rand(-0.3, 0.3),
   667→        color: 0xff7a2a, size: rand(0.1, 0.2), life: rand(0.8, 1.6), glow: 2.2, drag: 0.5,
   668→      });
   669→    });
   670→
   671→    if (frozen) return;
   672→
   673→    // estela de espada
   674→    if (this.player.state === 'attack') {
   675→      const b = new THREE.Vector3(), ti = new THREE.Vector3();
   676→      this.player.swordPoints(b, ti);
   677→      this.trail.emit(b, ti);
   678→    }
   679→    this.trail.update(dt, this.player.state === 'attack');
   680→
   681→    // despawn/respawn de errantes
   682→    this.roamerTimer -= dt;
   683→    if (this.roamerTimer <= 0) {
   684→      this.roamerTimer = rand(3.5, 6);
   685→      const aliveRoamers = this.enemies.filter(e => e.alive && e.guardianOf === -1 && !e.isBoss).length;
   686→      if (aliveRoamers < ROAMER_TARGET) this.spawnRoamer();
   687→    }
   688→
   689→    // despertar del jefe
   690→    if (this.awakenTimer > 0) {
   691→      this.awakenTimer -= dt;
   692→      if (this.awakenTimer <= 0) this.spawnBoss();
   693→    }
   694→    // reaparición del jefe tras morir el jugador en plena lucha
   695→    if (this.bossRespawnT > 0) {
   696→      this.bossRespawnT -= dt;
   697→      if (this.bossRespawnT <= 0 && this.shrinesCleansed >= 3) this.spawnBoss();
   698→    }
   699→
   700→    // interacción
   701→    this.updateInteract();
   702→
   703→    // música: intensidad
   704→    if (this.elapsed % 1 < dt) {
   705→      let intensity = 0.12;
   706→      if (this.bossActive) intensity = 1;
   707→      else {
   708→        const near = this.enemies.some(e => e.alive && e.aggroed && e.pos.distanceTo(this.player.pos) < 30);
   709→        intensity = near ? 0.75 : 0.12;
   710→      }
   711→      this.audio.setIntensity(intensity);
   712→    }
   713→
   714→    // victoria
   715→    if (this.victoryDelay > 0) {
   716→      this.victoryDelay -= dt;
   717→      if (this.victoryDelay <= 0) {
   718→        this.setPhase('victory');
   719→      }
   720→    }
   721→  }

   730→  private resolvePlayerStrike(def: AttackDef) {
   731→    const p = this.player;
   732→    const heavy = def === PLAYER_HEAVY;
   733→    let hitAny = false;
   734→    for (const e of this.enemies) {
   735→      if (!e.alive || e.state === 'spawn') continue;
   736→      const to = new THREE.Vector3().subVectors(e.pos, p.pos).setY(0);
   737→      const d = to.length() - e.radius;
   738→      if (d > def.range) continue;
   739→      const ang = Math.atan2(to.x, to.z);
   740→      let diff = Math.abs(((ang - p.yaw + Math.PI * 3) % (Math.PI * 2)) - Math.PI);
   741→      if (diff <= def.arc / 2) {
   742→        const dmg = def.dmg * p.dmgMul;
   743→        e.takeDamage(dmg, p.pos, this.ctx, heavy);
   744→        hitAny = true;
   745→        this.particles.burst({
   746→          x: e.pos.x - to.x * 0.3, y: e.pos.y + e.rig.height * 0.55, z: e.pos.z - to.z * 0.3,
   747→          count: heavy ? 14 : 8, speed: 5, color: 0xffd9a0, size: 0.2, life: 0.35, drag: 2, glow: 2.4, gravity: 4,
   748→        });
   749→      }
   750→    }
   751→    if (hitAny) {
   752→      this.hitStopT = Math.max(this.hitStopT, heavy ? 0.11 : 0.07);
   753→      this.shakeAmt = Math.min(1.2, this.shakeAmt + (heavy ? 0.4 : 0.16));
   754→      if (heavy) this.audio.heavyHit();
   755→      else this.audio.hitMetal();
   756→    }
   757→  }

   759→  private worldClamp(pos: THREE.Vector3, radius: number) {
   760→    const r = Math.hypot(pos.x, pos.z);
   761→    const max = WORLD.radius - 2 - radius;
   762→    if (r > max) {
   763→      pos.x *= max / r;
   764→      pos.z *= max / r;
   765→    }
   766→  }

   770→  private updateCamera(dt: number) {
   771→    // ratón → rotación
   772→    if (this.locked && this.phase === 'playing') {
   773→      this.camYaw -= this.mouseDX * 0.0023;
   774→      this.camPitch += this.mouseDY * 0.0018;
   775→      this.camPitch = clamp(this.camPitch, -0.25, 1.15);
   776→    }
   777→    this.mouseDX = 0; this.mouseDY = 0;
   778→
   779→    const target = this.player.pos;
   780→    const head = new THREE.Vector3(target.x, target.y + 1.55, target.z);
   781→
   782→    if (this.lockEnemy && this.lockEnemy.alive) {
   783→      // cámara de objetivo fijado
   784→      const desired = Math.atan2(target.x - this.lockEnemy.pos.x, target.z - this.lockEnemy.pos.z);
   785→      this.camYaw = dampAngle(this.camYaw, desired, 3.2, dt);
   786→      this.camPitch = damp(this.camPitch, 0.3, 3, dt);
   787→    }
   788→
   789→    const dist = this.camDist;
   790→    const cp = Math.cos(this.camPitch), sp = Math.sin(this.camPitch);
   791→    const off = new THREE.Vector3(Math.sin(this.camYaw) * cp, sp, Math.cos(this.camYaw) * cp);
   792→    const desiredPos = head.clone().addScaledVector(off, dist);
   793→    // hombro
   794→    const right = new THREE.Vector3(-Math.cos(this.camYaw), 0, Math.sin(this.camYaw));
   795→    desiredPos.addScaledVector(right, 0.55);
   796→    // colisión con terreno
   797→    const groundH = terrainHeight(desiredPos.x, desiredPos.z) + 0.5;
   798→    if (desiredPos.y < groundH) desiredPos.y = groundH;
   799→
   800→    this.camPos.lerp(desiredPos, 1 - Math.exp(-(this.lockEnemy ? 9 : 12) * dt));
   801→
   802→    // sacudida
   803→    const sh = this.shakeAmt;
   804→    this.camera.position.copy(this.camPos);
   805→    if (sh > 0.001) {
   806→      this.camera.position.x += rand(-1, 1) * sh * 0.18;
   807→      this.camera.position.y += rand(-1, 1) * sh * 0.18;
   808→      this.camera.position.z += rand(-1, 1) * sh * 0.18;
   809→      this.shakeAmt = Math.max(0, sh - dt * 2.4);
   810→    }
   811→    const lookAt = head.clone();
   812→    if (this.lockEnemy && this.lockEnemy.alive) {
   813→      lookAt.lerp(new THREE.Vector3(this.lockEnemy.pos.x, this.lockEnemy.pos.y + 1.2, this.lockEnemy.pos.z), 0.22);
   814→    }
   815→    this.camera.lookAt(lookAt);
   816→
   817→    // FOV dinámico al esprintar
   818→    const targetFov = this.baseFov + (this.player.sprinting ? 5 : 0);
   819→    if (Math.abs(this.camera.fov - targetFov) > 0.05) {
   820→      this.camera.fov = damp(this.camera.fov, targetFov, 6, dt);
   821→      this.camera.updateProjectionMatrix();
   822→    }
   823→  }

   920→  private promptText(): string {

   923→      case 'bonfire': return 'E · Descansar en la hoguera (cura y reabastece)';
   924→      case 'shrine': return `E · Purificar ${WORLD.shrines[this.interactTarget.idx!].name}`;
   925→      case 'sigil': return 'E · Despertar al jefe de nuevo';
   926→    }
   927→  }

   940→  }

   943→    if (this.phase !== 'paused') return;

   945→    this.requestLock();
   946→  }

   948→  requestLock() {
   949→    const c = this.renderer.domElement;
   950→    if (document.pointerLockElement !== c) {
   951→      c.requestPointerLock?.();
   952→    }
   953→  }

   955→  respawn() {
   956→    // limpiar enemigos vivos, proyectiles y drops
   957→    for (const e of this.enemies) this.scene.remove(e.root);
   958→    this.enemies = [];
   959→    this.boss = null;
   960→    this.bossActive = false;
   961→    this.lockEnemy = null;
   962→    this.player.lockTarget = null;
   963→    for (const pr of this.projectiles) this.scene.remove(pr.root);
   964→    this.projectiles = [];
   965→    for (const pk of this.pickups) this.scene.remove(pk.root);
   966→    this.pickups = [];
   967→    // guardianes frescos en santuarios sin purificar
   968→    this.spawnGuardians();
   969→    for (let i = 0; i < 4; i++) this.spawnRoamer(true);
   970→    // jugador
   971→    const p = this.player;
   972→    p.hp = p.maxHp; p.stamina = p.maxStamina; p.potions = Math.max(p.potions, p.maxPotions);
   973→    p.state = 'idle'; p.stateT = 0; p.iFrames = 2;
   974→    p.knock.set(0, 0, 0);
   975→    p.pos.set(WORLD.bonfire.x, 0, WORLD.bonfire.z + 3.4);
   976→    p.pos.y = terrainHeight(p.pos.x, p.pos.z);
   977→    p.yaw = Math.atan2(WORLD.bonfire.x - p.pos.x, WORLD.bonfire.z - p.pos.z) + Math.PI;
   978→    this.camYaw = p.yaw + Math.PI;
   979→    if (this.shrinesCleansed >= 3 && !this.bossDefeated) this.bossRespawnT = 6;
   980→    this.setPhase('playing');
   981→    this.requestLock();
   982→  }

   998→  private bindEvents() {
   999→    window.addEventListener('resize', this.onResize);
  1000→    window.addEventListener('keydown', this.onKeyDown);
  1001→    window.addEventListener('keyup', this.onKeyUp);
  1002→    document.addEventListener('mousemove', this.onMouseMove);
  1003→    document.addEventListener('pointerlockchange', this.onLockChange);
  1004→    const c = this.renderer.domElement;
  1005→    c.addEventListener('mousedown', this.onMouseDown);
  1006→    c.addEventListener('contextmenu', this.onContextMenu);
  1007→  }

  1009→  private onResize = () => {
  1010→    this.camera.aspect = window.innerWidth / window.innerHeight;
  1011→    this.camera.updateProjectionMatrix();
  1012→    this.renderer.setSize(window.innerWidth, window.innerHeight);
  1013→    this.composer.setSize(window.innerWidth, window.innerHeight);
  1014→  };
  1015→
  1016→  private onKeyDown = (e: KeyboardEvent) => {
  1017→    if (e.code === 'Tab' || e.code === 'Space') e.preventDefault();
  1018→    if (this.phase !== 'playing') return;
  1019→    this.keys.add(e.code);
  1020→    switch (e.code) {
  1021→      case 'Tab': this.toggleLockOn(); break;
  1022→      case 'Space': this.queued.roll = true; break;
  1023→      case 'KeyF': this.queued.potion = true; break;
  1024→      case 'KeyE': this.doInteract(); break;
  1025→    }
  1026→  };
  1027→
  1028→  private onKeyUp = (e: KeyboardEvent) => {
  1029→    this.keys.delete(e.code);
  1030→  };
  1031→
  1032→  private onMouseMove = (e: MouseEvent) => {
  1033→    if (!this.locked) return;
  1034→    this.mouseDX += e.movementX;
  1035→    this.mouseDY += e.movementY;
  1036→  };
  1037→
  1038→  private onMouseDown = (e: MouseEvent) => {
  1039→    if (this.phase !== 'playing') return;
  1040→    if (!this.locked) { this.requestLock(); return; }
  1041→    if (e.button === 0) this.queued.attack = true;
  1042→    if (e.button === 2) this.queued.heavy = true;
  1043→  };
  1044→
  1045→  private onContextMenu = (e: Event) => { e.preventDefault(); };
  1046→
  1047→  private onLockChange = () => {
  1048→    this.locked = document.pointerLockElement === this.renderer.domElement;
  1049→    if (!this.locked && this.phase === 'playing') this.pause();
  1050→  };

  1054→  dispose() {
  1055→    this.running = false;
  1056→    cancelAnimationFrame(this.raf);

  1058→    window.removeEventListener('keydown', this.onKeyDown);
  1059→    window.removeEventListener('keyup', this.onKeyUp);
  1060→    document.removeEventListener('mousemove', this.onMouseMove);
  1061→    document.removeEventListener('pointerlockchange', this.onLockChange);
  1062→    this.audio.dispose();
  1063→    this.renderer.dispose();
  1064→    this.renderer.domElement.remove();
  1065→    this.dmgLayer.remove();
  1066→  }
