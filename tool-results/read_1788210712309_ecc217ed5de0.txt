    44→const AURORA_FRAG = /* glsl */`
    45→varying vec2 vUv;
    46→uniform float uTime; uniform float uPhase; uniform float uGlobalA; uniform vec3 uColA; uniform vec3 uColB;
    47→void main(){
    48→  float x = vUv.x * 6.28318;
    49→  float band = sin(x*2.0 + uTime*0.11 + uPhase) * 0.5
    50→             + sin(x*5.0 - uTime*0.17 + uPhase*2.1) * 0.3
    51→             + sin(x*11.0 + uTime*0.29) * 0.2;
    52→  float y = vUv.y;
    53→  float curtain = smoothstep(0.02, 0.3, y) * (1.0 - smoothstep(0.45, 0.98, y));
    54→  float rays = 0.5 + 0.5 * sin(x*26.0 + band*4.0 + uTime*0.4 + uPhase*3.0);
    55→  float i = curtain * (0.35 + 0.65 * rays);
    56→  vec3 col = mix(uColA, uColB, clamp(y * 1.5, 0.0, 1.0));
    57→  gl_FragColor = vec4(col * i * 0.9 * uGlobalA, i * uGlobalA);
    58→}`;

   125→  resolve(pos: THREE.Vector3, radius: number) {
   126→    for (const c of this.colliders) {
   127→      const dx = pos.x - c.x, dz = pos.z - c.z;
   128→      const d2 = dx * dx + dz * dz;
   129→      const minD = c.r + radius;
   130→      if (d2 < minD * minD && d2 > 1e-6) {
   131→        const d = Math.sqrt(d2);
   132→        const push = (minD - d) / d;
   133→        pos.x += dx * push;
   134→        pos.z += dz * push;
   135→      }
   136→    }
   137→  }

   174→        uniform vec3 uSunDir; uniform vec3 uSunTint; uniform float uSunGlow;
   175→        void main(){
   176→          vec3 n = normalize(vP);

   182→          float sd = max(dot(n, normalize(uSunDir)), 0.0);

   329→  /**
   330→   * Aplica el estado del ciclo día/noche a cielo, luces, niebla,
   331→   * sol/luna, estrellas, aurora, luciérnagas y ambiente nocturno.
   332→   */
   333→  applyDayNight(s: DayNightSample, renderer: THREE.WebGLRenderer) {
   334→    this.nightK = s.night;
   335→    this.mistMul = s.mistMul;
   336→
   337→    // cielo
   338→    const su = this.skyMat.uniforms;
   339→    (su.top.value as THREE.Color).copy(s.skyTop);
   340→    (su.mid.value as THREE.Color).copy(s.skyMid);
   341→    (su.bottom.value as THREE.Color).copy(s.skyBottom);
   342→    this.dn.sunDir.value.copy(s.lightDir);
   343→    this.dn.sunTint.value.copy(s.sunTint);
   344→    this.dn.sunGlow.value = s.sunGlow;
   345→
   346→    // estrellas, aurora, luciérnagas
   347→    this.dn.starsA.value = s.stars;
   348→    this.dn.auroraA.value = s.aurora;
   349→    this.dn.ffA.value = s.fireflies;
   350→
   351→    // sol y luna visibles
   352→    this.sunSprites.forEach((sp) => { sp.position.copy(s.lightDir).multiplyScalar(385); });
   353→    if (this.sunMat) this.sunMat.opacity = 0.85 * s.sunA;
   354→    if (this.sunGlowMat) this.sunGlowMat.opacity = 0.34 * s.sunA;
   355→    if (this.moonGroup) {
   356→      this.moonGroup.visible = s.moonA > 0.02;
   357→      this.moonGroup.position.set(0, 0, 0);
   358→    }
   359→    if (this.moonMat) this.moonMat.opacity = s.moonA;
   360→    this.moonHaloMats.forEach((m, i) => { m.opacity = (i === 0 ? 0.5 : 0.16) * s.moonA; });
   361→
   362→    // luz direccional con sombras (sol de día / luna de noche)
   363→    this.moonLight.color.copy(s.lightColor);
   364→    this.moonLight.intensity = s.lightIntensity;
   365→
   366→    // hemisférica + relleno
   367→    this.hemi.color.copy(s.hemiSky);
   368→    this.hemi.groundColor.copy(s.hemiGround);
   369→    this.hemi.intensity = s.hemiIntensity;
   370→    this.fill.color.copy(s.fillColor);
   371→    this.fill.intensity = s.fillIntensity;
   372→
   373→    // niebla volumétrica global
   374→    if (this.scene.fog instanceof THREE.FogExp2) {
   375→      this.scene.fog.color.copy(s.fogColor);
   376→      this.scene.fog.density = s.fogDensity;
   377→    }
   378→
   379→    // agua de la Fuente Lunar
   380→    if (this.waterMat) {
   381→      (this.waterMat.uniforms.uSky.value as THREE.Color).copy(s.waterTint);
   382→      (this.waterMat.uniforms.uMoonDir.value as THREE.Vector3).copy(s.lightDir);
   383→    }
   384→
   385→    // exposición y reflejos de entorno
   386→    renderer.toneMappingExposure = s.exposure;
   387→    this.envIntensityTarget = s.envIntensity;
   388→    const sc = this.scene as THREE.Scene & { environmentIntensity?: number };
   389→    if (sc.environmentIntensity !== undefined) sc.environmentIntensity = s.envIntensity;
   390→  }

   394→  private buildTerrain() {
   395→    const seg = 150;
   396→    const geo = new THREE.PlaneGeometry(WORLD.size, WORLD.size, seg, seg);
   397→    geo.rotateX(-Math.PI / 2);
   398→    const pos = geo.getAttribute('position') as THREE.BufferAttribute;
   399→    const colors = new Float32Array(pos.count * 3);
   400→    const c = new THREE.Color();
   401→    for (let i = 0; i < pos.count; i++) {
   402→      const x = pos.getX(i), z = pos.getZ(i);
   403→      const h = terrainHeight(x, z);
   404→      pos.setY(i, h);
   405→      // sombreado por pendiente + variación
   406→      const n = fbm(x * 0.11, z * 0.11, 3) * 0.5 + 0.5;
   407→      const slope = 1 - Math.min(1, Math.abs(fbm(x * 0.05 + 31, z * 0.05 - 12, 2)) * 1.4);
   408→      const shade = (0.86 + n * 0.22) * lerp(0.82, 1, slope);
   409→      c.setRGB(shade, shade, shade);
   410→      colors[i * 3] = c.r; colors[i * 3 + 1] = c.g; colors[i * 3 + 2] = c.b;
   411→    }
   412→    geo.setAttribute('color', new THREE.BufferAttribute(colors, 3));
   413→    geo.computeVertexNormals();
   414→    const mat = new THREE.MeshStandardMaterial({
   415→      map: terrainSplat(),
   416→      normalMap: pbrTex('grass_normal.jpg', { srgb: false, repeat: 46 }),
   417→      normalScale: new THREE.Vector2(0.55, 0.55),
   418→      bumpMap: terrainDetailNormal(),
   419→      bumpScale: 0.06,
   420→      roughnessMap: pbrTex('grass_rough.jpg', { srgb: false, repeat: 46 }),
   421→      vertexColors: true,
   422→      roughness: 1.0, metalness: 0,
   423→      envMapIntensity: 0.35,
   424→    });
   425→    this.terrainMat = mat;
   426→    const terrain = new THREE.Mesh(geo, mat);
   427→    terrain.receiveShadow = true;
   428→    terrain.name = 'terrain';
   429→    this.scene.add(terrain);
   430→  }

   439→    // ==== Pinos: tronco + copa fusionada de 3 conos ====
   440→    const canopyGeo = (() => {
   441→      const parts: THREE.BufferGeometry[] = [];
   442→      const defs: [number, number, number][] = [ // [radio, alto, y]
   443→        [1.95, 3.1, 3.0], [1.5, 2.7, 4.7], [1.0, 2.3, 6.2],

   698→  private buildFireflies() {
   699→    const rng = mulberry32(555);
   700→    const N = 130;
   701→    const pos = new Float32Array(N * 3);
   702→    const phase = new Float32Array(N);
   703→    const speed = new Float32Array(N);
   704→    const size = new Float32Array(N);
   705→    let i = 0, guard = 0;
   706→    while (i < N && guard++ < 3000) {
   707→      const a = rng() * Math.PI * 2;
   708→      const r = 8 + rng() * (WORLD.radius - 16);
   709→      const x = Math.cos(a) * r, z = Math.sin(a) * r;
   710→      const h = terrainHeight(x, z);
   711→      if (h > 6) continue;
   712→      pos[i * 3] = x; pos[i * 3 + 1] = h + 0.4 + rng() * 1.4; pos[i * 3 + 2] = z;
   713→      phase[i] = rng() * Math.PI * 2;
   714→      speed[i] = 0.5 + rng() * 0.9;
   715→      size[i] = 3 + rng() * 4;
   716→      i++;
   717→    }
   718→    const geo = new THREE.BufferGeometry();
   719→    geo.setAttribute('position', new THREE.BufferAttribute(pos, 3));
   720→    geo.setAttribute('aPhase', new THREE.BufferAttribute(phase, 1));
   721→    geo.setAttribute('aSpeed', new THREE.BufferAttribute(speed, 1));
   722→    geo.setAttribute('aSize', new THREE.BufferAttribute(size, 1));
   723→    const mat = new THREE.ShaderMaterial({
   724→      transparent: true, depthWrite: false, blending: THREE.AdditiveBlending,
   725→      uniforms: { uTime: this.skyTime, uGlobalA: this.dn.ffA },
   726→      vertexShader: `attribute float aPhase; attribute float aSpeed; attribute float aSize;
   727→        uniform float uTime; uniform float uGlobalA; varying float vA;
   728→        void main(){
   729→          vec3 p = position;
   730→          p.x += sin(uTime * aSpeed + aPhase) * 1.3;
   731→          p.y += sin(uTime * aSpeed * 0.7 + aPhase * 2.0) * 0.55;
   732→          p.z += cos(uTime * aSpeed * 0.85 + aPhase) * 1.3;
   733→          vec4 mv = modelViewMatrix * vec4(p, 1.0);
   734→          gl_PointSize = clamp(aSize * (150.0 / max(0.1, -mv.z)), 1.0, 26.0);
   735→          float nearFade = smoothstep(2.2, 6.0, -mv.z);
   736→          vA = (pow(0.5 + 0.5 * sin(uTime * 2.1 + aPhase * 3.0), 2.2) * 0.9 + 0.06) * nearFade * uGlobalA;
   737→          gl_Position = projectionMatrix * mv; }`,
   738→      fragmentShader: `varying float vA;
   739→        void main(){ vec2 uv = gl_PointCoord - 0.5; float d = length(uv);
   740→          float a = smoothstep(0.5, 0.06, d) * vA;
   741→          if (a < 0.01) discard;
   742→          gl_FragColor = vec4(vec3(0.82, 1.0, 0.55) * 1.6, a); }`,
   743→    });
   744→    const pts = new THREE.Points(geo, mat);
   745→    pts.frustumCulled = false;
   746→    pts.renderOrder = 6;
   747→    this.scene.add(pts);
   748→  }

  1114→      const s = 1 + Math.sin(this.time * 11) * 0.12;
