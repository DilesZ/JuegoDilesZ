    22→/* Grading final cinematográfico: contraste S suave, viñeta, grano, CA */
    23→const GradeShader = {
    24→  uniforms: {
    25→    tDiffuse: { value: null as THREE.Texture | null },
    26→    uTime: { value: 0 },
    27→    uVignette: { value: 0.34 },
    28→    uGrain: { value: 0.013 },
    29→    uCA: { value: 0.0009 },
    30→    uSat: { value: 1.04 },
    31→    uCon: { value: 1.06 },
    32→  },
    33→  vertexShader: `varying vec2 vUv; void main(){ vUv = uv; gl_Position = projectionMatrix * modelViewMatrix * vec4(position,1.0); }`,
    34→  fragmentShader: `
    35→    uniform sampler2D tDiffuse; uniform float uTime; uniform float uVignette;
    36→    uniform float uGrain; uniform float uCA; uniform float uSat; uniform float uCon;
    37→    varying vec2 vUv;
    38→    float hash(vec2 p){ return fract(sin(dot(p, vec2(12.9898,78.233))) * 43758.5453); }
    39→    void main(){
    40→      vec2 uv = vUv;
    41→      vec2 d = uv - 0.5;
    42→      float r2 = dot(d, d);
    43→      vec2 off = d * r2 * uCA * 8.0;
    44→      vec3 col;
    45→      col.r = texture2D(tDiffuse, uv + off).r;
    46→      col.g = texture2D(tDiffuse, uv).g;
    47→      col.b = texture2D(tDiffuse, uv - off).b;
    48→      float l = dot(col, vec3(0.299, 0.587, 0.114));
    49→      col = mix(vec3(l), col, uSat);
    50→      // curva S suave (sombras densas, altas luces limpias)
    51→      col = col * col * (3.0 - 2.0 * col) * 0.28 + col * 0.72;
    52→      col = (col - 0.5) * uCon + 0.5;
    53→      col *= 1.0 - uVignette * smoothstep(0.12, 0.72, r2);
    54→      col += (hash(uv * vec2(1287.0, 731.0) + fract(uTime) * 43.7) - 0.5) * uGrain;
    55→      gl_FragColor = vec4(clamp(col, 0.0, 1.0), 1.0);
    56→    }`,
    57→};

    59→/* Onda expansiva en el suelo para impactos pesados (DMC) */
    60→class Shockwave {
    61→  mesh: THREE.Mesh;
    62→  private t = 0;
    63→  private maxR: number;
    64→  private mat: THREE.MeshBasicMaterial;
    65→  constructor(scene: THREE.Scene, pos: THREE.Vector3, color = 0xffd9a0, maxR = 3.4) {
    66→    this.maxR = maxR;
    67→    const geo = new THREE.RingGeometry(0.42, 0.62, 36);
    68→    geo.rotateX(-Math.PI / 2);
    69→    this.mat = new THREE.MeshBasicMaterial({
    70→      color, transparent: true, opacity: 0.85,
    71→      blending: THREE.AdditiveBlending, depthWrite: false, side: THREE.DoubleSide,
    72→    });
    73→    this.mesh = new THREE.Mesh(geo, this.mat);
    74→    this.mesh.position.copy(pos);
    75→    this.mesh.position.y += 0.12;
    76→    this.mesh.renderOrder = 8;
    77→    scene.add(this.mesh);
    78→  }
    79→  update(dt: number): boolean {
    80→    this.t += dt;
    81→    const k = this.t / 0.42;
    82→    if (k >= 1) return false;
    83→    const r = 0.4 + (this.maxR - 0.4) * (1 - Math.pow(1 - k, 2.4));
    84→    this.mesh.scale.setScalar(r);
    85→    this.mat.opacity = 0.85 * (1 - k) * (1 - k);
    86→    return true;
    87→  }
    88→  dispose(scene: THREE.Scene) {
    89→    scene.remove(this.mesh);
    90→    this.mesh.geometry.dispose();
    91→    this.mat.dispose();
    92→  }
    93→}

   709→  private handleEnemyDied(e: Enemy) {
   710→    this.player.kills++;
   711→    const cfg = ENEMY_CFG[e.type];
   712→    const gold = randInt(cfg.gold[0], cfg.gold[1]);
   713→    this.player.gold += gold;
   714→    this.audio.coin();
   715→    this.player.gainXp(cfg.xp, this.ctx);
   716→    // REAPARICIÓN ESTILO MMORPG: el punto de spawn repuebla tras un tiempo
   717→    if (!e.isBoss) {
   718→      const shrineCleansed = e.guardianOf >= 0 && this.world.shrines[e.guardianOf].cleansed;
   719→      if (!shrineCleansed) {
   720→        if (e.guardianOf >= 0) {
   721→          // guardias: los santuarios sin purificar vuelven a llenarse
   722→          this.respawnQueue.push({
   723→            type: e.type, x: e.home.x, z: e.home.z, guardianOf: e.guardianOf,
   724→            t: RESPAWN_T[e.type] * rand(0.85, 1.25),
   725→          });
   726→        } else {
   727→          // errantes: solo se reponen si la población no supera el objetivo
   728→          const pendingRoam = this.respawnQueue.filter(r => r.guardianOf === -1).length;
   729→          const aliveRoam = this.enemies.filter(o => o.alive && o.guardianOf === -1 && !o.isBoss).length;
   730→          if (aliveRoam + pendingRoam < ROAMER_TARGET + 2) {
   731→            this.respawnQueue.push({
   732→              type: e.type, x: e.home.x, z: e.home.z, guardianOf: -1,
   733→              t: RESPAWN_T[e.type] * rand(0.85, 1.25),
   734→            });
   735→          }
   736→        }
   737→      }
   738→    } else {
   739→      // jefe del mundo: reaparece en modo infinito, cada vez más fuerte
   740→      this.bossRespawnT = RESPAWN_T.boss;
   741→    }
   742→    // orbes de alma
   743→    for (let i = 0; i < 10; i++) {
   744→      this.particles.spawn({
   745→        x: e.pos.x + rand(-0.6, 0.6), y: e.pos.y + 0.6, z: e.pos.z + rand(-0.6, 0.6),
   746→        color: 0x6ae8ff, size: 0.26, life: 2.2, glow: 2.2, shrink: false,
   747→        seek: { getPos: () => this.player.pos, speed: 6 },
   748→      });
   749→    }
   750→    // drop de poción
   751→    if (Math.random() < cfg.potionChance) {
   752→      this.pickups.push(new Pickup(e.pos.clone(), 'potion'));
   753→      this.scene.add(this.pickups[this.pickups.length - 1].root);
   754→    }
   755→    // drop de objeto (armas, armaduras, accesorios…)
   756→    const dropChance = e.isBoss ? 1 : DROP_CHANCE[e.type];
   757→    if (Math.random() < dropChance) {
   758→      const def = rollDrop(this.player.level, e.isBoss);
   759→      const pk = new Pickup(
   760→        new THREE.Vector3(e.pos.x + rand(-0.8, 0.8), 0, e.pos.z + rand(-0.8, 0.8)),

   762→      );
   763→      this.pickups.push(pk);
   764→      this.scene.add(pk.root);
   765→    }
   766→    // guardianes → santuario purificable
   767→    if (e.guardianOf >= 0) {
   768→      const alive = this.enemies.filter(o => o.guardianOf === e.guardianOf && o.alive && o !== e).length;
   769→      if (alive === 0) {
   770→        // todos los guardias muertos
   771→        void e.guardianOf;
   772→      }
   773→    }
   774→    if (e.isBoss) {
   775→      this.bossActive = false;
   776→      this.bossDefeated = true;
   777→      this.endless = true;
   778→      this.world.activateSigil();
   779→      this.world.sigilReady();
   780→      this.slowmoT = 2.2;
   781→      this.timeScale = 0.3;
   782→      this.victoryDelay = 2.0;
   783→      this.audio.setIntensity(0.2);
   784→      this.audio.cleanse();
   785→      this.ctx.shake(1);
   786→      this.particles.burst({
   787→        x: e.pos.x, y: e.pos.y + 2, z: e.pos.z,
   788→        count: 90, speed: 10, color: 0xffc84a, size: 0.5, life: 2, gravity: -1, drag: 0.8, glow: 2.6,
   789→      });
   790→    }
   791→  }

   795→  private updateInteract() {
   796→    this.interactTarget = null;
   797→    const p = this.player.pos;
   798→    const dB = Math.hypot(p.x - WORLD.bonfire.x, p.z - WORLD.bonfire.z);

   800→      this.interactTarget = { kind: 'bonfire' };
   801→      return;
   802→    }
   803→    // puesto del mercader
   804→    if (merchantDist(p.x, p.z) < 2.7) {
   805→      this.interactTarget = { kind: 'merchant' };
   806→      return;
   807→    }
   808→    for (const sh of this.world.shrines) {
   809→      if (sh.cleansed) continue;
   810→      const guardians = this.enemies.filter(e => e.guardianOf === sh.idx && e.alive);
   811→      const d = Math.hypot(p.x - sh.pos.x, p.z - sh.pos.z);
   812→      if (d < 5 && guardians.length === 0) {
   813→        this.interactTarget = { kind: 'shrine', idx: sh.idx };
   814→        return;
   815→      }
   816→    }
   817→    if (this.world.sigil && this.world.sigil.group.visible && this.bossDefeated) {
   818→      const A = WORLD.arena;
   819→      const d = Math.hypot(p.x - A.x, p.z - A.z);
   820→      if (d < 4.5) this.interactTarget = { kind: 'sigil' };
   821→    }

   881→  start() {
   882→    if (this.running) return;
   883→    this.running = true;
   884→    this.lastT = performance.now();
   885→    this.loop(this.lastT);

   887→
   888→  private loop = (t: number) => {
   889→    if (!this.running) return;
   890→    this.raf = requestAnimationFrame(this.loop);
   891→    const dt = Math.min(0.05, (t - this.lastT) / 1000);
   892→    this.lastT = t;
   893→    this.fps = this.fps * 0.95 + (1 / Math.max(0.0001, dt)) * 0.05;
   894→
   895→    if (this.phase === 'menu') {
   896→      this.menuT += dt;
   897→      this.cycle.update(dt); // el mundo vive detrás del menú (con ciclo día/noche)
   898→      this.updateMenuScene(dt);
   899→      this.composer.render();
   900→      this.hudTimer -= dt;
   901→      if (this.hudTimer <= 0) { this.hudTimer = 0.3; this.emitHud(); }
   902→      return;
   903→    }
   904→
   905→    if (this.phase === 'playing') {
   906→      this.autoQuality(dt);
   907→      // ciclo día/noche (los enemigos son más rápidos de noche)
   908→      this.cycle.update(dt);
   909→      this.ctx.nightFactor = this.cycle.nightFactor;
   910→      if (this.noticeT > 0) { this.noticeT -= dt; if (this.noticeT <= 0) this.notice = ''; }
   911→      // pasos del jugador sobre el terreno
   912→      const stepDx = this.player.pos.x - this.lastStepPos.x;
   913→      const stepDz = this.player.pos.z - this.lastStepPos.z;
   914→      this.stepDist += Math.hypot(stepDx, stepDz);
   915→      this.lastStepPos.copy(this.player.pos);
   916→      if (this.stepDist > 2.2 && this.player.state !== 'roll' && this.player.state !== 'dead') {
   917→        this.stepDist = 0;
   918→        this.audio.footstep();
   919→        // polvo al correr (más denso esprintando)
   920→        if (this.player.moving) {
   921→          const n = this.player.sprinting ? 3 : 1;
   922→          for (let i = 0; i < n; i++) {
   923→            this.particles.spawn({
   924→              x: this.player.pos.x + rand(-0.25, 0.25),
   925→              y: this.player.pos.y + 0.08,
   926→              z: this.player.pos.z + rand(-0.25, 0.25),
   927→              vx: rand(-0.6, 0.6), vy: rand(0.3, 1.0), vz: rand(-0.6, 0.6),
   928→              color: 0x9a8a6a, size: rand(0.16, 0.3), life: 0.5, gravity: 0.5, drag: 2.2, glow: 0.35,
   929→            });
   930→          }
   931→        }
   932→      }
   933→      if (this.uiOpen) {
   934→        // inventario abierto: mundo congelado, escena viva
   935→        this.updateWorld(0.0001, true);
   936→      } else if (this.hitStopT > 0) {
   937→        // hit-stop y cámara lenta
   938→        this.hitStopT -= dt;
   939→        this.updateWorld(dt * 0.06);

  1202→  private resolvePlayerStrike(def: AttackDef) {
  1203→    const p = this.player;
  1204→    const heavy = def === PLAYER_HEAVY;
  1205→    let hitAny = false;
  1206→    for (const e of this.enemies) {
  1207→      if (!e.alive || e.state === 'spawn') continue;
  1208→      const to = new THREE.Vector3().subVectors(e.pos, p.pos).setY(0);
  1209→      const d = to.length() - e.radius;
  1210→      if (d > def.range) continue;
  1211→      const ang = Math.atan2(to.x, to.z);
  1212→      let diff = Math.abs(((ang - p.yaw + Math.PI * 3) % (Math.PI * 2)) - Math.PI);
  1213→      if (diff <= def.arc / 2) {
  1214→        const dmg = def.dmg * p.dmgMul;
  1215→        e.takeDamage(dmg, p.pos, this.ctx, heavy);
  1216→        hitAny = true;
  1217→        // chispas direccionales del impacto (DMC)
  1218→        const hitPos = new THREE.Vector3(
  1219→          e.pos.x - to.x * 0.3,
  1220→          e.pos.y + e.rig.height * 0.55,
  1221→          e.pos.z - to.z * 0.3,
  1222→        );
  1223→        this.particles.burst({
  1224→          x: hitPos.x, y: hitPos.y, z: hitPos.z,
  1225→          count: heavy ? 22 : 10, speed: heavy ? 7 : 5, color: 0xffd9a0,
  1226→          size: heavy ? 0.26 : 0.2, life: 0.38, drag: 2, glow: 2.6, gravity: 4,
  1227→        });
  1228→        if (heavy) {
  1229→          this.particles.burst({
  1230→            x: hitPos.x, y: hitPos.y, z: hitPos.z,
  1231→            count: 12, speed: 4.5, color: 0xff8848, size: 0.3, life: 0.5, drag: 1.6, glow: 2.2, gravity: 2,
  1232→          });
  1233→        }
  1234→      }
  1235→    }
  1236→    if (hitAny) {
  1237→      this.hitStopT = Math.max(this.hitStopT, heavy ? 0.13 : 0.07);
  1238→      this.shakeAmt = Math.min(1.2, this.shakeAmt + (heavy ? 0.5 : 0.16));

  1241→        this.fovKickAmt = Math.min(7, this.fovKickAmt + 4.5);
  1242→        const impact = p.pos.clone().addScaledVector(new THREE.Vector3(Math.sin(p.yaw), 0, Math.cos(p.yaw)), 1.7);
  1243→        impact.y = terrainHeight(impact.x, impact.z);
  1244→        this.ctx.shockwave(impact, 0xffc87d, 3.6);
  1245→      } else {
  1246→        this.audio.hitMetal();
  1247→      }
  1248→    }
  1249→  }

  1541→  private onKeyDown = (e: KeyboardEvent) => {
  1542→    if (e.code === 'Tab' || e.code === 'Space') e.preventDefault();
  1543→    // con un panel abierto (mochila/tienda) solo se permite cerrarlo
  1544→    if (this.uiOpen) {
  1545→      if (e.code === 'Escape' || e.code === 'KeyI' || e.code === 'KeyB') {
  1546→        if (this.uiPanel === 'shop') this.setPanel(null);
  1547→        else this.toggleInventory();
  1548→      }
  1549→      return;
  1550→    }
  1551→    if (this.phase !== 'playing') return;
  1552→    this.keys.add(e.code);
  1553→    switch (e.code) {
  1554→      case 'Tab': this.toggleLockOn(); break;
  1555→      case 'Space': this.queued.roll = true; break;
  1556→      case 'KeyF': this.queued.potion = true; break;
  1557→      case 'KeyE': this.doInteract(); break;
  1558→      case 'KeyI': case 'KeyB': this.toggleInventory(); break;
  1559→    }
  1560→  };
