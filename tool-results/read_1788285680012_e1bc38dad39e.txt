    22→    22→const GradeShader = {
    23→    23→  uniforms: {
    24→    24→    tDiffuse: { value: null as THREE.Texture | null },
    25→    25→    uTime: { value: 0 },
    26→    26→    uVignette: { value: 0.30 },
    27→    27→    uGrain: { value: 0.016 },
    28→    28→    uCA: { value: 0.0006 },
    29→    29→    uSat: { value: 1.18 },
    30→    30→    uCon: { value: 1.03 },
    31→    31→  },
    32→    32→  vertexShader: `varying vec2 vUv; void main(){ vUv = uv; gl_Position = projectionMatrix * modelViewMatrix * vec4(position,1.0); }`,
    33→    33→  fragmentShader: `
    34→    34→    uniform sampler2D tDiffuse; uniform float uTime; uniform float uVignette;
    35→    35→    uniform float uGrain; uniform float uCA; uniform float uSat; uniform float uCon;
    36→    36→    varying vec2 vUv;
    37→    37→    float hash(vec2 p){ return fract(sin(dot(p, vec2(12.9898,78.233))) * 43758.5453); }
    38→    38→    void main(){
    39→    39→      vec2 uv = vUv;
    40→    40→      vec2 d = uv - 0.5;
    41→    41→      float r2 = dot(d, d);
    42→    42→      vec2 off = d * r2 * uCA * 8.0;
    43→    43→      vec3 col;
    44→    44→      col.r = texture2D(tDiffuse, uv + off).r;
    45→    45→      col.g = texture2D(tDiffuse, uv).g;
    46→    46→      col.b = texture2D(tDiffuse, uv - off).b;
    47→    47→      float l = dot(col, vec3(0.299, 0.587, 0.114));
    48→    48→      col = mix(vec3(l), col, uSat);
    49→    49→      col = (col - 0.5) * uCon + 0.5;
    50→    50→      col *= 1.0 - uVignette * smoothstep(0.12, 0.72, r2);
    51→    51→      col += (hash(uv * vec2(1287.0, 731.0) + fract(uTime) * 43.7) - 0.5) * uGrain;
    52→    52→      gl_FragColor = vec4(clamp(col, 0.0, 1.0), 1.0);
    53→    53→    }`,
    54→    54→};

   403→   403→  private pixelRatioFor(q: QualityTier): number {
   404→   404→    if (q === 'bajo') return 0.85;
   405→   405→    if (q === 'medio') return 1.2;
   406→   406→    return Math.min(window.devicePixelRatio, 1.75);

   409→   409→  setQuality(q: QualityTier) {
   410→   410→    this.quality = q;
   411→   411→    const pr = this.pixelRatioFor(q);
   412→   412→    this.renderer.setPixelRatio(pr);
   413→   413→    this.composer.setPixelRatio(pr);

   664→   664→
   665→   665→  start() {
   666→   666→    if (this.running) return;
   667→   667→    this.running = true;

   671→   671→
   672→   672→  private loop = (t: number) => {
   673→   673→    if (!this.running) return;
   674→   674→    this.raf = requestAnimationFrame(this.loop);
   675→   675→    const dt = Math.min(0.05, (t - this.lastT) / 1000);
   676→   676→    this.lastT = t;
   677→   677→    this.fps = this.fps * 0.95 + (1 / Math.max(0.0001, dt)) * 0.05;
   678→   678→
   679→   679→    if (this.phase === 'menu') {
   680→   680→      this.menuT += dt;
   681→   681→      this.cycle.update(dt); // el mundo vive detrás del menú (con ciclo día/noche)
   682→   682→      this.updateMenuScene(dt);
   683→   683→      this.composer.render();
   684→   684→      this.hudTimer -= dt;
   685→   685→      if (this.hudTimer <= 0) { this.hudTimer = 0.3; this.emitHud(); }
   686→   686→      return;
   687→   687→    }
   688→   688→
   689→   689→    if (this.phase === 'playing') {
   690→   690→      this.autoQuality(dt);
   691→   691→      // ciclo día/noche (los enemigos son más rápidos de noche)
   692→   692→      this.cycle.update(dt);
   693→   693→      this.ctx.nightFactor = this.cycle.nightFactor;
   694→   694→      if (this.noticeT > 0) { this.noticeT -= dt; if (this.noticeT <= 0) this.notice = ''; }
   695→   695→      // pasos del jugador sobre el terreno
   696→   696→      const stepDx = this.player.pos.x - this.lastStepPos.x;
   697→   697→      const stepDz = this.player.pos.z - this.lastStepPos.z;
   698→   698→      this.stepDist += Math.hypot(stepDx, stepDz);
   699→   699→      this.lastStepPos.copy(this.player.pos);
   700→   700→      if (this.stepDist > 2.2 && this.player.state !== 'roll' && this.player.state !== 'dead') {
   701→   701→        this.stepDist = 0;
   702→   702→        this.audio.footstep();
   703→   703→      }
   704→   704→      if (this.uiOpen) {
   705→   705→        // inventario abierto: mundo congelado, escena viva
   706→   706→        this.updateWorld(0.0001, true);
   707→   707→      } else if (this.hitStopT > 0) {
   708→   708→        // hit-stop y cámara lenta
   709→   709→        this.hitStopT -= dt;
   710→   710→        this.updateWorld(dt * 0.06);
   711→   711→      } else if (this.slowmoT > 0) {
   712→   712→        this.slowmoT -= dt;
   713→   713→        if (this.slowmoT <= 0) this.timeScale = 1;
   714→   714→        this.updateWorld(dt * this.timeScale);
   715→   715→      } else {
   716→   716→        this.updateWorld(dt);
   717→   717→      }
   718→   718→    } else if (this.phase === 'dead' || this.phase === 'victory') {
   719→   719→      // el mundo sigue con animaciones reducidas
   720→   720→      this.cycle.update(dt * 0.25);
   721→   721→      this.updateWorld(dt * 0.25, true);
   722→   722→    }
   723→   723→
   724→   724→    this.updateCamera(dt);
   725→   725→    this.updateEffects(dt);
   726→   726→    this.composer.render();
   727→   727→
   728→   728→    this.hudTimer -= dt;
   729→   729→    if (this.hudTimer <= 0) { this.hudTimer = 0.08; this.emitHud(); this.drawMinimap(); }
   730→   730→  };

   769→   769→  private updateWorld(dt: number, frozen = false) {
   770→   770→    this.elapsed += frozen ? 0 : dt;
   771→   771→
   772→   772→    // input → estado
   773→   773→    const k = this.keys;
   774→   774→    this.inputState.fwd = k.has('KeyW') || k.has('ArrowUp');
   775→   775→    this.inputState.back = k.has('KeyS') || k.has('ArrowDown');
   776→   776→    this.inputState.left = k.has('KeyA') || k.has('ArrowLeft');
   777→   777→    this.inputState.right = k.has('KeyD') || k.has('ArrowRight');
   778→   778→    this.inputState.sprint = k.has('ShiftLeft') || k.has('ShiftRight');
   779→   779→
   780→   780→    // jugador
   781→   781→    this.ctx.input = frozen ? this.frozenInput : this.inputState;
   782→   782→    this.ctx.camYaw = this.camYaw;
   783→   783→    this.player.update(dt, this.ctx);
   784→   784→    this.world.resolve(this.player.pos, this.player.radius);
   785→   785→    this.worldClamp(this.player.pos, this.player.radius);
   786→   786→
   787→   787→    // resolver golpe pendiente del jugador
   788→   788→    if (this.player.pendingStrike) {
   789→   789→      this.resolvePlayerStrike(this.player.pendingStrike);
   790→   790→      this.player.pendingStrike = null;
   791→   791→    }
   792→   792→
   793→   793→    // lock-on: limpiar si muere o se aleja
   794→   794→    if (this.lockEnemy && (!this.lockEnemy.alive || this.lockEnemy.pos.distanceTo(this.player.pos) > 38)) {
   795→   795→      this.lockEnemy = null;
   796→   796→      this.player.lockTarget = null;
   797→   797→    }
   798→   798→
   799→   799→    // enemigos
   800→   800→    for (const e of this.enemies) {
   801→   801→      if (e.removable) continue;
   802→   802→      e.update(dt, this.ctx);
   803→   803→      if (e.alive) {
   804→   804→        this.world.resolve(e.pos, e.radius);
   805→   805→        this.worldClamp(e.pos, e.radius);

   818→   818→          const push = (minD - d) / d * 0.5;

   820→   820→          b.pos.x += dx * push; b.pos.z += dz * push;

   978→   978→      // cámara de objetivo fijado

   987→   987→    const cp = Math.cos(this.camPitch), sp = Math.sin(this.camPitch);
   988→   988→    const off = new THREE.Vector3(Math.sin(this.camYaw) * cp, sp, Math.cos(this.camYaw) * cp);

   990→   990→    // hombro

   992→   992→    desiredPos.addScaledVector(right, 0.55);

   995→   995→    if (desiredPos.y < groundH) desiredPos.y = groundH;

  1005→  1005→      this.camera.position.z += rand(-1, 1) * sh * 0.18;

  1017→  1017→      this.camera.fov = damp(this.camera.fov, targetFov, 6, dt);

  1019→  1019→    }
  1020→  1020→  }

  1235→  1235→    if (e.code === 'Tab' || e.code === 'Space') e.preventDefault();
  1236→  1236→    // con la mochila abierta solo se permite cerrarla
  1237→  1237→    if (this.uiOpen) {
  1238→  1238→      if (e.code === 'Escape' || e.code === 'KeyI' || e.code === 'KeyB') this.toggleInventory();

  1240→  1240→    }
  1241→  1241→    if (this.phase !== 'playing') return;
  1242→  1242→    this.keys.add(e.code);
  1243→  1243→    switch (e.code) {
  1244→  1244→      case 'Tab': this.toggleLockOn(); break;
  1245→  1245→      case 'Space': this.queued.roll = true; break;
  1246→  1246→      case 'KeyF': this.queued.potion = true; break;
  1247→  1247→      case 'KeyE': this.doInteract(); break;

  1249→  1249→    }
