   206→   206→  private spawnRoamer(initial = false) {
   207→   207→    if (this.enemies.filter(e => e.alive && e.guardianOf === -1 && !e.isBoss).length >= ROAMER_TARGET + 4) return;

   213→   213→      const dBonfire = Math.hypot(x - WORLD.bonfire.x, z - WORLD.bonfire.z);

   215→   215→      if (dPlayer > (initial ? 20 : 26) && dBonfire > 16 && !nearShrine) {
   216→   216→        // tipos según nivel

   224→   224→    }

   303→   303→  private updateInteract() {
   304→   304→    this.interactTarget = null;
   305→   305→    const p = this.player.pos;
   306→   306→    const dB = Math.hypot(p.x - WORLD.bonfire.x, p.z - WORLD.bonfire.z);
   307→   307→    if (dB < 3.4) {
   308→   308→      this.interactTarget = { kind: 'bonfire' };
   309→   309→      return;
   310→   310→    }
   311→   311→    for (const sh of this.world.shrines) {
   312→   312→      if (sh.cleansed) continue;
   313→   313→      const guardians = this.enemies.filter(e => e.guardianOf === sh.idx && e.alive);
   314→   314→      const d = Math.hypot(p.x - sh.pos.x, p.z - sh.pos.z);
   315→   315→      if (d < 5 && guardians.length === 0) {
   316→   316→        this.interactTarget = { kind: 'shrine', idx: sh.idx };
   317→   317→        return;
   318→   318→      }
   319→   319→    }
   320→   320→    if (this.world.sigil && this.world.sigil.group.visible && this.bossDefeated) {
   321→   321→      const A = WORLD.arena;
   322→   322→      const d = Math.hypot(p.x - A.x, p.z - A.z);
   323→   323→      if (d < 4.5) this.interactTarget = { kind: 'sigil' };
   324→   324→    }
   325→   325→  }

   327→   327→  private doInteract() {
   328→   328→    const t = this.interactTarget;
   329→   329→    if (!t) return;
   330→   330→    if (t.kind === 'bonfire') {
   331→   331→      const p = this.player;
   332→   332→      p.hp = p.maxHp; p.stamina = p.maxStamina; p.potions = p.maxPotions;
   333→   333→      this.audio.bonfireRest();
   334→   334→      this.particles.burst({
   335→   335→        x: p.pos.x, y: p.pos.y + 0.6, z: p.pos.z,
   336→   336→        count: 26, speed: 2.6, color: 0xffb347, size: 0.3, life: 1.1, gravity: -1.5, drag: 1.5, glow: 2,
   337→   337→      });
   338→   338→      this.addDamageNumber(new THREE.Vector3(p.pos.x, p.pos.y + 2.3, p.pos.z), 'REPOSO', '#ffb347');
   339→   339→    } else if (t.kind === 'shrine' && t.idx !== undefined) {
   340→   340→      const sh = this.world.shrines[t.idx];
   341→   341→      this.world.cleanseShrine(t.idx);
   342→   342→      this.shrinesCleansed++;
   343→   343→      this.audio.cleanse();
   344→   344→      this.player.gainXp(60, this.ctx);
   345→   345→      this.player.gold += 40;
   346→   346→      this.particles.burst({
   347→   347→        x: sh.pos.x, y: sh.pos.y + 4, z: sh.pos.z,
   348→   348→        count: 70, speed: 7, color: 0x37d8c8, size: 0.42, life: 1.8, gravity: -0.5, drag: 0.9, glow: 2.4,
   349→   349→      });
   350→   350→      this.ctx.shake(0.5);
   351→   351→      if (this.shrinesCleansed >= 3 && !this.bossDefeated && !this.bossActive) {
   352→   352→        this.awakenTimer = 2.5;
   353→   353→      }
   354→   354→      this.emitHud();
   355→   355→    } else if (t.kind === 'sigil') {
   356→   356→      this.world.sigilReady();
   357→   357→      this.spawnBoss(1.35 ** this.bossKills());
   358→   358→      this.addDamageNumber(new THREE.Vector3(this.player.pos.x, this.player.pos.y + 2.3, this.player.pos.z), 'EL JEFE DESPIERTA DE NUEVO', '#ff5a4e', true);
   359→   359→    }
   360→   360→  }

   368→   368→  private toggleLockOn() {
   369→   369→    if (this.lockEnemy) { this.lockEnemy = null; this.player.lockTarget = null; return; }
   370→   370→    let best: Enemy | null = null, bestD = 34;
   371→   371→    for (const e of this.enemies) {
   372→   372→      if (!e.alive || e.state === 'spawn') continue;
   373→   373→      const d = e.pos.distanceTo(this.player.pos);
   374→   374→      if (d < bestD) { bestD = d; best = e; }
   375→   375→    }
   376→   376→    this.lockEnemy = best;
   377→   377→    this.player.lockTarget = best;
   378→   378→  }

   381→   381→
   382→   382→  start() {
   383→   383→    if (this.running) return;
   384→   384→    this.running = true;
   385→   385→    this.lastT = performance.now();
   386→   386→    this.loop(this.lastT);
   387→   387→  }
   388→   388→
   389→   389→  private loop = (t: number) => {
   390→   390→    if (!this.running) return;
   391→   391→    this.raf = requestAnimationFrame(this.loop);
   392→   392→    let dt = Math.min(0.05, (t - this.lastT) / 1000);
   393→   393→    this.lastT = t;
   394→   394→    this.fps = this.fps * 0.95 + (1 / Math.max(0.0001, dt)) * 0.05;
   395→   395→
   396→   396→    if (this.phase === 'playing') {
   397→   397→      // hit-stop y cámara lenta
   398→   398→      if (this.hitStopT > 0) {
   399→   399→        this.hitStopT -= dt;
   400→   400→        this.updateWorld(dt * 0.06);
   401→   401→      } else if (this.slowmoT > 0) {
   402→   402→        this.slowmoT -= dt;
   403→   403→        if (this.slowmoT <= 0) this.timeScale = 1;
   404→   404→        this.updateWorld(dt * this.timeScale);
   405→   405→      } else {
   406→   406→        this.updateWorld(dt);
   407→   407→      }
   408→   408→    } else if (this.phase === 'dead' || this.phase === 'victory') {
   409→   409→      // el mundo sigue con animaciones reducidas
   410→   410→      this.updateWorld(dt * 0.25, true);
   411→   411→    }
   412→   412→
   413→   413→    this.updateCamera(dt);
   414→   414→    this.updateEffects(dt);
   415→   415→    this.composer.render();
   416→   416→
   417→   417→    this.hudTimer -= dt;
   418→   418→    if (this.hudTimer <= 0) { this.hudTimer = 0.08; this.emitHud(); this.drawMinimap(); }
   419→   419→  };

   421→   421→  private updateWorld(dt: number, frozen = false) {
   422→   422→    this.elapsed += frozen ? 0 : dt;
   423→   423→
   424→   424→    // input → estado
   425→   425→    const k = this.keys;
   426→   426→    this.inputState.fwd = k.has('KeyW') || k.has('ArrowUp');
   427→   427→    this.inputState.back = k.has('KeyS') || k.has('ArrowDown');
   428→   428→    this.inputState.left = k.has('KeyA') || k.has('ArrowLeft');
   429→   429→    this.inputState.right = k.has('KeyD') || k.has('ArrowRight');
   430→   430→    this.inputState.sprint = k.has('ShiftLeft') || k.has('ShiftRight');
   431→   431→
   432→   432→    // jugador
   433→   433→    this.ctx.input = frozen ? this.frozenInput : this.inputState;
   434→   434→    this.ctx.camYaw = this.camYaw;
   435→   435→    this.player.update(dt, this.ctx);
   436→   436→    this.world.resolve(this.player.pos, this.player.radius);
   437→   437→    this.worldClamp(this.player.pos, this.player.radius);
   438→   438→
   439→   439→    // resolver golpe pendiente del jugador
   440→   440→    if (this.player.pendingStrike) {
   441→   441→      this.resolvePlayerStrike(this.player.pendingStrike);
   442→   442→      this.player.pendingStrike = null;
   443→   443→    }
   444→   444→
   445→   445→    // lock-on: limpiar si muere o se aleja
   446→   446→    if (this.lockEnemy && (!this.lockEnemy.alive || this.lockEnemy.pos.distanceTo(this.player.pos) > 38)) {
   447→   447→      this.lockEnemy = null;
   448→   448→      this.player.lockTarget = null;
   449→   449→    }
   450→   450→
   451→   451→    // enemigos
   452→   452→    for (const e of this.enemies) {
   453→   453→      if (e.removable) continue;
   454→   454→      e.update(dt, this.ctx);
   455→   455→      if (e.alive) {
   456→   456→        this.world.resolve(e.pos, e.radius);
   457→   457→        this.worldClamp(e.pos, e.radius);
   458→   458→      }
   459→   459→    }
   460→   460→    // separación entre enemigos
   461→   461→    const live = this.enemies.filter(e => e.alive && e.state !== 'spawn');
   462→   462→    for (let i = 0; i < live.length; i++) {
   463→   463→      for (let j = i + 1; j < live.length; j++) {
   464→   464→        const a = live[i], b = live[j];
   465→   465→        const dx = b.pos.x - a.pos.x, dz = b.pos.z - a.pos.z;
   466→   466→        const minD = a.radius + b.radius;
   467→   467→        const d2 = dx * dx + dz * dz;
   468→   468→        if (d2 < minD * minD && d2 > 1e-5) {
   469→   469→          const d = Math.sqrt(d2);
   470→   470→          const push = (minD - d) / d * 0.5;
   471→   471→          a.pos.x -= dx * push; a.pos.z -= dz * push;
   472→   472→          b.pos.x += dx * push; b.pos.z += dz * push;
   473→   473→        }
   474→   474→      }
   475→   475→      // el jugador empuja y es empujado ligeramente
   476→   476→      const p = this.player;
   477→   477→      const dx = p.pos.x - live[i].pos.x, dz = p.pos.z - live[i].pos.z;
   478→   478→      const minD = live[i].radius + p.radius;
   479→   479→      const d2 = dx * dx + dz * dz;
   480→   480→      if (d2 < minD * minD && d2 > 1e-5) {
   481→   481→        const d = Math.sqrt(d2);
   482→   482→        const push = (minD - d) / d * 0.6;
   483→   483→        live[i].pos.x -= dx * push; live[i].pos.z -= dz * push;
   484→   484→        p.pos.x += dx * push * 0.3; p.pos.z += dz * push * 0.3;
   485→   485→      }
   486→   486→    }
   487→   487→    // limpiar cadáveres
   488→   488→    if (this.enemies.some(e => e.removable)) {
   489→   489→      for (const e of this.enemies) {
   490→   490→        if (e.removable) { this.scene.remove(e.root); }
   491→   491→      }
   492→   492→      this.enemies = this.enemies.filter(e => !e.removable);
   493→   493→    }
   494→   494→
   495→   495→    // proyectiles y drops
   496→   496→    for (const pr of this.projectiles) pr.update(dt, this.ctx);
   497→   497→    this.projectiles = this.projectiles.filter(pr => {
   498→   498→      if (pr.dead) { this.scene.remove(pr.root); return false; }
   499→   499→      return true;
   500→   500→    });
   501→   501→    for (const pk of this.pickups) {
   502→   502→      pk.update(dt, this.ctx);
   503→   503→      this.world.resolve(pk.pos, 0.3);
   504→   504→    }
   505→   505→    this.pickups = this.pickups.filter(pk => {
   506→   506→      if (pk.dead) { this.scene.remove(pk.root); return false; }
   507→   507→      return true;
   508→   508→    });
   509→   509→
   510→   510→    // mundo visual
   511→   511→    this.world.update(dt, this.camera, (x, y, z) => {
   512→   512→      this.particles.spawn({
   513→   513→        x, y, z, vx: rand(-0.3, 0.3), vy: rand(1, 2.2), vz: rand(-0.3, 0.3),
   514→   514→        color: 0xff7a2a, size: rand(0.1, 0.2), life: rand(0.8, 1.6), glow: 2.2, drag: 0.5,
   515→   515→      });
   516→   516→    });
   517→   517→
   518→   518→    if (frozen) return;
   519→   519→
   520→   520→    // estela de espada
   521→   521→    if (this.player.state === 'attack') {
   522→   522→      const b = new THREE.Vector3(), ti = new THREE.Vector3();
   523→   523→      this.player.swordPoints(b, ti);
   524→   524→      this.trail.emit(b, ti);
   525→   525→    }
   526→   526→    this.trail.update(dt, this.player.state === 'attack');
   527→   527→
   528→   528→    // despawn/respawn de errantes
   529→   529→    this.roamerTimer -= dt;
   530→   530→    if (this.roamerTimer <= 0) {
   531→   531→      this.roamerTimer = rand(3.5, 6);
   532→   532→      const aliveRoamers = this.enemies.filter(e => e.alive && e.guardianOf === -1 && !e.isBoss).length;
   533→   533→      if (aliveRoamers < ROAMER_TARGET) this.spawnRoamer();
   534→   534→    }
   535→   535→
   536→   536→    // despertar del jefe
   537→   537→    if (this.awakenTimer > 0) {
   538→   538→      this.awakenTimer -= dt;
   539→   539→      if (this.awakenTimer <= 0) this.spawnBoss();
   540→   540→    }
   541→   541→    // reaparición del jefe tras morir el jugador en plena lucha
   542→   542→    if (this.bossRespawnT > 0) {
   543→   543→      this.bossRespawnT -= dt;
   544→   544→      if (this.bossRespawnT <= 0 && this.shrinesCleansed >= 3) this.spawnBoss();
   545→   545→    }
   546→   546→
   547→   547→    // interacción
   548→   548→    this.updateInteract();
   549→   549→
   550→   550→    // música: intensidad
   551→   551→    if (this.elapsed % 1 < dt) {
   552→   552→      let intensity = 0.12;
   553→   553→      if (this.bossActive) intensity = 1;
   554→   554→      else {
   555→   555→        const near = this.enemies.some(e => e.alive && e.aggroed && e.pos.distanceTo(this.player.pos) < 30);
   556→   556→        intensity = near ? 0.75 : 0.12;
   557→   557→      }
   558→   558→      this.audio.setIntensity(intensity);
   559→   559→    }
   560→   560→
   561→   561→    // victoria
   562→   562→    if (this.victoryDelay > 0) {
   563→   563→      this.victoryDelay -= dt;
   564→   564→      if (this.victoryDelay <= 0) {
   565→   565→        this.setPhase('victory');
   566→   566→      }
   567→   567→    }
   568→   568→  }

   571→   571→  private frozenInput: InputState = {

   577→   577→  private resolvePlayerStrike(def: AttackDef) {
   578→   578→    const p = this.player;
   579→   579→    const heavy = def === PLAYER_HEAVY;
   580→   580→    let hitAny = false;
   581→   581→    for (const e of this.enemies) {
   582→   582→      if (!e.alive || e.state === 'spawn') continue;
   583→   583→      const to = new THREE.Vector3().subVectors(e.pos, p.pos).setY(0);
   584→   584→      const d = to.length() - e.radius;
   585→   585→      if (d > def.range) continue;
   586→   586→      const ang = Math.atan2(to.x, to.z);
   587→   587→      let diff = Math.abs(((ang - p.yaw + Math.PI * 3) % (Math.PI * 2)) - Math.PI);
   588→   588→      if (diff <= def.arc / 2) {
   589→   589→        const dmg = def.dmg * p.dmgMul;
   590→   590→        e.takeDamage(dmg, p.pos, this.ctx, heavy);
   591→   591→        hitAny = true;
   592→   592→        this.particles.burst({
   593→   593→          x: e.pos.x - to.x * 0.3, y: e.pos.y + e.rig.height * 0.55, z: e.pos.z - to.z * 0.3,
   594→   594→          count: heavy ? 14 : 8, speed: 5, color: 0xffd9a0, size: 0.2, life: 0.35, drag: 2, glow: 2.4, gravity: 4,
   595→   595→        });
   596→   596→      }
   597→   597→    }
   598→   598→    if (hitAny) {
   599→   599→      this.hitStopT = Math.max(this.hitStopT, heavy ? 0.11 : 0.07);
   600→   600→      this.shakeAmt = Math.min(1.2, this.shakeAmt + (heavy ? 0.4 : 0.16));
   601→   601→      if (heavy) this.audio.heavyHit();
   602→   602→      else this.audio.hitMetal();
   603→   603→    }
   604→   604→  }

   606→   606→  private worldClamp(pos: THREE.Vector3, radius: number) {
   607→   607→    const r = Math.hypot(pos.x, pos.z);
   608→   608→    const max = WORLD.radius - 2 - radius;
   609→   609→    if (r > max) {
   610→   610→      pos.x *= max / r;
   611→   611→      pos.z *= max / r;
   612→   612→    }
   613→   613→  }

   617→   617→  private updateCamera(dt: number) {
   618→   618→    // ratón → rotación
   619→   619→    if (this.locked && this.phase === 'playing') {
   620→   620→      this.camYaw -= this.mouseDX * 0.0023;
   621→   621→      this.camPitch += this.mouseDY * 0.0018;
   622→   622→      this.camPitch = clamp(this.camPitch, -0.25, 1.15);
   623→   623→    }
   624→   624→    this.mouseDX = 0; this.mouseDY = 0;
   625→   625→
   626→   626→    const target = this.player.pos;
   627→   627→    const head = new THREE.Vector3(target.x, target.y + 1.55, target.z);
   628→   628→
   629→   629→    if (this.lockEnemy && this.lockEnemy.alive) {
   630→   630→      // cámara de objetivo fijado
   631→   631→      const desired = Math.atan2(target.x - this.lockEnemy.pos.x, target.z - this.lockEnemy.pos.z);
   632→   632→      this.camYaw = dampAngle(this.camYaw, desired, 3.2, dt);
   633→   633→      this.camPitch = damp(this.camPitch, 0.3, 3, dt);
   634→   634→    }
   635→   635→
   636→   636→    const dist = this.camDist;
   637→   637→    const cp = Math.cos(this.camPitch), sp = Math.sin(this.camPitch);
   638→   638→    const off = new THREE.Vector3(Math.sin(this.camYaw) * cp, sp, Math.cos(this.camYaw) * cp);
   639→   639→    const desiredPos = head.clone().addScaledVector(off, dist);
   640→   640→    // hombro
   641→   641→    const right = new THREE.Vector3(-Math.cos(this.camYaw), 0, Math.sin(this.camYaw));
   642→   642→    desiredPos.addScaledVector(right, 0.55);
   643→   643→    // colisión con terreno
   644→   644→    const groundH = terrainHeight(desiredPos.x, desiredPos.z) + 0.5;
   645→   645→    if (desiredPos.y < groundH) desiredPos.y = groundH;
   646→   646→
   647→   647→    this.camPos.lerp(desiredPos, 1 - Math.exp(-(this.lockEnemy ? 9 : 12) * dt));
   648→   648→
   649→   649→    // sacudida
   650→   650→    const sh = this.shakeAmt;
   651→   651→    this.camera.position.copy(this.camPos);
   652→   652→    if (sh > 0.001) {
   653→   653→      this.camera.position.x += rand(-1, 1) * sh * 0.18;
   654→   654→      this.camera.position.y += rand(-1, 1) * sh * 0.18;
   655→   655→      this.camera.position.z += rand(-1, 1) * sh * 0.18;
   656→   656→      this.shakeAmt = Math.max(0, sh - dt * 2.4);
   657→   657→    }
   658→   658→    const lookAt = head.clone();
   659→   659→    if (this.lockEnemy && this.lockEnemy.alive) {
   660→   660→      lookAt.lerp(new THREE.Vector3(this.lockEnemy.pos.x, this.lockEnemy.pos.y + 1.2, this.lockEnemy.pos.z), 0.22);
   661→   661→    }
   662→   662→    this.camera.lookAt(lookAt);
   663→   663→
   664→   664→    // FOV dinámico al esprintar
   665→   665→    const targetFov = this.baseFov + (this.player.sprinting ? 5 : 0);
   666→   666→    if (Math.abs(this.camera.fov - targetFov) > 0.05) {
   667→   667→      this.camera.fov = damp(this.camera.fov, targetFov, 6, dt);
   668→   668→      this.camera.updateProjectionMatrix();
   669→   669→    }
   670→   670→  }

   672→   672→  private updateEffects(dt: number) {
   673→   673→    this.particles.update(dt, this.camera.position);
   674→   674→    // números de daño
   675→   675→    const w = window.innerWidth, h = window.innerHeight;
   676→   676→    const v = new THREE.Vector3();
   677→   677→    for (const d of this.dmgPool) {
   678→   678→      if (!d.active) continue;

   681→   681→      d.worldPos.y += d.vy * dt;
   682→   682→      d.vy = Math.max(0.2, d.vy - dt * 1.4);
   683→   683→      v.copy(d.worldPos).project(this.camera);
   684→   684→      if (v.z > 1) { d.el.style.display = 'none'; continue; }
   685→   685→      d.el.style.display = 'block';
   686→   686→      d.el.style.left = `${(v.x * 0.5 + 0.5) * w}px`;
   687→   687→      d.el.style.top = `${(-v.y * 0.5 + 0.5) * h}px`;
   688→   688→      d.el.style.opacity = `${Math.min(1, d.life * 2.5)}`;
   689→   689→    }
   690→   690→    // viñeta de daño
   691→   691→    this.hurtFlash = Math.max(0, this.hurtFlash - dt * 2.2);

   695→   695→  }

   841→   841→  private bindEvents() {
   842→   842→    window.addEventListener('resize', this.onResize);
   843→   843→    window.addEventListener('keydown', this.onKeyDown);
   844→   844→    window.addEventListener('keyup', this.onKeyUp);
   845→   845→    document.addEventListener('mousemove', this.onMouseMove);
   846→   846→    document.addEventListener('pointerlockchange', this.onLockChange);
   847→   847→    const c = this.renderer.domElement;
   848→   848→    c.addEventListener('mousedown', this.onMouseDown);
   849→   849→    c.addEventListener('contextmenu', this.onContextMenu);
   850→   850→  }
   851→   851→
   852→   852→  private onResize = () => {
   853→   853→    this.camera.aspect = window.innerWidth / window.innerHeight;
   854→   854→    this.camera.updateProjectionMatrix();
   855→   855→    this.renderer.setSize(window.innerWidth, window.innerHeight);
   856→   856→    this.composer.setSize(window.innerWidth, window.innerHeight);
   857→   857→  };
   858→   858→
   859→   859→  private onKeyDown = (e: KeyboardEvent) => {
   860→   860→    if (e.code === 'Tab' || e.code === 'Space') e.preventDefault();
   861→   861→    if (this.phase !== 'playing') return;
   862→   862→    this.keys.add(e.code);
   863→   863→    switch (e.code) {
   864→   864→      case 'Tab': this.toggleLockOn(); break;
   865→   865→      case 'Space': this.queued.roll = true; break;
   866→   866→      case 'KeyF': this.queued.potion = true; break;
   867→   867→      case 'KeyE': this.doInteract(); break;
   868→   868→    }
   869→   869→  };
   870→   870→
   871→   871→  private onKeyUp = (e: KeyboardEvent) => {
   872→   872→    this.keys.delete(e.code);
   873→   873→  };
   874→   874→
   875→   875→  private onMouseMove = (e: MouseEvent) => {
   876→   876→    if (!this.locked) return;
   877→   877→    this.mouseDX += e.movementX;
   878→   878→    this.mouseDY += e.movementY;
   879→   879→  };
   880→   880→
   881→   881→  private onMouseDown = (e: MouseEvent) => {
   882→   882→    if (this.phase !== 'playing') return;
   883→   883→    if (!this.locked) { this.requestLock(); return; }
   884→   884→    if (e.button === 0) this.queued.attack = true;
   885→   885→    if (e.button === 2) this.queued.heavy = true;
   886→   886→  };
   887→   887→
   888→   888→  private onContextMenu = (e: Event) => { e.preventDefault(); };
   889→   889→
   890→   890→  private onLockChange = () => {
   891→   891→    this.locked = document.pointerLockElement === this.renderer.domElement;
   892→   892→    if (!this.locked && this.phase === 'playing') this.pause();
   893→   893→  };

   897→   897→  dispose() {
   898→   898→    this.running = false;
   899→   899→    cancelAnimationFrame(this.raf);
   900→   900→    window.removeEventListener('resize', this.onResize);
   901→   901→    window.removeEventListener('keydown', this.onKeyDown);
   902→   902→    window.removeEventListener('keyup', this.onKeyUp);
   903→   903→    document.removeEventListener('mousemove', this.onMouseMove);
   904→   904→    document.removeEventListener('pointerlockchange', this.onLockChange);
   905→   905→    this.audio.dispose();
   906→   906→    this.renderer.dispose();
   907→   907→    this.renderer.domElement.remove();
   908→   908→    this.dmgLayer.remove();
   909→   909→  }
