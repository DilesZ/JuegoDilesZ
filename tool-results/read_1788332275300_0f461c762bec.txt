    55→export abstract class Entity {
    56→  root = new THREE.Group();
    57→  pos = new THREE.Vector3();
    58→  yaw = 0;
    59→  hp = 1; maxHp = 1;
    60→  radius = 0.5;
    61→  height = 1.8;
    62→  alive = true;
    63→  removable = false;
    64→  knock = new THREE.Vector3();
    65→  protected mats: CharMat[] = [];
    66→  protected flashT = 0;
    67→
    68→  protected collectMats() {
    69→    const set = new Set<CharMat>();
    70→    this.root.traverse(o => {
    71→      if (o instanceof THREE.Mesh) {
    72→        const m = o.material as CharMat;
    73→        if (m && ((m as THREE.MeshStandardMaterial).isMeshStandardMaterial || (m as THREE.MeshToonMaterial).isMeshToonMaterial)) {
    74→          set.add(m);
    75→        }
    76→      }
    77→    });
    78→    this.mats = [...set];
    79→    this.originals = this.mats.map(m => ({ e: m.emissive.clone(), i: m.emissiveIntensity }));
    80→  }
    81→  private originals: { e: THREE.Color; i: number }[] = [];
    82→
    83→  flash(intensity = 2.2, time = 0.12) {
    84→    this.flashT = time;
    85→    for (let i = 0; i < this.mats.length; i++) {
    86→      this.mats[i].emissive.setRGB(1, 0.9, 0.8);
    87→      this.mats[i].emissiveIntensity = intensity;
    88→    }
    89→  }
    90→  protected updateFlash(dt: number) {
    91→    if (this.flashT > 0) {
    92→      this.flashT -= dt;
    93→      if (this.flashT <= 0) {
    94→        for (let i = 0; i < this.mats.length; i++) {
    95→          this.mats[i].emissive.copy(this.originals[i].e);
    96→          this.mats[i].emissiveIntensity = this.originals[i].i;
    97→        }
    98→      }
    99→    }
   100→  }
   101→  abstract update(dt: number, ctx: GameCtx): void;
   102→}

   104→/* ---------- Estela de espada (cinta aditiva, buffer de anillo sin allocations) ---------- */
   105→
   106→export class SwordTrail {
   107→  private segs = 16;
   108→  private geo: THREE.BufferGeometry;
   109→  private posAttr: Float32Array;
   110→  private alphaAttr: Float32Array;
   111→  mesh: THREE.Mesh;
   112→  private strength = 0;
   113→  /** anillo preasignado: cada segmento reutiliza sus dos Vector3 (cero GC) */
   114→  private ring: { b: THREE.Vector3; t: THREE.Vector3 }[];
   115→  private head = 0;   // siguiente índice a escribir
   116→  private count = 0;  // puntos vivos (máx segs)
   117→
   118→  constructor(scene: THREE.Scene, color = 0xffd9a0) {
   119→    this.geo = new THREE.BufferGeometry();
   120→    this.posAttr = new Float32Array(this.segs * 2 * 3);
   121→    this.alphaAttr = new Float32Array(this.segs * 2);
   122→    this.geo.setAttribute('position', new THREE.BufferAttribute(this.posAttr, 3));
   123→    this.geo.setAttribute('aAlpha', new THREE.BufferAttribute(this.alphaAttr, 1));
   124→    const idx: number[] = [];
   125→    for (let i = 0; i < this.segs - 1; i++) {
   126→      const a = i * 2, b = i * 2 + 1, c = (i + 1) * 2, d = (i + 1) * 2 + 1;
   127→      idx.push(a, b, c, b, d, c);
   128→    }
   129→    this.geo.setIndex(idx);
   130→    const mat = new THREE.ShaderMaterial({
   131→      uniforms: { uColor: { value: new THREE.Color(color) } },
   132→      vertexShader: `attribute float aAlpha; varying float vA;
   133→        void main(){ vA = aAlpha; gl_Position = projectionMatrix * modelViewMatrix * vec4(position,1.0); }`,
   134→      fragmentShader: `uniform vec3 uColor; varying float vA;
   135→        void main(){ if (vA <= 0.003) discard; gl_FragColor = vec4(uColor * (0.6 + vA), vA * 0.85); }`,
   136→      transparent: true, depthWrite: false, side: THREE.DoubleSide, blending: THREE.AdditiveBlending,
   137→    });
   138→    this.mesh = new THREE.Mesh(this.geo, mat);
   139→    this.mesh.frustumCulled = false;
   140→    this.mesh.renderOrder = 9;
   141→    this.ring = [];
   142→    for (let i = 0; i < this.segs; i++) {
   143→      this.ring.push({ b: new THREE.Vector3(0, -999, 0), t: new THREE.Vector3(0, -999, 0) });
   144→    }
   145→    scene.add(this.mesh);
   146→  }
   147→
   148→  /** Registra un punto de la estela reutilizando la memoria del anillo */
   149→  emit(base: THREE.Vector3, tip: THREE.Vector3) {
   150→    const slot = this.ring[this.head];
   151→    slot.b.copy(base);
   152→    slot.t.copy(tip);
   153→    this.head = (this.head + 1) % this.segs;
   154→    if (this.count < this.segs) this.count++;
   155→    this.strength = 1;
   156→  }
   157→
   158→  update(dt: number, active: boolean) {
   159→    if (!active) {
   160→      this.strength -= dt * 4;
   161→      if (this.strength < -1) this.strength = -1;
   162→      // drena el anillo suavemente al dejar de atacar
   163→      if (this.strength <= 0 && this.count > 0) this.count--;
   164→    }
   165→    const n = this.count;
   166→    if (n === 0 && this.strength <= 0) {
   167→      // nada que dibujar: atributos ya limpios, evita subidas a GPU cada frame
   168→      if (this.posAttr[1] !== -999) {
   169→        for (let i = 0; i < this.segs * 2; i++) this.posAttr[i * 3 + 1] = -999;
   170→        (this.geo.getAttribute('position') as THREE.BufferAttribute).needsUpdate = true;
   171→      }
   172→      return;
   173→    }
   174→    // el más reciente es head-1; se recorre hacia atrás en orden de edad
   175→    for (let i = 0; i < this.segs; i++) {
   176→      let src: { b: THREE.Vector3; t: THREE.Vector3 } | undefined;
   177→      if (i < n) {
   178→        const idx = (this.head - 1 - i + this.segs * 2) % this.segs;
   179→        src = this.ring[idx];
   180→      }
   181→      const o = i * 6;
   182→      if (src) {
   183→        this.posAttr[o] = src.b.x; this.posAttr[o + 1] = src.b.y; this.posAttr[o + 2] = src.b.z;
   184→        this.posAttr[o + 3] = src.t.x; this.posAttr[o + 4] = src.t.y; this.posAttr[o + 5] = src.t.z;
   185→      } else {
   186→        this.posAttr[o + 1] = -999; this.posAttr[o + 4] = -999;
   187→      }
   188→      // el más reciente (i=0) con alpha alto, la cola se funde
   189→      const a = n > 1 ? (1 - i / n) : 1;
   190→      const fade = this.strength > 0 ? 1 : Math.max(0, 1 + this.strength);
   191→      const alpha = a * a * Math.max(0, fade) * 0.9;
   192→      this.alphaAttr[i * 2] = alpha;
   193→      this.alphaAttr[i * 2 + 1] = alpha;
   194→    }
   195→    (this.geo.getAttribute('position') as THREE.BufferAttribute).needsUpdate = true;
   196→    (this.geo.getAttribute('aAlpha') as THREE.BufferAttribute).needsUpdate = true;
   197→  }
   198→}

   366→  /** Reproduce un clip de acción UNA vez sincronizado con la duración de jugabilidad */

   379→  tryAttack(heavy: boolean, ctx: GameCtx, fwd: THREE.Vector3): boolean {
   380→    let def: AttackDef;
   381→    if (heavy) {
   382→      if (this.stamina < PLAYER_HEAVY.stam * 0.35) return false;
   383→      def = PLAYER_HEAVY;
   384→      this.comboNext = 0; // el mandoble cargado reinicia la cadena
   385→    } else {
   386→      if (this.stamina <= 0) return false;
   387→      def = PLAYER_ATTACKS[this.comboNext] ?? PLAYER_ATTACKS[0];
   388→      if (this.stamina < def.stam * 0.5) return false;
   389→      this.comboNext = def.kind === 'finisher' ? 0 : Math.min(this.comboNext + 1, PLAYER_ATTACKS.length - 1);
   390→    }
   391→    this.stamina = Math.max(0, this.stamina - def.stam);
   392→    this.staminaDelay = 0.75;
   393→    this.state = 'attack';
   394→    this.stateT = 0;
   395→    this.heavy = def.kind !== 'light';
   396→    this.attackIdx = PLAYER_ATTACKS.indexOf(def) < 0 ? 0 : PLAYER_ATTACKS.indexOf(def);
   397→    this.currentAttack = def;
   398→    this.didHit = false;
   399→    this.buffered = false;
   400→    this.bufferedHeavy = false;
   401→    // impulso inicial (sin allocations)
   402→    this.knock.add(_tmpA.copy(fwd).multiplyScalar(def.impulse * 0.5));
   403→    // la animación se dispara UNA vez por golpe (antes se reiniciaba cada frame)
   404→    this.playAction(def.clip, def.dur);
   405→    ctx.audio.swing(def.soundPitch);
   406→    return true;
   407→  }

   409→  tryRoll(ctx: GameCtx, dir: THREE.Vector3): boolean {
   410→    if (this.stamina <= 0) return false;
   411→    if (this.state === 'roll' || this.state === 'dead') return false;
   412→    if (this.state === 'attack' && this.stateT < this.currentAttack!.dur * 0.5) return false;
   413→    this.stamina = Math.max(0, this.stamina - 22);
   414→    this.staminaDelay = 0.75;
   415→    this.state = 'roll'; this.stateT = 0;
   416→    this.buffered = false; this.bufferedHeavy = false;
   417→    this.rollDir.copy(dir.lengthSq() > 0.01 ? dir : _tmpA.set(Math.sin(this.yaw), 0, Math.cos(this.yaw)));
   418→    this.iFrames = 0.42;
   419→    this.playAction('roll', CLIPS.roll.dur, 0.05);
   420→    ctx.audio.roll();
   421→    ctx.particles.burst({
   422→      x: this.pos.x, y: this.pos.y + 0.15, z: this.pos.z,
   423→      count: 10, speed: 2.2, color: 0x8a7a5e, size: 0.3, life: 0.55, gravity: 1.5, drag: 2.4, glow: 0.7,
   424→    });
   425→    return true;
   426→  }

   428→  takeDamage(dmg: number, srcPos: THREE.Vector3, ctx: GameCtx): boolean {
   429→    if (this.iFrames > 0 || this.state === 'dead') return false;
   430→    // la defensa de la armadura reduce el daño entrante
   431→    const final = dmg * (1 - this.damageReduction);
   432→    this.hp -= final;
   433→    this.iFrames = 0.28;
   434→    ctx.audio.hurt();
   435→    ctx.playerHurt();
   436→    ctx.addDamageNumber(new THREE.Vector3(this.pos.x + rand(-0.3, 0.3), this.pos.y + 2.1, this.pos.z), `-${Math.round(final)}`, '#ff5a4e');
   437→    ctx.particles.burst({
   438→      x: this.pos.x, y: this.pos.y + 1.2, z: this.pos.z,
   439→      count: 14, speed: 3.2, color: 0xd8323c, size: 0.24, life: 0.7, gravity: 5, drag: 1.5,
   440→    });
   441→    // knockback (sin allocations)
   442→    const d = _tmpB.copy(this.pos).sub(srcPos).setY(0);
   443→    if (d.lengthSq() > 0.001) d.normalize();
   444→    this.knock.add(d.multiplyScalar(4));
   445→    if (this.hp <= 0) {
   446→      this.hp = 0;
   447→      this.state = 'dead'; this.stateT = 0;
   448→      this.alive = false;
   449→      this.buffered = false; this.bufferedHeavy = false;
   450→      this.playAction('death', CLIPS.death.dur, 0.12);
   451→      ctx.audio.die();
   452→      ctx.shake(0.8);
   453→      return true;
   454→    }
   455→    if (this.state !== 'roll') {
   456→      this.state = 'hurt'; this.stateT = 0;
   457→      this.buffered = false; this.bufferedHeavy = false;
   458→      this.playAction('hurt', CLIPS.hurt.dur, 0.08);
   459→    }
   460→    ctx.shake(0.45);
   461→    return true;
   462→  }
   463→
   464→  update(dt: number, ctx: GameCtx) {
   465→    const input = ctx.input;
   466→    const camYaw = ctx.camYaw;
   467→    this.animT += dt;
   468→    this.updateFlash(dt);
   469→    this.iFrames = Math.max(0, this.iFrames - dt);
   470→    this.comboTimer = Math.max(0, this.comboTimer - dt);
   471→    if (this.comboTimer === 0 && this.state !== 'attack') { this.comboNext = 0; this.attackIdx = 0; }
   472→
   473→    // regeneración de aguante (mejorada por equipo y consumibles)
   474→    this.staminaDelay = Math.max(0, this.staminaDelay - dt);
   475→    if (this.staminaDelay === 0 && this.state !== 'dead') {
   476→      const regen = 26 * (1 + this.equip.stam + this.perm.stam);
   477→      this.stamina = Math.min(this.maxStamina, this.stamina + regen * dt);
   478→    }
   479→
   480→    // dirección de movimiento relativa a cámara (scratches reutilizables)
   481→    const ix = (input.right ? 1 : 0) - (input.left ? 1 : 0);
   482→    const iz = (input.fwd ? 1 : 0) - (input.back ? 1 : 0);
   483→    const fwd = _fwd.set(Math.sin(camYaw), 0, Math.cos(camYaw)).multiplyScalar(-1);
   484→    const right = _right.set(-fwd.z, 0, fwd.x);
   485→    const moveDir = _moveDir.set(0, 0, 0).addScaledVector(fwd, iz).addScaledVector(right, ix);
   486→    const hasMove = moveDir.lengthSq() > 0.001;
   487→    if (hasMove) moveDir.normalize();
   488→
   489→    const locked = !!(this.lockTarget && this.lockTarget.alive);
   490→    const targetPos = locked ? this.lockTarget!.pos : null;
   491→
   492→    switch (this.state) {
   493→      case 'dead': {
   494→        this.stateT += dt;
   495→        if (!this.glb) {
   496→          const clip = CLIPS.death;
   497→          const p = sampleClip(clip, Math.min(this.stateT, clip.dur));
   498→          this.applier.snap(p, dt);
   499→        }
   500→        this.root.rotation.y = this.yaw;
   501→        this.pos.addScaledVector(this.knock, dt);
   502→        this.knock.multiplyScalar(Math.max(0, 1 - 6 * dt));
   503→        this.applyTransform(ctx, dt);
   504→        return;
   505→      }
   506→      case 'hurt': {
   507→        this.stateT += dt;
   508→        if (!this.glb) {
   509→          const clip = CLIPS.hurt;
   510→          this.applier.snap(sampleClip(clip, Math.min(this.stateT, clip.dur)), dt);
   511→        }
   512→        if (this.stateT >= CLIPS.hurt.dur) this.state = 'idle';
   513→        break;
   514→      }
   515→      case 'potion': {
   516→        this.stateT += dt;
   517→        if (!this.glb) this.applier.apply(idlePose(this.animT), dt, 14);
   518→        if (this.stateT > 0.45) this.state = 'idle';
   519→        break;
   520→      }
   521→      case 'roll': {
   522→        this.stateT += dt;
   523→        if (!this.glb) {
   524→          const clip = CLIPS.roll;
   525→          const p = sampleClip(clip, Math.min(this.stateT, clip.dur));
   526→          this.applier.snap(p, dt);
   527→        }
   528→        this.pos.addScaledVector(this.rollDir, 10.5 * dt);
   529→        if (Math.random() < 0.4) {
   530→          ctx.particles.spawn({
   531→            x: this.pos.x, y: this.pos.y + 0.1, z: this.pos.z,
   532→            vx: rand(-1, 1), vy: rand(0.5, 1.5), vz: rand(-1, 1),
   533→            color: 0x8a7a5e, size: 0.25, life: 0.4, gravity: 1, drag: 2, glow: 0.6,
   534→          });
   535→        }
   536→        if (this.stateT >= CLIPS.roll.dur) { this.state = 'idle'; this.applier.apply(ZERO_POSE, dt, 20); }
   537→        break;
   538→      }
   539→      case 'attack': {
   540→        this.stateT += dt;
   541→        const def = this.currentAttack!;
   542→        // buffers de combo (se consumen aquí para no arrastrar pulsos viejos)
   543→        if (input.consumeAttack()) this.buffered = true;
   544→        if (input.consumeHeavy()) this.bufferedHeavy = true;
   545→        // seguimiento del objetivo
   546→        if (targetPos) {
   547→          const ty = Math.atan2(targetPos.x - this.pos.x, targetPos.z - this.pos.z);
   548→          this.yaw = dampAngle(this.yaw, ty, 10, dt);
   549→        } else {
   550→          this.yaw = dampAngle(this.yaw, Math.atan2(fwd.x, fwd.z), 8, dt);
   551→        }
   552→        const prog = this.stateT / def.dur;
   553→        // impulso hacia delante durante la fase activa (embestida del tajo)
   554→        if (prog > 0.30 && prog < 0.72) {
   555→          this.pos.addScaledVector(_tmpA.set(Math.sin(this.yaw), 0, Math.cos(this.yaw)), def.impulse * dt);
   556→        }
   557→        // pose procedural (fallback sin GLB) sincronizada con la jugabilidad
   558→        if (!this.glb) {
   559→          const clip = CLIPS[def.clip];
   560→          const speedK = clip.dur > 0.01 ? clip.dur / def.dur : 1;
   561→          const p = sampleClip(clip, Math.min(this.stateT * speedK, clip.dur));
   562→          this.applier.snap(p, dt);
   563→        }
   564→        // momento del golpe
   565→        if (!this.didHit && this.stateT >= def.hitAt) {
   566→          this.didHit = true;
   567→          this.strike(ctx, def);
   568→        }
   569→        // ESQUIVA-CANCEL: rodar cancela la recuperación del ataque
   570→        if (input.consumeRoll() && prog >= Math.min(def.chainAt, 1.0) * 0.82) {
   571→          const dir = hasMove ? moveDir : _tmpA.set(Math.sin(this.yaw), 0, Math.cos(this.yaw));
   572→          this.yaw = Math.atan2(dir.x, dir.z);
   573→          if (this.tryRoll(ctx, dir)) break;
   574→        }
   575→        // ENCADENADO CON CANCELACIÓN (DMC): el siguiente golpe arranca en
   576→        // cuanto termina la fase activa — sin esperar la recuperación
   577→        if (prog >= def.chainAt && (this.buffered || this.bufferedHeavy)) {
   578→          const wantHeavy = this.bufferedHeavy && this.stamina > PLAYER_HEAVY.stam * 0.4;
   579→          if (this.tryAttack(wantHeavy, ctx, _tmpA.set(Math.sin(this.yaw), 0, Math.cos(this.yaw)))) break;
   580→        }
   581→        if (this.stateT >= def.dur) {
   582→          if (this.buffered && !this.bufferedHeavy && this.stamina > 8) {
   583→            // golpes ligeros se auto-encadenan al acabar si hubo pulso tardío
   584→            if (this.tryAttack(false, ctx, _tmpA.set(Math.sin(this.yaw), 0, Math.cos(this.yaw)))) break;
   585→          }
   586→          // fin del ataque sin encadenar: la cadena sigue viva 0.9 s
   587→          this.state = 'idle';
   588→          this.comboTimer = 0.9;
   589→          if (def.kind !== 'light') this.comboNext = 0;
   590→          this.buffered = false;
   591→          this.bufferedHeavy = false;
   592→        }
   593→        break;
   594→      }
   595→      case 'idle': {
   596→        // acciones nuevas
   597→        if (input.consumePotion()) {
   598→          if (this.healPotion(ctx)) break;
   599→        }
   600→        if (input.consumeRoll()) {
   601→          const dir = hasMove ? moveDir : _tmpA.set(Math.sin(this.yaw), 0, Math.cos(this.yaw));
   602→          this.yaw = Math.atan2(dir.x, dir.z);
   603→          if (this.tryRoll(ctx, dir)) break;
   604→        }
   605→        const wantHeavy = input.consumeHeavy();
   606→        const wantLight = input.consumeAttack();
   607→        if (wantHeavy || wantLight) {
   608→          const faceDir = targetPos
   609→            ? _tmpA.set(targetPos.x - this.pos.x, 0, targetPos.z - this.pos.z).normalize()
   610→            : (hasMove ? moveDir : fwd);
   611→          this.yaw = Math.atan2(faceDir.x, faceDir.z);
   612→          this.tryAttack(wantHeavy, ctx, faceDir);
   613→        }
   614→        break;
   615→      }
   616→    }
   617→
   618→    // movimiento normal (solo fuera de acciones que bloquean)
   619→    if (this.state === 'idle') {
   620→      this.sprinting = input.sprint && hasMove;
   621→      const speed = (this.sprinting ? 9.2 : 6.2) * this.moveSpeedMul;
   622→      if (hasMove) {
   623→        this.pos.addScaledVector(moveDir, speed * dt);
   624→        if (!locked) {
   625→          this.yaw = dampAngle(this.yaw, Math.atan2(moveDir.x, moveDir.z), 14, dt);
   626→        } else {
   627→          this.yaw = dampAngle(this.yaw, Math.atan2(targetPos!.x - this.pos.x, targetPos!.z - this.pos.z), 12, dt);
   628→        }
   629→      }
   630→      this.moving = hasMove;
   631→      // pose (mocap GLB con crossfade o procedural)
   632→      if (this.glb) {
   633→        if (hasMove) {
   634→          if (this.sprinting) {
   635→            // esprintar: ciclo de carrera siempre orientado al avance
   636→            this.anim('run', { fade: 0.22, timeScale: 1.32 });
   637→          } else if (locked) {
   638→            // lock-on: strafes/backpedal reales según dirección relativa
   639→            const facing = _tmpB.set(Math.sin(this.yaw), 0, Math.cos(this.yaw));
   640→            const dot = moveDir.dot(facing);
   641→            const side = moveDir.dot(_tmpA.set(-facing.z, 0, facing.x));
   642→            if (dot > 0.45) this.anim('walk', { fade: 0.22, timeScale: 1.42 });
   643→            else if (dot < -0.45) this.anim('back', { fade: 0.22, timeScale: 1.35 });
   644→            else if (side > 0) this.anim('strafeR', { fade: 0.25, timeScale: 1.3 });
   645→            else this.anim('strafeL', { fade: 0.25, timeScale: 1.3 });
   646→          } else {
   647→            this.anim('walk', { fade: 0.22, timeScale: 1.42 });
   648→          }
   649→        } else {
   650→          this.anim('idle', { fade: 0.3 });
   651→        }
   652→      } else if (hasMove) {
   653→        this.runPhase += dt * (this.sprinting ? 1.3 : 1);
   654→        this.applier.apply(runPose(this.runPhase, this.sprinting ? 1.15 : 1), dt, 12);
   655→      } else if (locked) {
   656→        this.applier.apply(strafePose(this.animT, false), dt, 10);
   657→      } else {
   658→        this.applier.apply(idlePose(this.animT), dt, 8);
   659→      }
   660→    }
   661→
   662→    this.glb?.animator.update(dt);
   663→
   664→    // knockback decay
   665→    this.pos.addScaledVector(this.knock, dt);
   666→    this.knock.multiplyScalar(Math.max(0, 1 - 7 * dt));
   667→
   668→    this.applyTransform(ctx, dt);
   669→  }
   670→
   671→  private strike(ctx: GameCtx, def: AttackDef) {
   672→    // (la resolución de impactos la hace Game mediante pendingStrike)
   673→    this.pendingStrike = def;
   674→  }
   675→  pendingStrike: AttackDef | null = null;
   676→
   677→  private applyTransform(ctx: GameCtx, dt: number) {
   678→    const h = ctx.world.height(this.pos.x, this.pos.z);
   679→    this.pos.y = damp(this.pos.y, h, 25, dt);
   680→    this.root.position.copy(this.pos);
   681→    this.root.rotation.y = this.yaw;
   682→  }
   683→
   684→  /** Ancla de la punta/base de la espada para la estela */
   685→  swordPoints(outBase: THREE.Vector3, outTip: THREE.Vector3) {
   686→    const w = this.rig.weapon;
   687→    if (!w) return;
   688→    outBase.set(0, 0.15, 0);
   689→    outTip.set(0, 1.05, 0);
   690→    w.updateWorldMatrix(true, false);
   691→    outBase.applyMatrix4(w.matrixWorld);
   692→    outTip.applyMatrix4(w.matrixWorld);
   693→  }
   694→}

   698→export class Projectile {
   699→  root = new THREE.Group();
   700→  pos = new THREE.Vector3();
   701→  vel = new THREE.Vector3();
   702→  dmg: number;
   703→  kind: 'arrow' | 'orb';
   704→  life = 4;
   705→  dead = false;
   706→  gravity: number;
   707→
   708→  constructor(o: { pos: THREE.Vector3; dir: THREE.Vector3; speed: number; dmg: number; kind: 'arrow' | 'orb' }) {
   709→    this.pos.copy(o.pos);
   710→    this.vel.copy(o.dir).normalize().multiplyScalar(o.speed);
   711→    this.dmg = o.dmg;
   712→    this.kind = o.kind;
   713→    this.gravity = o.kind === 'arrow' ? 4.5 : 0;
   714→    if (o.kind === 'arrow') {
   715→      this.root.add(buildArrowMesh());
   716→    } else {
   717→      const core = new THREE.Mesh(
   718→        new THREE.IcosahedronGeometry(0.22, 0),
   719→        new THREE.MeshStandardMaterial({ color: 0x2a0a0a, emissive: 0xff2211, emissiveIntensity: 2.6, roughness: 0.5 })
   720→      );
   721→      this.root.add(core);
   722→      const halo = new THREE.Mesh(
   723→        new THREE.SphereGeometry(0.4, 8, 6),
   724→        new THREE.MeshBasicMaterial({ color: 0xff3a22, transparent: true, opacity: 0.3 })
   725→      );
   726→      this.root.add(halo);
   727→    }
   728→    this.root.position.copy(this.pos);
   729→  }
   730→
   731→  update(dt: number, ctx: GameCtx) {
   732→    this.life -= dt;
   733→    if (this.life <= 0) { this.dead = true; return; }
   734→    this.vel.y -= this.gravity * dt;
   735→    if (this.kind === 'orb') {
   736→      // ligero homing
   737→      const to = ctx.player.pos.clone().setY(ctx.player.pos.y + 1.1).sub(this.pos).normalize().multiplyScalar(this.vel.length());
   738→      this.vel.lerp(to, 1 - Math.exp(-1.6 * dt));
   739→    }
   740→    this.pos.addScaledVector(this.vel, dt);
   741→    this.root.position.copy(this.pos);
   742→    if (this.kind === 'arrow') {
   743→      this.root.lookAt(this.pos.clone().add(this.vel));
   744→      this.root.rotateX(Math.PI / 2);
   745→    } else {
   746→      this.root.rotation.y += dt * 6;
   747→      if (Math.random() < 0.6) ctx.particles.spawn({
   748→        x: this.pos.x, y: this.pos.y, z: this.pos.z,
   749→        color: 0xff3a22, size: 0.22, life: 0.4, glow: 2.2, shrink: false,
   750→        vx: rand(-0.5, 0.5), vy: rand(-0.5, 0.5), vz: rand(-0.5, 0.5),
   751→      });
   752→    }
   753→    // impacto con el jugador
   754→    const p = ctx.player.pos;
   755→    const dx = p.x - this.pos.x, dy = (p.y + 1.1) - this.pos.y, dz = p.z - this.pos.z;
   756→    if (dx * dx + dy * dy + dz * dz < 0.9 * 0.9) {
   757→      if (ctx.player.takeDamage(this.dmg, this.pos, ctx)) {
   758→        if (this.kind === 'orb') ctx.particles.burst({ x: this.pos.x, y: this.pos.y, z: this.pos.z, count: 18, speed: 4, color: 0xff3a22, size: 0.28, life: 0.6, glow: 2 });
   759→      }
   760→      this.dead = true;
   761→      return;
   762→    }
   763→    // impacto con el suelo
   764→    if (this.pos.y <= ctx.world.height(this.pos.x, this.pos.z) + 0.05) {
   765→      this.dead = true;
   766→      ctx.particles.burst({ x: this.pos.x, y: this.pos.y, z: this.pos.z, count: 6, speed: 1.5, color: 0x8a7a5e, size: 0.18, life: 0.4, gravity: 3 });
   767→    }
   768→  }
   769→}

   773→export class Pickup {
   774→  root: THREE.Group;
   775→  pos = new THREE.Vector3();
   776→  life = 40;
   777→  dead = false;
   778→  kind: 'potion' | 'item';
   779→  item: ItemDef | null = null;
   780→  private core: THREE.Mesh;
   781→  private t = Math.random() * 10;
   782→
   783→  constructor(pos: THREE.Vector3, kind: 'potion' | 'item', item?: ItemDef) {
   784→    this.kind = kind;
   785→    const color = kind === 'item' && item ? RARITY_INFO[item.rarity].hex : 0xff4a4a;
   786→    const { group, core } = buildPickupOrb(color);
   787→    this.root = group;
   788→    this.core = core;
   789→    this.item = item ?? null;
   790→    if (kind === 'item') {
   791→      // los objetos flotan un poco más alto y giran una cartera de destellos
   792→      this.root.scale.setScalar(1.25);
   793→    }
   794→    this.pos.copy(pos);
   795→    this.pos.y = terrainHeight(pos.x, pos.z) + (kind === 'item' ? 0.7 : 0.5);
   796→    this.root.position.copy(this.pos);
   797→  }
   798→
   799→  update(dt: number, ctx: GameCtx) {
   800→    this.t += dt;
   801→    this.life -= dt;
   802→    if (this.life <= 0) { this.dead = true; return; }
   803→    this.root.position.y = this.pos.y + Math.sin(this.t * 2.4) * 0.12;
   804→    this.core.rotation.y += dt * 2.5;
   805→    this.core.rotation.x += dt * 1.2;
   806→    const p = ctx.player.pos;
   807→    const d = Math.hypot(p.x - this.pos.x, p.z - this.pos.z);
   808→    if (d < 1.5 && Math.abs(p.y - this.pos.y) < 2.5 && !ctx.player.invulnHit) {
   809→      if (this.kind === 'item' && this.item) {
   810→        ctx.gainItem(this.item);
   811→        this.dead = true;
   812→        return;
   813→      }
   814→      if (ctx.player.potions < ctx.player.maxPotions) {
   815→        ctx.player.potions++;
   816→        ctx.audio.potion();
   817→        ctx.addDamageNumber(this.root.position.clone(), '+1 POCIÓN', '#ff8a8a');
   818→        this.dead = true;
   819→      }
   820→    }
   821→  }
   822→}
