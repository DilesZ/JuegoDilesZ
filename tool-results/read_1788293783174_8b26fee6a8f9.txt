    29→    29→
    30→    30→export class GlbAnimator {
    31→    31→  mixer: THREE.AnimationMixer;
    32→    32→  private actions = new Map<string, THREE.AnimationAction>();
    33→    33→  current = '';
    34→    34→  speed = 1;
    35→    35→
    36→    36→  constructor(root: THREE.Object3D, clips: Record<string, THREE.AnimationClip>) {
    37→    37→    this.mixer = new THREE.AnimationMixer(root);
    38→    38→    for (const [name, clip] of Object.entries(clips)) {
    39→    39→      this.actions.set(name, this.mixer.clipAction(clip));
    40→    40→    }
    41→    41→  }
    42→    42→
    43→    43→  has(name: string) { return this.actions.has(name); }
    44→    44→
    45→    45→  play(name: string, opts: { fade?: number; once?: boolean; restart?: boolean; timeScale?: number } = {}) {
    46→    46→    const next = this.actions.get(name);
    47→    47→    if (!next) return;
    48→    48→    if (this.current === name && !opts.restart) return;
    49→    49→    const fade = opts.fade ?? 0.18;
    50→    50→    const prev = this.current ? this.actions.get(this.current) : null;
    51→    51→    next.reset();
    52→    52→    next.setLoop(opts.once ? THREE.LoopOnce : THREE.LoopRepeat, opts.once ? 1 : Infinity);
    53→    53→    next.clampWhenFinished = !!opts.once;
    54→    54→    next.setEffectiveTimeScale(opts.timeScale ?? 1);
    55→    55→    next.setEffectiveWeight(1);
    56→    56→    next.fadeIn(fade).play();
    57→    57→    if (prev && prev !== next) prev.fadeOut(fade);
    58→    58→    this.current = name;
    59→    59→  }
    60→    60→
    61→    61→  update(dt: number) { this.mixer.update(dt * this.speed); }
    62→    62→}

    92→    92→/* ============================================================
    93→    93→   HORNEADO DE CLIPS: pose procedural → pistas del esqueleto real
    94→    94→   La rotación procedural (euler XYZ) se aplica en ejes del mundo
    95→    95→   sobre la dirección de reposo alineada del hueso:
    96→    96→     D  = qAnim · R_align          (espacio mundo del personaje)
    97→    97→     L' = Pw⁻¹ · D · Pw · L0       (local del hueso)

    99→    99→
   100→   100→const _q1 = new THREE.Quaternion();
   101→   101→const _q2 = new THREE.Quaternion();
   102→   102→const _q3 = new THREE.Quaternion();
   103→   103→const _v1 = new THREE.Vector3();
   104→   104→const _v2 = new THREE.Vector3();
   105→   105→const AX = new THREE.Vector3(1, 0, 0);
   106→   106→const AY = new THREE.Vector3(0, 1, 0);
   107→   107→const AZ = new THREE.Vector3(0, 0, 1);
   108→   108→const DOWN = new THREE.Vector3(0, -1, 0);

   110→   110→function findBone(model: THREE.Object3D, ...candidates: string[]): THREE.Object3D | null {
   111→   111→  let found: THREE.Object3D | null = null;
   112→   112→  const wanted = candidates.map(c => c.toLowerCase());
   113→   113→  model.traverse(o => {
   114→   114→    if (found) return;
   115→   115→    const raw = o.name.includes(':') ? o.name.slice(o.name.lastIndexOf(':') + 1) : o.name;
   116→   116→    if (wanted.includes(raw.toLowerCase())) found = o;
   117→   117→  });
   118→   118→  return found;
   119→   119→}

   121→   121→/** dirección de reposo del hueso hacia su primer hijo (espacio raíz) */
   122→   122→function boneDirection(bone: THREE.Object3D, model: THREE.Object3D): THREE.Vector3 {
   123→   123→  model.updateMatrixWorld(true);
   124→   124→  const a = bone.getWorldPosition(new THREE.Vector3());
   125→   125→  const child = bone.children.find(c => c.type === 'Bone') ?? bone.children[0];
   126→   126→  const b = child ? child.getWorldPosition(new THREE.Vector3()) : a.clone().add(new THREE.Vector3(0, 0.1, 0));
   127→   127→  const d = b.sub(a);
   128→   128→  if (d.lengthSq() < 1e-8) d.set(0, -1, 0);
   129→   129→  return d.normalize();
   130→   130→}

   132→   132→class SkeletonBinder {
   133→   133→  hips: THREE.Object3D | null = null;
   134→   134→  private hipsParentInv = new THREE.Matrix4();
   135→   135→  private hipsRestWorld = new THREE.Vector3();
   136→   136→  private joints = new Map<string, { bone: THREE.Object3D; align: THREE.Quaternion; parentW: THREE.Quaternion; restLocal: THREE.Quaternion }>();
   137→   137→
   138→   138→  constructor(model: THREE.Object3D, withLimbs: boolean) {
   139→   139→    model.updateMatrixWorld(true);
   140→   140→    const hips = findBone(model, 'Hips', 'Pelvis');
   141→   141→    this.hips = hips;
   142→   142→    if (hips && hips.parent) {
   143→   143→      this.hipsParentInv.copy(hips.parent.matrixWorld).invert();
   144→   144→      this.hipsRestWorld.copy(hips.getWorldPosition(new THREE.Vector3()));
   145→   145→    }
   146→   146→    const bind = (key: string, bone: THREE.Object3D | null, alignDir: THREE.Vector3 | null) => {
   147→   147→      if (!bone || !bone.parent) return;
   148→   148→      const parentW = new THREE.Quaternion();
   149→   149→      bone.parent.getWorldQuaternion(parentW);
   150→   150→      const align = new THREE.Quaternion();
   151→   151→      if (alignDir) align.setFromUnitVectors(boneDirection(bone, model), alignDir);
   152→   152→      this.joints.set(key, { bone, align, parentW, restLocal: bone.quaternion.clone() });
   153→   153→    };
   154→   154→    bind('body', this.hips, null);
   155→   155→    bind('torso', findBone(model, 'Spine1', 'Spine2', 'Spine', 'Chest'), null);
   156→   156→    bind('head', findBone(model, 'Head', 'Neck'), null);
   157→   157→    if (withLimbs) {
   158→   158→      bind('armL', findBone(model, 'LeftArm', 'LeftForeArm'), DOWN);
   159→   159→      bind('armR', findBone(model, 'RightArm', 'RightForeArm'), DOWN);
   160→   160→      bind('legL', findBone(model, 'LeftUpLeg', 'LeftLeg'), DOWN);
   161→   161→      bind('legR', findBone(model, 'RightUpLeg', 'RightLeg'), DOWN);
   162→   162→    }
   163→   163→  }
   164→   164→
   165→   165→  /** Construye las pistas del esqueleto para poses muestreadas */
   166→   166→  tracks(samples: { t: number; pose: Pose }[], height: number): THREE.KeyframeTrack[] {
   167→   167→    const tracks: THREE.KeyframeTrack[] = [];
   168→   168→    const times = samples.map(s => s.t);
   169→   169→    const k = height / PROC_HEIGHT;
   170→   170→    for (const [key, b] of this.joints) {
   171→   171→      const vals = new Float32Array(samples.length * 4);
   172→   172→      samples.forEach((s, i) => {
   173→   173→        const e = s.pose[key as keyof Pose] as [number, number, number];
   174→   174→        // qAnim = qX·qY·qZ (mismo orden que Euler 'XYZ' del rig procedural)
   175→   175→        _q1.setFromAxisAngle(AX, e[0]);
   176→   176→        _q2.setFromAxisAngle(AY, e[1]);
   177→   177→        _q1.multiply(_q2);
   178→   178→        _q2.setFromAxisAngle(AZ, e[2]);
   179→   179→        _q1.multiply(_q2);        // _q1 = qAnim
   180→   180→        _q1.multiply(b.align);    // D = qAnim · R_align
   181→   181→        _q3.copy(b.parentW).invert().multiply(_q1).multiply(b.parentW).multiply(b.restLocal);
   182→   182→        vals[i * 4] = _q3.x; vals[i * 4 + 1] = _q3.y; vals[i * 4 + 2] = _q3.z; vals[i * 4 + 3] = _q3.w;
   183→   183→      });
   184→   184→      tracks.push(new THREE.QuaternionKeyframeTrack(`${b.bone.name}.quaternion`, times, vals));
   185→   185→    }
   186→   186→    // posición de caderas (bodyY del rig procedural)
   187→   187→    if (this.hips) {
   188→   188→      const vals = new Float32Array(samples.length * 3);
   189→   189→      samples.forEach((s, i) => {
   190→   190→        _v1.set(0, s.pose.bodyY * k, 0).add(this.hipsRestWorld);
   191→   191→        _v2.copy(_v1).applyMatrix4(this.hipsParentInv);
   192→   192→        vals[i * 3] = _v2.x; vals[i * 3 + 1] = _v2.y; vals[i * 3 + 2] = _v2.z;
   193→   193→      });
   194→   194→      tracks.push(new THREE.VectorKeyframeTrack(`${this.hips.name}.position`, times, vals));
   195→   195→    }
   196→   196→    return tracks;
   197→   197→  }
   198→   198→}

   200→   200→/** Hornea un clip de acción (duración EXACTA de CLIPS → jugabilidad intacta) */
   201→   201→function bakeActionClip(name: string, clipName: string, binder: SkeletonBinder, height: number, steps = 14): THREE.AnimationClip {
   202→   202→  const dur = CLIPS[clipName].dur;
   203→   203→  const samples: { t: number; pose: Pose }[] = [];
   204→   204→  for (let i = 0; i <= steps; i++) {
   205→   205→    const t = dur * (i / steps);
   206→   206→    samples.push({ t, pose: sampleClip(CLIPS[clipName], t) });
   207→   207→  }
   208→   208→  return new THREE.AnimationClip(name, dur, binder.tracks(samples, height));
   209→   209→}

   212→   212→function bakeLoopClip(name: string, dur: number, steps: number, fn: (t: number) => Pose, binder: SkeletonBinder, height: number): THREE.AnimationClip {
   213→   213→  const samples: { t: number; pose: Pose }[] = [];
   214→   214→  for (let i = 0; i <= steps; i++) {
   215→   215→    const t = dur * (i / steps);
   216→   216→    samples.push({ t, pose: fn(t) });
   217→   217→  }
   218→   218→  return new THREE.AnimationClip(name, dur, binder.tracks(samples, height));
   219→   219→}

   221→   221→/** Saludo del mercader: brazo derecho saludando (1.2 s) */
   222→   222→function greetPose(t: number): Pose {
   223→   223→  const dur = 1.2;
   224→   224→  const p = idlePose(t);
   225→   225→  const w = Math.min(1, Math.min(t, dur - t) / 0.35);
   226→   226→  p.armR = [-2.35 * w, 0, (-0.45 + Math.sin(t * 13) * 0.28) * w];
   227→   227→  p.head = [0, 0, -0.06 * w];
   228→   228→  return p;
   229→   229→}

   231→   231→/** Beber poción (0.45 s): lleva la mano izquierda a la cara */
   232→   232→function potionPose(t: number): Pose {
   233→   233→  const dur = 0.45;
   234→   234→  const p = idlePose(t);
   235→   235→  const w = Math.sin((t / dur) * Math.PI);
   236→   236→  p.armL = [-1.9 * w, 0, 0.35 * w];
   237→   237→  p.head = [0.12 * w, 0, 0];
   238→   238→  return p;
   239→   239→}

   243→   243→/** Escala el modelo a una altura objetivo y apoya los pies en y=0 */
   244→   244→function normalizeModel(model: THREE.Object3D, targetH: number): number {
   245→   245→  model.updateMatrixWorld(true);
   246→   246→  const bbox = new THREE.Box3().setFromObject(model);
   247→   247→  const h = Math.max(0.001, bbox.max.y - bbox.min.y);
   248→   248→  model.scale.setScalar(targetH / h);
   249→   249→  model.updateMatrixWorld(true);
   250→   250→  const bb2 = new THREE.Box3().setFromObject(model);
   251→   251→  model.position.y -= bb2.min.y;
   252→   252→  return targetH;
   253→   253→}

   286→   286→/* ---------- Retarget de locomoción (Soldier → héroe) ---------- */
   287→   287→
   288→   288→function retargetLocomotion(
   289→   289→  targetModel: THREE.Object3D,
   290→   290→  sourceModel: THREE.Object3D,
   291→   291→  sourceClips: THREE.AnimationClip[],
   292→   292→): Record<string, THREE.AnimationClip> | null {
   293→   293→  try {
   294→   294→    const targetSkin = targetModel.getObjectByProperty('isSkinnedMesh', true) as THREE.SkinnedMesh | undefined;
   295→   295→    const sourceSkin = sourceModel.getObjectByProperty('isSkinnedMesh', true) as THREE.SkinnedMesh | undefined;
   296→   296→    if (!targetSkin || !sourceSkin) return null;
   297→   297→    const sourceNames = new Set(sourceSkin.skeleton.bones.map(b => b.name));
   298→   298→    // GLTFLoader sanea los nombres ('mixamorig:Hips' → 'mixamorigHips'):
   299→   299→    // probamos las tres variantes al construir el mapa destino→fuente
   300→   300→    const names: Record<string, string> = {};
   301→   301→    for (const b of targetSkin.skeleton.bones) {
   302→   302→      for (const cand of [`mixamorig:${b.name}`, `mixamorig${b.name}`, b.name]) {
   303→   303→        if (sourceNames.has(cand) && cand !== b.name) { names[b.name] = cand; break; }
   304→   304→      }
   305→   305→    }
   306→   306→    const hipName = names['Hips'] ?? 'mixamorigHips';
   307→   307→    const out: Record<string, THREE.AnimationClip> = {};
   308→   308→    for (const clip of sourceClips) {
   309→   309→      if (!/^(idle|walk|run)$/i.test(clip.name)) continue;
   310→   310→      const retargeted = SkeletonUtils.retargetClip(targetSkin, sourceSkin, clip, {
   311→   311→        hip: hipName,
   312→   312→        names,
   313→   313→      });
   314→   314→      if (retargeted && retargeted.tracks.length > 0) {
   315→   315→        // retargetClip genera tracks '.bones[X].quaternion' (requieren mixer
   316→   316→        // sobre el SkinnedMesh); los reescribimos a 'X.quaternion' para que
   317→   317→        // convivan con los clips horneados en el mixer de la escena.
   318→   318→        for (const tr of retargeted.tracks) {
   319→   319→          tr.name = tr.name.replace(/^\.bones\[(.+?)\]\./, '$1.');
   320→   320→        }
   321→   321→        out[clip.name.toLowerCase()] = retargeted;
   322→   322→      }
   323→   323→    }
   324→   324→    return Object.keys(out).length > 0 ? out : null;
   325→   325→  } catch (err) {
   326→   326→    console.warn('[AETHERIA] retarget de locomoción falló:', err);
   327→   327→    return null;
   328→   328→  }
   329→   329→}

   334→   334→
   335→   335→function makeCharacter(
   336→   336→  model: THREE.Object3D,
   337→   337→  height: number,
   338→   338→  clips: Record<string, THREE.AnimationClip>,
   339→   339→  weapon: THREE.Group | null,
   340→   340→  weaponMat: THREE.Material | null,
   341→   341→  shield: boolean,
   342→   342→): GlbCharacter {
   343→   343→  const root = new THREE.Group();
   344→   344→  root.add(model);
   345→   345→  let handR: THREE.Object3D | null = null;
   346→   346→  let handL: THREE.Object3D | null = null;
   347→   347→  if (weapon || shield) {
   348→   348→    handR = findBone(model, 'RightHand', 'RightForeArm');
   349→   349→    handL = findBone(model, 'LeftHand', 'LeftForeArm');
   350→   350→  }
   351→   351→  model.updateMatrixWorld(true);
   352→   352→  if (weapon && handR) {
   353→   353→    handR.add(weapon);
   354→   354→    // La orientación de los huesos de mano varía por rig: fijamos la hoja
   355→   355→    // en espacio mundo (arriba y ligeramente al frente) calculando
   356→   356→    // q_local = q_mundo_mano⁻¹ · q_deseado, con el modelo en pose de reposo.
   357→   357→    const handWorld = new THREE.Quaternion();
   358→   358→    handR.getWorldQuaternion(handWorld);
   359→   359→    const desired = new THREE.Quaternion().setFromEuler(new THREE.Euler(0.5, 0, 0));
   360→   360→    weapon.quaternion.copy(handWorld.invert().multiply(desired));
   361→   361→    weapon.position.set(0, 0.06, 0.02);
   362→   362→  }
   363→   363→  if (shield && handL) {
   364→   364→    const sh = buildShield();
   365→   365→    sh.scale.setScalar(0.9);
   366→   366→    handL.add(sh);
   367→   367→    const handWorldL = new THREE.Quaternion();
   368→   368→    handL.getWorldQuaternion(handWorldL);
   369→   369→    // cara del escudo (+Z local) hacia fuera-izquierda y algo al frente
   370→   370→    const desiredL = new THREE.Quaternion().setFromEuler(new THREE.Euler(0, -1.15, 0));
   371→   371→    sh.quaternion.copy(handWorldL.invert().multiply(desiredL));
   372→   372→    sh.position.set(0, -0.07, 0);
   373→   373→  }
   374→   374→  const animator = new GlbAnimator(model, clips);
   375→   375→  return {
   376→   376→    root, model,
   377→   377→    rig: { root, weapon, weaponMat: (weaponMat ?? null) as VisualRig['weaponMat'], handR, handL, height },
   378→   378→    animator, height,
   379→   379→  };
   380→   380→}

   382→   382→/** Héroe: readyplayer.me + locomoción retargeteada + combate horneado */
   383→   383→export function createHeroCharacter(pack: CharacterPack): GlbCharacter {
   384→   384→  const targetH = 1.62;
   385→   385→  const model = SkeletonUtils.clone(pack.heroSource);
   386→   386→  normalizeModel(model, targetH);
   387→   387→  enableShadows(model);

   417→   417→      if (mm && mm.emissive && mm.emissiveIntensity > 0) weaponMat = mm;
   418→   418→    }

   421→   421→}

   451→   451→  enableShadows(model);
   452→   452→
   453→   453→  const clips: Record<string, THREE.AnimationClip> = { ...pack.xbotClips, ...pack.xbotCombat };

   459→   459→    case 'axe': weapon = buildAxe(); break;
   460→   460→    case 'greatsword': weapon = buildGreatsword(1); break;

   463→   463→    weapon.scale.setScalar(look.ws);

   465→   465→      if (m instanceof THREE.Mesh) {
   466→   466→        const mm = m.material as THREE.MeshStandardMaterial;
   467→   467→        if (mm && mm.emissive && mm.emissiveIntensity > 0) weaponMat = mm;
   468→   468→      }
   469→   469→    });
   470→   470→  }
   471→   471→  return makeCharacter(model, look.h, clips, weapon, weaponMat, false);
   472→   472→}

   500→   500→export class Fox {
   501→   501→  root = new THREE.Group();
   502→   502→  pos = new THREE.Vector3();
   503→   503→  yaw = 0;
   504→   504→  private animator: GlbAnimator;
   505→   505→  private target = new THREE.Vector3();
   506→   506→  private state: 'survey' | 'walk' | 'run' = 'survey';
   507→   507→  private t = rand(0, 5);
   508→   508→  private cd = rand(1, 4);
   509→   509→
   510→   510→  constructor(pack: CharacterPack, x: number, z: number) {
   511→   511→    const model = SkeletonUtils.clone(pack.foxSource);
   512→   512→    normalizeModel(model, 0.58);
   513→   513→    enableShadows(model);
   514→   514→    const clips = lowercaseClips(pack.foxClips);
   515→   515→    if (clips.survey) clips.idle = clips.survey;
   516→   516→    this.animator = new GlbAnimator(model, clips);
   517→   517→    this.root.add(model);
   518→   518→    this.pos.set(x, terrainHeight(x, z), z);
   519→   519→    this.root.position.copy(this.pos);
   520→   520→    this.pickTarget();
   521→   521→  }
   522→   522→
   523→   523→  private pickTarget() {
   524→   524→    const a = rand(0, Math.PI * 2);
   525→   525→    const r = rand(8, 46);
   526→   526→    this.target.set(Math.cos(a) * r, 0, Math.sin(a) * r);
   527→   527→    this.target.y = terrainHeight(this.target.x, this.target.z);
   528→   528→  }
   529→   529→
   530→   530→  update(dt: number, playerPos: THREE.Vector3) {
   531→   531→    this.t += dt;
   532→   532→    this.cd -= dt;
   533→   533→    const dPlayer = Math.hypot(playerPos.x - this.pos.x, playerPos.z - this.pos.z);
   534→   534→    if (dPlayer < 4.5) {
   535→   535→      const away = new THREE.Vector3(this.pos.x - playerPos.x, 0, this.pos.z - playerPos.z).normalize();
   536→   536→      this.target.copy(this.pos).addScaledVector(away, 12);
   537→   537→      this.target.y = terrainHeight(this.target.x, this.target.z);
   538→   538→      this.state = 'run';
   539→   539→    } else if (this.cd <= 0) {
   540→   540→      if (this.state !== 'survey') {
   541→   541→        this.state = 'survey'; this.cd = rand(2, 5);
   542→   542→      } else {
   543→   543→        this.pickTarget();
   544→   544→        this.state = rand(0, 1) > 0.35 ? 'walk' : 'run';
   545→   545→        this.cd = rand(4, 9);
   546→   546→      }
   547→   547→    }
   548→   548→    if (this.state !== 'survey') {
   549→   549→      const to = new THREE.Vector3().subVectors(this.target, this.pos).setY(0);
   550→   550→      if (to.length() < 1.4) { this.state = 'survey'; this.cd = rand(2, 6); }
   551→   551→      else {
   552→   552→        to.normalize();
   553→   553→        const spd = this.state === 'run' ? 5.2 : 1.7;
   554→   554→        this.pos.addScaledVector(to, spd * dt);
   555→   555→        this.pos.y = terrainHeight(this.pos.x, this.pos.z);
   556→   556→        this.yaw = dampAngle(this.yaw, Math.atan2(to.x, to.z), 6, dt);
   557→   557→      }
   558→   558→    }
   559→   559→    const r = Math.hypot(this.pos.x, this.pos.z);
   560→   560→    if (r > 78) { this.pos.multiplyScalar(78 / r); this.pickTarget(); }
   561→   561→    this.animator.play(this.state, { fade: 0.25 });
   562→   562→    this.animator.update(dt);
   563→   563→    this.root.position.copy(this.pos);
   564→   564→    this.root.rotation.y = this.yaw;
   565→   565→  }
   566→   566→}

   581→   581→
   582→   582→function mergeDungeon(scene: THREE.Object3D): { group: THREE.Group; radius: number } {
   583→   583→  const byMat = new Map<THREE.Material, THREE.BufferGeometry[]>();
   584→   584→  scene.updateMatrixWorld(true);
   585→   585→  scene.traverse(o => {
   586→   586→    if (!(o instanceof THREE.Mesh) || !o.material || Array.isArray(o.material)) return;
   587→   587→    const geo = o.geometry.clone().applyMatrix4(o.matrixWorld);
   588→   588→    const list = byMat.get(o.material as THREE.Material) ?? [];
   589→   589→    list.push(geo);
   590→   590→    byMat.set(o.material as THREE.Material, list);
   591→   591→  });
   592→   592→  const group = new THREE.Group();
   593→   593→  let radius = 6;
   594→   594→  // Material de piedra estilizada del juego: cohesiona las ruinas con
   595→   595→  // obeliscos/pilares y evita materiales negros del GLB (metálicos sin IBL)
   596→   596→  const stone = stoneMat();
   597→   597→  for (const [, geos] of byMat) {
   598→   598→    const merged = mergeDungeonGeos(geos);
   599→   599→    if (!merged) continue;
   600→   600→    merged.computeBoundingSphere();
   601→   601→    radius = Math.max(radius, merged.boundingSphere?.radius ?? 6);
   602→   602→    const mesh = new THREE.Mesh(merged, stone);
   603→   603→    mesh.receiveShadow = true;
   604→   604→    mesh.castShadow = false;
   605→   605→    mesh.name = 'ruina';
   606→   606→    group.add(mesh);
   607→   607→  }
   608→   608→  // normalizar: la mazmorra llega en unidades arbitrarias → landmark de ~34 m
   609→   609→  const bbox = new THREE.Box3().setFromObject(group);
   610→   610→  const maxDim = Math.max(bbox.max.x - bbox.min.x, bbox.max.y - bbox.min.y, bbox.max.z - bbox.min.z);
   611→   611→  const s = 34 / Math.max(1, maxDim);
   612→   612→  group.scale.setScalar(s);
   613→   613→  group.updateMatrixWorld(true);
   614→   614→  const bbox2 = new THREE.Box3().setFromObject(group);
   615→   615→  group.position.y = -bbox2.min.y;
   616→   616→  radius *= s;
   617→   617→  return { group, radius };
   618→   618→}
   619→   619→
   620→   620→function mergeDungeonGeos(geos: THREE.BufferGeometry[]): THREE.BufferGeometry | null {
   621→   621→  try {
   622→   622→    const clean = geos.map(g => {
   623→   623→      const c = g.index ? g.toNonIndexed() : g;
   624→   624→      for (const name of [...Object.keys(c.attributes)]) {
   625→   625→        if (!['position', 'normal', 'uv', 'color'].includes(name)) c.deleteAttribute(name);
   626→   626→      }
   627→   627→      if (!c.attributes.uv) {
   628→   628→        const count = c.attributes.position.count;
   629→   629→        c.setAttribute('uv', new THREE.BufferAttribute(new Float32Array(count * 2), 2));
   630→   630→      }
   631→   631→      if (!c.attributes.normal) c.computeVertexNormals();
   632→   632→      return c;
   633→   633→    });
   634→   634→    return mergeGeometries(clean, false);
   635→   635→  } catch (err) {
   636→   636→    console.warn('[AETHERIA] no se pudieron fusionar las ruinas:', err);
   637→   637→    return null;
   638→   638→  }
   639→   639→}
