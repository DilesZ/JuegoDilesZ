    16→    16→interface WindEntry { mat: THREE.Material; amp: number }
    17→    17→const windMats: WindEntry[] = [];
    18→    18→const flameMats: THREE.ShaderMaterial[] = [];
    19→    19→const _time = { value: 0 };

    21→    21→export function registerWind(mat: THREE.Material, amp: number, anchor: 'top' | 'bottom' = 'top') {
    22→    22→  if (windMats.some(w => w.mat === mat)) return;
    23→    23→  windMats.push({ mat, amp });
    24→    24→  mat.onBeforeCompile = (shader) => {
    25→    25→    shader.uniforms.uTime = _time;
    26→    26→    shader.uniforms.uAmp = { value: amp };
    27→    27→    shader.vertexShader = shader.vertexShader
    28→    28→      .replace('#include <common>', '#include <common>\nuniform float uTime;\nuniform float uAmp;')
    29→    29→      .replace('#include <begin_vertex>', `#include <begin_vertex>
    30→    30→        {
    31→    31→          float wx = 0.0, wz = 0.0;
    32→    32→          #ifdef USE_INSTANCING
    33→    33→            wx = instanceMatrix[3][0]; wz = instanceMatrix[3][2];
    34→    34→          #endif
    35→    35→          float swayF = pow(${anchor === 'top' ? 'max(uv.y, 0.0)' : 'max(1.0 - uv.y, 0.0)'}, 1.5);
    36→    36→          float sway = sin(uTime * 1.6 + wx * 0.35 + wz * 0.5) + 0.45 * sin(uTime * 2.9 + wx * 1.4 + wz * 0.8);
    37→    37→          transformed.x += sway * uAmp * swayF;
    38→    38→          transformed.z += sway * uAmp * 0.6 * swayF;
    39→    39→        }`);
    40→    40→  };
    41→    41→  mat.customProgramCacheKey = () => `wind_${amp}_${anchor}`;
    42→    42→}

    45→    45→  _time.value = t;
    46→    46→  for (const m of flameMats) m.uniforms.uTime.value = t;
    47→    47→}

    49→    49→/** Material de llama procedural (plano con forma de fuego animado) */
    50→    50→export function makeFlameMaterial(core: number, outer: number, scaleY = 1): THREE.ShaderMaterial {
    51→    51→  const mat = new THREE.ShaderMaterial({
    52→    52→    transparent: true, depthWrite: false, blending: THREE.AdditiveBlending, side: THREE.DoubleSide,
    53→    53→    uniforms: {
    54→    54→      uTime: { value: 0 },
    55→    55→      uCore: { value: new THREE.Color(core) },
    56→    56→      uOuter: { value: new THREE.Color(outer) },
    57→    57→      uScaleY: { value: scaleY },
    58→    58→    },
    59→    59→    vertexShader: `varying vec2 vUv;
    60→    60→      void main(){ vUv = uv; gl_Position = projectionMatrix * modelViewMatrix * vec4(position,1.0); }`,
    61→    61→    fragmentShader: `varying vec2 vUv; uniform float uTime; uniform vec3 uCore; uniform vec3 uOuter; uniform float uScaleY;
    62→    62→      float hash(vec2 p){ return fract(sin(dot(p, vec2(127.1,311.7))) * 43758.5453); }
    63→    63→      float noise(vec2 p){ vec2 i = floor(p), f = fract(p); f = f*f*(3.0-2.0*f);
    64→    64→        return mix(mix(hash(i), hash(i+vec2(1,0)), f.x), mix(hash(i+vec2(0,1)), hash(i+vec2(1,1)), f.x), f.y); }
    65→    65→      void main(){
    66→    66→        vec2 uv = vUv; uv.y *= uScaleY;
    67→    67→        float t = uTime;
    68→    68→        // ondulación horizontal creciente con la altura
    69→    69→        float wob = (noise(vec2(uv.y*3.0 - t*2.4, t*0.8)) - 0.5) * 0.34 * uv.y;
    70→    70→        float x = uv.x - 0.5 + wob;
    71→    71→        // silueta de llama: ancho decreciente hacia arriba
    72→    72→        float width = mix(0.42, 0.02, pow(uv.y, 0.72));
    73→    73→        float body = 1.0 - smoothstep(width*0.15, width, abs(x));
    74→    74→        // flicker vertical
    75→    75→        float flick = 0.82 + 0.18 * noise(vec2(t*3.1, uv.y*2.0));
    76→    76→        body *= flick * smoothstep(0.0, 0.10, uv.y) * (1.0 - smoothstep(0.72, 1.0, uv.y + abs(x)*0.8));
    77→    77→        if (body <= 0.004) discard;
    78→    78→        float core = smoothstep(0.12, 0.75, body) * (1.0 - uv.y*0.55);
    79→    79→        vec3 col = mix(uOuter, uCore, clamp(core, 0.0, 1.0));
    80→    80→        col *= 1.35 + 0.65 * body;
    81→    81→        gl_FragColor = vec4(col, body);
    82→    82→      }`,
    83→    83→  });
    84→    84→  flameMats.push(mat);
    85→    85→  return mat;
    86→    86→}

   157→   157→export function addRim(mat: THREE.MeshStandardMaterial, color: number, strength = 0.35, power = 3.2) {
   158→   158→  const c = new THREE.Color(color);
   159→   159→  mat.onBeforeCompile = (shader) => {
   160→   160→    shader.uniforms.uRimColor = { value: c };
   161→   161→    shader.uniforms.uRimStrength = { value: strength };
   162→   162→    shader.uniforms.uRimPower = { value: power };
   163→   163→    shader.fragmentShader = shader.fragmentShader
   164→   164→      .replace('#include <common>', `#include <common>
   165→   165→        uniform vec3 uRimColor; uniform float uRimStrength; uniform float uRimPower;`)
   166→   166→      .replace('#include <emissivemap_fragment>', `#include <emissivemap_fragment>
   167→   167→        {
   168→   168→          float rimF = pow(1.0 - saturate(dot(normalize(vViewPosition), normal)), uRimPower);
   169→   169→          totalEmissiveRadiance += uRimColor * rimF * uRimStrength;
   170→   170→        }`);
   171→   171→  };
   172→   172→  mat.customProgramCacheKey = () => `rim_${color}_${strength}_${power}`;
   173→   173→  return mat;
   174→   174→}

   186→   186→function bladeGeo(len: number, wBase: number, tipFrac = 0.22): THREE.BufferGeometry {
   187→   187→  const g = new THREE.CylinderGeometry(wBase * 0.22, wBase, len * (1 - tipFrac), 4, 1);
   188→   188→  g.translate(0, len * (1 - tipFrac) / 2, 0);
   189→   189→  const tip = new THREE.ConeGeometry(wBase * 0.22, len * tipFrac, 4);
   190→   190→  tip.rotateY(Math.PI / 4);
   191→   191→  tip.translate(0, len * (1 - tipFrac) + (len * tipFrac) / 2, 0);
   192→   192→  const merged = mergeGeometries([g.toNonIndexed(), tip.toNonIndexed()])!;
   193→   193→  merged.rotateY(Math.PI / 4);
   194→   194→  merged.scale(1, 1, 0.42);
   195→   195→  merged.computeVertexNormals();
   196→   196→  return merged;
   197→   197→}

   199→   199→export function buildSword(scale = 1): THREE.Group {
   200→   200→  const g = new THREE.Group();
   201→   201→  const steel = forgedMat(0xcdd6e2);
   202→   202→  const gold = forgedMat(0xc7a24a);
   203→   203→  const blade = mesh(bladeGeo(0.85 * scale, 0.05 * scale), steel, 0, 0.1 * scale, 0);
   204→   204→  // ranura luminosa (runa encendida)
   205→   205→  const rune = mesh(new THREE.BoxGeometry(0.012 * scale, 0.5 * scale, 0.024 * scale),
   206→   206→    emisMat(0x54e0ff, 1.5), 0, 0.5 * scale, 0, false);

   210→   210→  const pommel = mesh(new THREE.OctahedronGeometry(0.045 * scale, 0), gold, 0, -0.14 * scale, 0);

   336→   336→export function buildHumanoid(o: HumanoidOpts): HumanoidRig {
   337→   337→  const s = o.scale ?? 1;
   338→   338→  const root = new THREE.Group();
   339→   339→
   340→   340→  const mkBody = (color: number, rough: number) => {
   341→   341→    const m = stdMat(color, { roughness: rough });
   342→   342→    if (o.rim) addRim(m, o.rim.color, o.rim.strength);
   343→   343→    return m;
   344→   344→  };
   345→   345→  const skin = mkBody(o.skin, 0.68);
   346→   346→  const torsoMat = mkBody(o.torso, 0.82);
   347→   347→  const legMat = mkBody(o.legs, 0.82);
   348→   348→  const armMat = mkBody(o.arms ?? o.skin, 0.72);
   349→   349→
   350→   350→  const legLen = 0.52, torsoH = 0.55, headR = o.bigHead ? 0.21 : 0.155;
   351→   351→
   352→   352→  const body = new THREE.Group();
   353→   353→  body.position.y = legLen;
   354→   354→  root.add(body);
   355→   355→
   356→   356→  const pelvis = mesh(new THREE.BoxGeometry(0.3, 0.16, 0.2), legMat, 0, 0.04, 0);
   357→   357→  body.add(pelvis);
   358→   358→
   359→   359→  const torso = new THREE.Group();
   360→   360→  body.add(torso);
   361→   361→  const chest = mesh(new THREE.BoxGeometry(0.34, torsoH, 0.22), torsoMat, 0, torsoH / 2, 0);
   362→   362→  torso.add(chest);
   363→   363→  if (o.chest !== null && o.chest !== undefined) {
   364→   364→    torso.add(mesh(new THREE.BoxGeometry(0.37, torsoH * 0.55, 0.25), forgedMat(o.chest), 0, torsoH * 0.62, 0));
   365→   365→  }
   366→   366→  if (o.hunched) torso.rotation.x = 0.35;
   367→   367→
   368→   368→  // Capa
   369→   369→  if (o.cape !== null && o.cape !== undefined) {
   370→   370→    const capeGroup = new THREE.Group();
   371→   371→    capeGroup.position.set(0, torsoH * 0.9, -0.15);
   372→   372→    capeGroup.rotation.x = 0.14;
   373→   373→    capeGroup.rotation.y = Math.PI;
   374→   374→    const geo = new THREE.PlaneGeometry(0.5, 0.68, 4, 6);

   491→   491→    aL.hand.add(sh);
   492→   492→  }

   498→   498→    handR: aR.hand, handL: aL.hand, weapon, weaponMat: weaponMat ?? null,
   499→   499→    height: (legLen + torsoH + headR * 2.4) * s,
   500→   500→  };
   501→   501→}

   674→   674→export function grassGeometry(): THREE.BufferGeometry {
   675→   675→  const blades: THREE.BufferGeometry[] = [];
   676→   676→  const H = 0.6;
   677→   677→  for (let i = 0; i < 3; i++) {
   678→   678→    const p = new THREE.PlaneGeometry(0.2, H, 1, 3);
   679→   679→    p.translate(0, H / 2, 0);
   680→   680→    // curvatura de la hoja
   681→   681→    const pos = p.getAttribute('position') as THREE.BufferAttribute;
   682→   682→    for (let v = 0; v < pos.count; v++) {
   683→   683→      const y = pos.getY(v);
   684→   684→      const k = Math.pow(y / H, 2);
   685→   685→      pos.setZ(v, k * 0.16);
   686→   686→      pos.setX(v, pos.getX(v) * (1 - k * 0.35));
   687→   687→    }
   688→   688→    p.rotateY((i / 3) * Math.PI + 0.3);
   689→   689→    p.rotateX(((i * 37) % 10 - 5) * 0.02);
   690→   690→    blades.push(p);
   691→   691→  }
   692→   692→  const merged = mergeGeometries(blades)!;
   693→   693→  merged.computeVertexNormals();
   694→   694→  return merged;
   695→   695→}
