    20→/* Grading final: viñeta, grano fílmico, aberración cromática y color */
    21→const GradeShader = {
    22→  uniforms: {
    23→    tDiffuse: { value: null as THREE.Texture | null },
    24→    uTime: { value: 0 },
    25→    uVignette: { value: 0.44 },
    26→    uGrain: { value: 0.038 },
    27→    uCA: { value: 0.0015 },
    28→    uSat: { value: 1.08 },
    29→    uCon: { value: 1.04 },
    30→  },
    31→  vertexShader: `varying vec2 vUv; void main(){ vUv = uv; gl_Position = projectionMatrix * modelViewMatrix * vec4(position,1.0); }`,
    32→  fragmentShader: `
    33→    uniform sampler2D tDiffuse; uniform float uTime; uniform float uVignette;
    34→    uniform float uGrain; uniform float uCA; uniform float uSat; uniform float uCon;
    35→    varying vec2 vUv;
    36→    float hash(vec2 p){ return fract(sin(dot(p, vec2(12.9898,78.233))) * 43758.5453); }
    37→    void main(){
    38→      vec2 uv = vUv;
    39→      vec2 d = uv - 0.5;
    40→      float r2 = dot(d, d);
    41→      vec2 off = d * r2 * uCA * 8.0;
    42→      vec3 col;
    43→      col.r = texture2D(tDiffuse, uv + off).r;
    44→      col.g = texture2D(tDiffuse, uv).g;
    45→      col.b = texture2D(tDiffuse, uv - off).b;
    46→      float l = dot(col, vec3(0.299, 0.587, 0.114));
    47→      col = mix(vec3(l), col, uSat);
    48→      col = (col - 0.5) * uCon + 0.5;
    49→      col *= 1.0 - uVignette * smoothstep(0.12, 0.72, r2);
    50→      col += (hash(uv * vec2(1287.0, 731.0) + fract(uTime) * 43.7) - 0.5) * uGrain;
    51→      gl_FragColor = vec4(clamp(col, 0.0, 1.0), 1.0);
    52→    }`,
    53→};

   335→  private spawnRoamer(initial = false) {
   336→    if (this.enemies.filter(e => e.alive && e.guardianOf === -1 && !e.isBoss).length >= ROAMER_TARGET + 4) return;
   337→    for (let tries = 0; tries < 20; tries++) {
   338→      const a = rand(0, Math.PI * 2);
   339→      const r = rand(24, WORLD.radius - 12);
   340→      const x = Math.cos(a) * r, z = Math.sin(a) * r;
   341→      const dPlayer = Math.hypot(x - this.player.pos.x, z - this.player.pos.z);
   342→      const dBonfire = Math.hypot(x - WORLD.bonfire.x, z - WORLD.bonfire.z);
   343→      const nearShrine = WORLD.shrines.some((s, i) => !this.world.shrines[i].cleansed && Math.hypot(x - s.x, z - s.z) < 16);
   344→      if (dPlayer > (initial ? 20 : 26) && dBonfire > 16 && !nearShrine) {
   345→        // tipos según nivel
   346→        const roll = Math.random();
   347→        let type: EnemyType = 'goblin';
   348→        if (roll > 0.82 - Math.min(0.15, this.player.level * 0.01)) type = 'orc';
   349→        else if (roll > 0.5) type = 'archer';
   350→        this.spawnEnemy(type, x, z);
   351→        return;
   352→      }
   353→    }
   354→  }

   392→    for (let i = 0; i < 10; i++) {
   393→      this.particles.spawn({
   394→        x: e.pos.x + rand(-0.6, 0.6), y: e.pos.y + 0.6, z: e.pos.z + rand(-0.6, 0.6),
   395→        color: 0x6ae8ff, size: 0.26, life: 2.2, glow: 2.2, shrink: false,
   396→        seek: { getPos: () => this.player.pos, speed: 6 },
   397→      });
   398→    }
   399→    // drop de poción
   400→    if (Math.random() < cfg.potionChance) {
   401→      this.pickups.push(new Pickup(e.pos.clone(), 'potion'));
   402→      this.scene.add(this.pickups[this.pickups.length - 1].root);
   403→    }
   404→    // guardianes → santuario purificable
   405→    if (e.guardianOf >= 0) {
   406→      const alive = this.enemies.filter(o => o.guardianOf === e.guardianOf && o.alive && o !== e).length;
   407→      if (alive === 0) {
   408→        // todos los guardias muertos
   409→        void e.guardianOf;
   410→      }
   411→    }
   412→    if (e.isBoss) {
   413→      this.bossActive = false;
   414→      this.bossDefeated = true;
   415→      this.endless = true;
   416→      this.world.activateSigil();
   417→      this.world.sigilReady();
   418→      this.slowmoT = 2.2;
   419→      this.timeScale = 0.3;
   420→      this.victoryDelay = 2.0;
   421→      this.audio.setIntensity(0.2);
   422→      this.audio.cleanse();
   423→      this.ctx.shake(1);
   424→      this.particles.burst({
   425→        x: e.pos.x, y: e.pos.y + 2, z: e.pos.z,
   426→        count: 90, speed: 10, color: 0xffc84a, size: 0.5, life: 2, gravity: -1, drag: 0.8, glow: 2.6,
   427→      });
   428→    }

   431→  /* ---------- Interacción ---------- */
   432→
   433→  private updateInteract() {
   434→    this.interactTarget = null;
   435→    const p = this.player.pos;
   436→    const dB = Math.hypot(p.x - WORLD.bonfire.x, p.z - WORLD.bonfire.z);
   437→    if (dB < 3.4) {
   438→      this.interactTarget = { kind: 'bonfire' };
   439→      return;
   440→    }
   441→    for (const sh of this.world.shrines) {
   442→      if (sh.cleansed) continue;
   443→      const guardians = this.enemies.filter(e => e.guardianOf === sh.idx && e.alive);
   444→      const d = Math.hypot(p.x - sh.pos.x, p.z - sh.pos.z);
   445→      if (d < 5 && guardians.length === 0) {
   446→        this.interactTarget = { kind: 'shrine', idx: sh.idx };
   447→        return;
   448→      }
   449→    }
   450→    if (this.world.sigil && this.world.sigil.group.visible && this.bossDefeated) {
   451→      const A = WORLD.arena;
   452→      const d = Math.hypot(p.x - A.x, p.z - A.z);
   453→      if (d < 4.5) this.interactTarget = { kind: 'sigil' };
   454→    }
   455→  }

   457→  private doInteract() {
   458→    const t = this.interactTarget;
   459→    if (!t) return;
   460→    if (t.kind === 'bonfire') {
   461→      const p = this.player;
   462→      p.hp = p.maxHp; p.stamina = p.maxStamina; p.potions = p.maxPotions;
   463→      this.audio.bonfireRest();
   464→      this.particles.burst({
   465→        x: p.pos.x, y: p.pos.y + 0.6, z: p.pos.z,
   466→        count: 26, speed: 2.6, color: 0xffb347, size: 0.3, life: 1.1, gravity: -1.5, drag: 1.5, glow: 2,
   467→      });
   468→      this.addDamageNumber(new THREE.Vector3(p.pos.x, p.pos.y + 2.3, p.pos.z), 'REPOSO', '#ffb347');
   469→    } else if (t.kind === 'shrine' && t.idx !== undefined) {
   470→      const sh = this.world.shrines[t.idx];
   471→      this.world.cleanseShrine(t.idx);
   472→      this.shrinesCleansed++;
   473→      this.audio.cleanse();
   474→      this.player.gainXp(60, this.ctx);
   475→      this.player.gold += 40;

   499→    if (this.lockEnemy) { this.lockEnemy = null; this.player.lockTarget = null; return; }
   500→    let best: Enemy | null = null, bestD = 34;

   503→      const d = e.pos.distanceTo(this.player.pos);

   512→  start() {
   513→    if (this.running) return;
   514→    this.running = true;
   515→    this.lastT = performance.now();
   516→    this.loop(this.lastT);

   518→
   519→  private loop = (t: number) => {
   520→    if (!this.running) return;
   521→    this.raf = requestAnimationFrame(this.loop);
   522→    const dt = Math.min(0.05, (t - this.lastT) / 1000);
   523→    this.lastT = t;
   524→    this.fps = this.fps * 0.95 + (1 / Math.max(0.0001, dt)) * 0.05;
   525→
   526→    if (this.phase === 'menu') {
   527→      this.menuT += dt;
   528→      this.cycle.update(dt); // el mundo vive detrás del menú (con ciclo día/noche)
   529→      this.updateMenuScene(dt);
   530→      this.composer.render();
   531→      this.hudTimer -= dt;
   532→      if (this.hudTimer <= 0) { this.hudTimer = 0.3; this.emitHud(); }
   533→      return;
   534→    }
   535→
   536→    if (this.phase === 'playing') {
   537→      this.autoQuality(dt);
   538→      // ciclo día/noche (los enemigos son más rápidos de noche)
   539→      this.cycle.update(dt);
   540→      this.ctx.nightFactor = this.cycle.nightFactor;
   541→      if (this.noticeT > 0) { this.noticeT -= dt; if (this.noticeT <= 0) this.notice = ''; }
   542→      // pasos del jugador sobre el terreno
   543→      const stepDx = this.player.pos.x - this.lastStepPos.x;
   544→      const stepDz = this.player.pos.z - this.lastStepPos.z;
   545→      this.stepDist += Math.hypot(stepDx, stepDz);
   546→      this.lastStepPos.copy(this.player.pos);
   547→      if (this.stepDist > 2.2 && this.player.state !== 'roll' && this.player.state !== 'dead') {
   548→        this.stepDist = 0;
   549→        this.audio.footstep();
   550→      }
   551→      // hit-stop y cámara lenta
   552→      if (this.hitStopT > 0) {
   553→        this.hitStopT -= dt;
   554→        this.updateWorld(dt * 0.06);
   555→      } else if (this.slowmoT > 0) {
   556→        this.slowmoT -= dt;
   557→        if (this.slowmoT <= 0) this.timeScale = 1;
   558→        this.updateWorld(dt * this.timeScale);
   559→      } else {
   560→        this.updateWorld(dt);
   561→      }
   562→    } else if (this.phase === 'dead' || this.phase === 'victory') {
   563→      // el mundo sigue con animaciones reducidas
   564→      this.cycle.update(dt * 0.25);
   565→      this.updateWorld(dt * 0.25, true);
   566→    }
   567→
   568→    this.updateCamera(dt);
   569→    this.updateEffects(dt);
   570→    this.composer.render();
   571→
   572→    this.hudTimer -= dt;
   573→    if (this.hudTimer <= 0) { this.hudTimer = 0.08; this.emitHud(); this.drawMinimap(); }
   574→  };

   613→  private updateWorld(dt: number, frozen = false) {
   614→    this.elapsed += frozen ? 0 : dt;
   615→
   616→    // input → estado
   617→    const k = this.keys;
   618→    this.inputState.fwd = k.has('KeyW') || k.has('ArrowUp');
   619→    this.inputState.back = k.has('KeyS') || k.has('ArrowDown');
   620→    this.inputState.left = k.has('KeyA') || k.has('ArrowLeft');
   621→    this.inputState.right = k.has('KeyD') || k.has('ArrowRight');
   622→    this.inputState.sprint = k.has('ShiftLeft') || k.has('ShiftRight');
   623→
   624→    // jugador
   625→    this.ctx.input = frozen ? this.frozenInput : this.inputState;
   626→    this.ctx.camYaw = this.camYaw;
   627→    this.player.update(dt, this.ctx);
   628→    this.world.resolve(this.player.pos, this.player.radius);
   629→    this.worldClamp(this.player.pos, this.player.radius);
   630→
   631→    // resolver golpe pendiente del jugador
   632→    if (this.player.pendingStrike) {
   633→      this.resolvePlayerStrike(this.player.pendingStrike);
   634→      this.player.pendingStrike = null;
   635→    }
   636→
   637→    // lock-on: limpiar si muere o se aleja
   638→    if (this.lockEnemy && (!this.lockEnemy.alive || this.lockEnemy.pos.distanceTo(this.player.pos) > 38)) {
   639→      this.lockEnemy = null;
   640→      this.player.lockTarget = null;
   641→    }
   642→
   643→    // enemigos
   644→    for (const e of this.enemies) {
   645→      if (e.removable) continue;
   646→      e.update(dt, this.ctx);
   647→      if (e.alive) {
   648→        this.world.resolve(e.pos, e.radius);
   649→        this.worldClamp(e.pos, e.radius);
   650→      }
   651→    }
   652→    // separación entre enemigos
   653→    const live = this.enemies.filter(e => e.alive && e.state !== 'spawn');
   654→    for (let i = 0; i < live.length; i++) {
   655→      for (let j = i + 1; j < live.length; j++) {
   656→        const a = live[i], b = live[j];
   657→        const dx = b.pos.x - a.pos.x, dz = b.pos.z - a.pos.z;
   658→        const minD = a.radius + b.radius;
   659→        const d2 = dx * dx + dz * dz;
   660→        if (d2 < minD * minD && d2 > 1e-5) {
   661→          const d = Math.sqrt(d2);
   662→          const push = (minD - d) / d * 0.5;
   663→          a.pos.x -= dx * push; a.pos.z -= dz * push;
   664→          b.pos.x += dx * push; b.pos.z += dz * push;
   665→        }
   666→      }
   667→      // el jugador empuja y es empujado ligeramente
   668→      const p = this.player;
   669→      const dx = p.pos.x - live[i].pos.x, dz = p.pos.z - live[i].pos.z;
   670→      const minD = live[i].radius + p.radius;
   671→      const d2 = dx * dx + dz * dz;
   672→      if (d2 < minD * minD && d2 > 1e-5) {
   673→        const d = Math.sqrt(d2);
   674→        const push = (minD - d) / d * 0.6;
   675→        live[i].pos.x -= dx * push; live[i].pos.z -= dz * push;
   676→        p.pos.x += dx * push * 0.3; p.pos.z += dz * push * 0.3;
   677→      }
   678→    }
   679→    // limpiar cadáveres
   680→    if (this.enemies.some(e => e.removable)) {
   681→      for (const e of this.enemies) {
   682→        if (e.removable) { this.scene.remove(e.root); }
   683→      }
   684→      this.enemies = this.enemies.filter(e => !e.removable);
   685→    }
   686→
   687→    // proyectiles y drops
   688→    for (const pr of this.projectiles) pr.update(dt, this.ctx);
   689→    this.projectiles = this.projectiles.filter(pr => {
   690→      if (pr.dead) { this.scene.remove(pr.root); return false; }
   691→      return true;
   692→    });
   693→    for (const pk of this.pickups) {
   694→      pk.update(dt, this.ctx);
   695→      this.world.resolve(pk.pos, 0.3);
   696→    }
   697→    this.pickups = this.pickups.filter(pk => {
   698→      if (pk.dead) { this.scene.remove(pk.root); return false; }
   699→      return true;
   700→    });
   701→
   702→    // mundo visual
   703→    this.world.update(dt, this.camera, (x, y, z) => {
   704→      this.particles.spawn({
   705→        x, y, z, vx: rand(-0.3, 0.3), vy: rand(1, 2.2), vz: rand(-0.3, 0.3),
   706→        color: 0xff7a2a, size: rand(0.1, 0.2), life: rand(0.8, 1.6), glow: 2.2, drag: 0.5,
   707→      });
   708→    });
   709→
   710→    if (frozen) return;
   711→
   712→    // estela de espada
   713→    if (this.player.state === 'attack') {
   714→      const b = new THREE.Vector3(), ti = new THREE.Vector3();
   715→      this.player.swordPoints(b, ti);
   716→      this.trail.emit(b, ti);
   717→    }
   718→    this.trail.update(dt, this.player.state === 'attack');
   719→
   720→    // despawn/respawn de errantes
   721→    this.roamerTimer -= dt;
   722→    if (this.roamerTimer <= 0) {
   723→      this.roamerTimer = rand(3.5, 6);
   724→      const aliveRoamers = this.enemies.filter(e => e.alive && e.guardianOf === -1 && !e.isBoss).length;
   725→      if (aliveRoamers < ROAMER_TARGET) this.spawnRoamer();
   726→    }
   727→
   728→    // despertar del jefe
   729→    if (this.awakenTimer > 0) {
   730→      this.awakenTimer -= dt;
   731→      if (this.awakenTimer <= 0) this.spawnBoss();
   732→    }
   733→    // reaparición del jefe tras morir el jugador en plena lucha
   734→    if (this.bossRespawnT > 0) {
   735→      this.bossRespawnT -= dt;
   736→      if (this.bossRespawnT <= 0 && this.shrinesCleansed >= 3) this.spawnBoss();
   737→    }
   738→
   739→    // interacción
   740→    this.updateInteract();
   741→
   742→    // música: intensidad
   743→    if (this.elapsed % 1 < dt) {
   744→      let intensity = 0.12;
   745→      if (this.bossActive) intensity = 1;
   746→      else {
   747→        const near = this.enemies.some(e => e.alive && e.aggroed && e.pos.distanceTo(this.player.pos) < 30);
   748→        intensity = near ? 0.75 : 0.12;
   749→      }
   750→      this.audio.setIntensity(intensity);
   751→    }
   752→
   753→    // victoria
   754→    if (this.victoryDelay > 0) {
   755→      this.victoryDelay -= dt;
   756→      if (this.victoryDelay <= 0) {
   757→        this.setPhase('victory');
   758→      }
   759→    }
   760→  }

   769→  private resolvePlayerStrike(def: AttackDef) {
   770→    const p = this.player;
   771→    const heavy = def === PLAYER_HEAVY;
   772→    let hitAny = false;
   773→    for (const e of this.enemies) {
   774→      if (!e.alive || e.state === 'spawn') continue;
   775→      const to = new THREE.Vector3().subVectors(e.pos, p.pos).setY(0);
   776→      const d = to.length() - e.radius;
   777→      if (d > def.range) continue;
   778→      const ang = Math.atan2(to.x, to.z);
   779→      let diff = Math.abs(((ang - p.yaw + Math.PI * 3) % (Math.PI * 2)) - Math.PI);
   780→      if (diff <= def.arc / 2) {
   781→        const dmg = def.dmg * p.dmgMul;
   782→        e.takeDamage(dmg, p.pos, this.ctx, heavy);
   783→        hitAny = true;
   784→        this.particles.burst({
   785→          x: e.pos.x - to.x * 0.3, y: e.pos.y + e.rig.height * 0.55, z: e.pos.z - to.z * 0.3,
   786→          count: heavy ? 14 : 8, speed: 5, color: 0xffd9a0, size: 0.2, life: 0.35, drag: 2, glow: 2.4, gravity: 4,
   787→        });
   788→      }
   789→    }
   790→    if (hitAny) {
   791→      this.hitStopT = Math.max(this.hitStopT, heavy ? 0.11 : 0.07);
   792→      this.shakeAmt = Math.min(1.2, this.shakeAmt + (heavy ? 0.4 : 0.16));
   793→      if (heavy) this.audio.heavyHit();
   794→      else this.audio.hitMetal();
   795→    }
   796→  }

   804→    }

   809→  private updateCamera(dt: number) {
   810→    // ratón → rotación
   811→    if (this.locked && this.phase === 'playing') {
   812→      this.camYaw -= this.mouseDX * 0.0023;
   813→      this.camPitch += this.mouseDY * 0.0018;
   814→      this.camPitch = clamp(this.camPitch, -0.25, 1.15);
   815→    }
   816→    this.mouseDX = 0; this.mouseDY = 0;
   817→
   818→    const target = this.player.pos;
   819→    const head = new THREE.Vector3(target.x, target.y + 1.55, target.z);
   820→
   821→    if (this.lockEnemy && this.lockEnemy.alive) {
   822→      // cámara de objetivo fijado
   823→      const desired = Math.atan2(target.x - this.lockEnemy.pos.x, target.z - this.lockEnemy.pos.z);
   824→      this.camYaw = dampAngle(this.camYaw, desired, 3.2, dt);
   825→      this.camPitch = damp(this.camPitch, 0.3, 3, dt);
   826→    }
   827→
   828→    const dist = this.camDist;
   829→    const cp = Math.cos(this.camPitch), sp = Math.sin(this.camPitch);
   830→    const off = new THREE.Vector3(Math.sin(this.camYaw) * cp, sp, Math.cos(this.camYaw) * cp);
   831→    const desiredPos = head.clone().addScaledVector(off, dist);
   832→    // hombro
   833→    const right = new THREE.Vector3(-Math.cos(this.camYaw), 0, Math.sin(this.camYaw));
   834→    desiredPos.addScaledVector(right, 0.55);
   835→    // colisión con terreno
   836→    const groundH = terrainHeight(desiredPos.x, desiredPos.z) + 0.5;
   837→    if (desiredPos.y < groundH) desiredPos.y = groundH;
   838→
   839→    this.camPos.lerp(desiredPos, 1 - Math.exp(-(this.lockEnemy ? 9 : 12) * dt));
   840→
   841→    // sacudida
   842→    const sh = this.shakeAmt;
   843→    this.camera.position.copy(this.camPos);
   844→    if (sh > 0.001) {
   845→      this.camera.position.x += rand(-1, 1) * sh * 0.18;
   846→      this.camera.position.y += rand(-1, 1) * sh * 0.18;
   847→      this.camera.position.z += rand(-1, 1) * sh * 0.18;
   848→      this.shakeAmt = Math.max(0, sh - dt * 2.4);
   849→    }
   850→    const lookAt = head.clone();
   851→    if (this.lockEnemy && this.lockEnemy.alive) {
   852→      lookAt.lerp(new THREE.Vector3(this.lockEnemy.pos.x, this.lockEnemy.pos.y + 1.2, this.lockEnemy.pos.z), 0.22);
   853→    }
   854→    this.camera.lookAt(lookAt);
   855→
   856→    // FOV dinámico al esprintar
   857→    const targetFov = this.baseFov + (this.player.sprinting ? 5 : 0);
   858→    if (Math.abs(this.camera.fov - targetFov) > 0.05) {
   859→      this.camera.fov = damp(this.camera.fov, targetFov, 6, dt);
   860→      this.camera.updateProjectionMatrix();
   861→    }
   862→  }

   920→      entities,
   921→      this.world.shrines.map(s => ({ x: s.pos.x, z: s.pos.z, cleansed: s.cleansed })),
   922→      this.bossActive,

   931→    if (this.bossActive) return `Derrota a ${ENEMY_NAMES.boss}`;
   932→    if (this.shrinesCleansed >= 3) return 'El aire vibra… algo despierta en la arena del norte';

   963→  private promptText(): string {
   964→    if (!this.interactTarget || this.player.state === 'dead') return '';
   965→    switch (this.interactTarget.kind) {
   966→      case 'bonfire': return 'E · Descansar en la hoguera (cura y reabastece)';
   967→      case 'shrine': return `E · Purificar ${WORLD.shrines[this.interactTarget.idx!].name}`;
   968→      case 'sigil': return 'E · Despertar al jefe de nuevo';

   972→  private setPhase(p: GamePhase) {
   973→    const prev = this.phase;
   974→    this.phase = p;
   975→    if (p !== prev) {
   976→      if (p === 'victory') { this.audio.victory(); this.audio.duckTheme(0.2, 1.5); }
   977→      else if (p === 'dead') { this.audio.defeat(); this.audio.duckTheme(0.2, 1.5); }
   978→      else if (p === 'playing') { this.audio.duckTheme(1); }
   979→    }
   980→    this.emitHud();
   981→  }

   984→
   985→  pause() {
   986→    if (this.phase !== 'playing') return;
   987→    this.setPhase('paused');
   988→    if (document.pointerLockElement) document.exitPointerLock();
   989→  }
   990→
   991→  resume() {
   992→    if (this.phase !== 'paused') return;
   993→    this.setPhase('playing');
   994→    this.requestLock();
   995→  }
   996→
   997→  requestLock() {
   998→    const c = this.renderer.domElement;
   999→    if (document.pointerLockElement !== c) {
  1000→      try {
  1001→        // Chrome moderno devuelve Promise; sin gesto válido rechaza (NotAllowedError)
  1002→        const p = c.requestPointerLock?.() as unknown;
  1003→        if (p && typeof (p as Promise<void>).catch === 'function') {
  1004→          (p as Promise<void>).catch(() => {
  1005→            /* el jugador puede hacer clic en el lienzo para recuperar el bloqueo */
  1006→          });
  1007→        }
  1008→      } catch { /* navegadores antiguos: API síncrona, ignorar */ }
  1009→    }
  1010→  }

  1012→  respawn() {
  1013→    // limpiar enemigos vivos, proyectiles y drops
  1014→    for (const e of this.enemies) this.scene.remove(e.root);
  1015→    this.enemies = [];
  1016→    this.boss = null;
  1017→    this.bossActive = false;
  1018→    this.lockEnemy = null;
  1019→    this.player.lockTarget = null;
  1020→    for (const pr of this.projectiles) this.scene.remove(pr.root);
  1021→    this.projectiles = [];
  1022→    for (const pk of this.pickups) this.scene.remove(pk.root);
  1023→    this.pickups = [];
  1024→    // guardianes frescos en santuarios sin purificar
  1025→    this.spawnGuardians();
  1026→    for (let i = 0; i < 4; i++) this.spawnRoamer(true);
  1027→    // jugador
  1028→    const p = this.player;
  1029→    p.hp = p.maxHp; p.stamina = p.maxStamina; p.potions = Math.max(p.potions, p.maxPotions);
  1030→    p.state = 'idle'; p.stateT = 0; p.iFrames = 2;
  1031→    p.knock.set(0, 0, 0);
  1032→    p.pos.set(WORLD.bonfire.x, 0, WORLD.bonfire.z + 3.4);
  1033→    p.pos.y = terrainHeight(p.pos.x, p.pos.z);
  1034→    p.yaw = Math.atan2(WORLD.bonfire.x - p.pos.x, WORLD.bonfire.z - p.pos.z) + Math.PI;
  1035→    this.camYaw = p.yaw + Math.PI;
  1036→    if (this.shrinesCleansed >= 3 && !this.bossDefeated) this.bossRespawnT = 6;
  1037→    this.setPhase('playing');
  1038→    this.requestLock();
  1039→  }

  1066→  private onResize = () => {
  1067→    this.camera.aspect = window.innerWidth / window.innerHeight;

  1073→  private onKeyDown = (e: KeyboardEvent) => {
  1074→    if (e.code === 'Tab' || e.code === 'Space') e.preventDefault();
  1075→    if (this.phase !== 'playing') return;
  1076→    this.keys.add(e.code);
  1077→    switch (e.code) {
  1078→      case 'Tab': this.toggleLockOn(); break;
  1079→      case 'Space': this.queued.roll = true; break;
  1080→      case 'KeyF': this.queued.potion = true; break;
  1081→      case 'KeyE': this.doInteract(); break;
  1082→    }
  1083→  };
  1084→
  1085→  private onKeyUp = (e: KeyboardEvent) => {
  1086→    this.keys.delete(e.code);
  1087→  };
  1088→
  1089→  private onMouseMove = (e: MouseEvent) => {
  1090→    if (!this.locked) return;
  1091→    this.mouseDX += e.movementX;
  1092→    this.mouseDY += e.movementY;
  1093→  };
  1094→
  1095→  private onMouseDown = (e: MouseEvent) => {
  1096→    if (this.phase !== 'playing') return;
  1097→    if (!this.locked) { this.requestLock(); return; }
  1098→    if (e.button === 0) this.queued.attack = true;
  1099→    if (e.button === 2) this.queued.heavy = true;
  1100→  };
  1101→
  1102→  private onContextMenu = (e: Event) => { e.preventDefault(); };
  1103→
  1104→  private onLockChange = () => {
  1105→    this.locked = document.pointerLockElement === this.renderer.domElement;
  1106→    if (!this.locked && this.phase === 'playing') this.pause();
  1107→  };
